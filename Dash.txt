/* =========================================================
   DASHBOARD HELPERS
   ========================================================= */

function getPromoId(x) {

    const raw =
        x?.promo_id ??
        x?.promotion_id ??
        x?.promoId;

    const n = Number(raw);

    return Number.isFinite(n) ? n : null;
}


function getSubmissionId(x) {

    const raw =
        x?.incentive_id ??
        x?.submission_id ??
        x?.id;

    const n = Number(raw);

    return Number.isFinite(n) ? n : null;
}


function getType(x) {

    const direct = String(
        x?.type ??
        x?.promotion_type ??
        x?.promo_type ??
        x?.promotionType ??
        ""
    ).trim();

    const promoId = getPromoId(x);

    /*
     * Important:
     * Resolve the type from the promotion table first.
     */
    if (promoId !== null && Array.isArray(state.promotions)) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {

            const promoType = String(
                promo?.type ??
                promo?.promotion_type ??
                promo?.promo_type ??
                promo?.promotionType ??
                ""
            ).trim();

            if (promoType) {
                return promoType;
            }
        }
    }

    return direct;
}


function normalizedType(x) {

    return getType(x)
        .toLowerCase()
        .replace(/[\s_-]/g, "");
}


function isBundle(x) {

    const type = normalizedType(x);

    return (
        type === "bundle" ||
        type === "outletbundle"
    );
}


function displayType(x) {

    if (isBundle(x)) {
        return "Bundle";
    }

    return "Individual";
}


function getStatus(x) {

    return String(
        x?.status ??
        x?.submission_status ??
        ""
    ).trim().toLowerCase();
}


function getQuantity(x) {

    const quantity = Number(
        x?.quantity ??
        x?.qty ??
        1
    );

    return Number.isFinite(quantity) && quantity > 0
        ? quantity
        : 1;
}


function getRate(x) {

    const promoId = getPromoId(x);

    /*
     * Get the rate from the promotion table.
     */
    if (promoId !== null && Array.isArray(state.promotions)) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {

            const rate = Number(
                promo?.Rate ??
                promo?.rate ??
                0
            );

            if (Number.isFinite(rate)) {
                return rate;
            }
        }
    }

    /*
     * Fallback if backend already sends rate.
     */
    const rate = Number(
        x?.Rate ??
        x?.rate ??
        0
    );

    return Number.isFinite(rate) ? rate : 0;
}


function getSubmissionDate(x) {

    return String(
        x?.submission_date ??
        x?.submitted_at ??
        x?.created_at ??
        x?.date ??
        ""
    ).trim();
}


function formatMoney(amount) {

    const value = Number(amount) || 0;

    return "$" + value.toFixed(2);
}


/* =========================================================
   STAFF NAME
   ========================================================= */

function getStaffName(submission) {

    return (
        submission?.staff_name ??
        submission?.name ??
        submission?.staff ??
        submission?.SMC ??
        submission?.sales_man_code ??
        "-"
    );
}


/* =========================================================
   PROMOTION NAME
   ========================================================= */

function getPromotionName(submission) {

    const promoId = getPromoId(submission);

    if (promoId !== null && Array.isArray(state.promotions)) {

        const promotion = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promotion) {

            return (
                promotion?.promo ??
                promotion?.promotion_name ??
                promotion?.name ??
                "-"
            );
        }
    }

    return (
        submission?.promo ??
        submission?.promotion ??
        submission?.promotion_name ??
        "-"
    );
}


/* =========================================================
   DASHBOARD CALCULATION
   ========================================================= */

function calculateDashboardStats(submissions) {

    let individualAmount = 0;
    let bundleAmount = 0;

    let individualCount = 0;
    let bundleCount = 0;

    let individualQty = 0;
    let bundleQty = 0;

    let pending = 0;
    let approved = 0;
    let rejected = 0;

    submissions.forEach(submission => {

        const status = getStatus(submission);

        const quantity = getQuantity(submission);
        const rate = getRate(submission);

        const amount = quantity * rate;

        if (status === "approved") {

            approved++;

            if (isBundle(submission)) {

                bundleCount++;
                bundleQty += quantity;
                bundleAmount += amount;

            } else {

                individualCount++;
                individualQty += quantity;
                individualAmount += amount;
            }

        } else if (status === "pending") {

            pending++;

        } else if (status === "rejected") {

            rejected++;
        }
    });

    return {

        individualAmount,
        bundleAmount,

        totalAmount:
            individualAmount +
            bundleAmount,

        individualCount,
        bundleCount,

        individualQty,
        bundleQty,

        totalCount:
            individualCount +
            bundleCount,

        totalQty:
            individualQty +
            bundleQty,

        pending,
        approved,
        rejected
    };
}


/* =========================================================
   RECENT SUBMISSIONS
   ========================================================= */

function renderRecentSubmissions(submissions) {

    const tbody =
        document.getElementById(
            "dashboardRecentSubmissions"
        );

    if (!tbody) {
        return;
    }

    /*
     * Sort newest first.
     */
    const recent = [...submissions]
        .sort((a, b) => {

            const dateA =
                new Date(getSubmissionDate(a)).getTime() || 0;

            const dateB =
                new Date(getSubmissionDate(b)).getTime() || 0;

            return dateB - dateA;
        })
        .slice(0, 10);


    if (recent.length === 0) {

        tbody.innerHTML = `
            <tr>
                <td colspan="6" class="empty">
                    No submissions found.
                </td>
            </tr>
        `;

        return;
    }


    tbody.innerHTML = recent.map(submission => {

        const id =
            getSubmissionId(submission) ?? "-";

        const staff =
            getStaffName(submission);

        const promotion =
            getPromotionName(submission);

        const type =
            displayType(submission);

        const quantity =
            getQuantity(submission);

        const rate =
            getRate(submission);

        const amount =
            quantity * rate;

        const status =
            getStatus(submission);


        let statusClass =
            "status-default";

        if (status === "approved") {
            statusClass = "status-approved";
        } else if (status === "pending") {
            statusClass = "status-pending";
        } else if (status === "rejected") {
            statusClass = "status-rejected";
        }


        const statusText =
            status
                ? status.charAt(0).toUpperCase() +
                  status.slice(1)
                : "-";


        return `
            <tr>

                <td>
                    ${escapeHtml(String(id))}
                </td>

                <td>
                    ${escapeHtml(String(staff))}
                </td>

                <td>
                    ${escapeHtml(String(promotion))}
                </td>

                <td>
                    <span class="type-badge ${isBundle(submission) ? "type-bundle" : ""}">
                        ${escapeHtml(type)}
                    </span>
                </td>

                <td class="money">
                    ${formatMoney(amount)}
                </td>

                <td>
                    <span class="status ${statusClass}">
                        ${escapeHtml(statusText)}
                    </span>
                </td>

            </tr>
        `;

    }).join("");
}


/* =========================================================
   ESCAPE HTML
   ========================================================= */

function escapeHtml(value) {

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}

/* =========================================================
   MONTH FILTER
   ========================================================= */

function calculateDashboardStatsForMonth(month) {

    const submissions =
        Array.isArray(state.submissions)
            ? state.submissions
            : [];


    if (!month) {
        return submissions;
    }


    return submissions.filter(submission => {

        const date =
            getSubmissionDate(submission);

        if (!date) {
            return false;
        }

        return date.substring(0, 7) === month;
    });
}
