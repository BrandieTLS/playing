    function escapeHtml(value) {

        if (
            value === null ||
            value === undefined
        ) {
            return "";
        }

        return String(value)
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;")
            .replace(/'/g, "&#039;");
    }


    function getValue(
        obj,
        keys,
        fallback = ""
    ) {

        if (!obj) {
            return fallback;
        }

        for (const key of keys) {

            if (
                obj[key] !== undefined &&
                obj[key] !== null &&
                obj[key] !== ""
            ) {
                return obj[key];
            }

        }

        return fallback;
    }

    async function apiRequest(
        action,
        data = {}
    ) {

        try {

            const response =
                await fetch(
                    "admin.php?action=" +
                    encodeURIComponent(action),
                    {
                        method: "POST",
                        credentials: "same-origin",
                        headers: {
                            "Content-Type":
                                "application/json"
                        },
                        body:
                            JSON.stringify(data)
                    }
                );


            const text =
                await response.text();


            let result;


            try {

                result =
                    JSON.parse(text);

            } catch (error) {

                console.error(
                    "Invalid server response:",
                    text
                );

                throw new Error(
                    "Server returned an invalid response."
                );

            }


            if (
                !response.ok ||
                result.success === false
            ) {

                throw new Error(
                    result.message ||
                    "Request failed."
                );

            }


            return result;

        } catch (error) {

            console.error(error);

            showError(
                error.message
            );

            throw error;

        }

    }

        adminData.submissions
            .forEach(item => {

                const smc =
                    String(
                        getSMC(item)
                    );


                const name =
                    String(
                        getStaffName(item)
                    );


                if (
                    smc !== "-" &&
                    smc !== ""
                ) {

                    staffMap.set(
                        smc,
                        name +
                        " (" +
                        smc +
                        ")"
                    );

                }


                const outlet =
                    String(
                        getOutlet(item)
                    );


                if (
                    outlet !== "-" &&
                    outlet !== ""
                ) {

                    outletSet.add(
                        outlet
                    );

                }

            });


        staffSelect.innerHTML =
            `<option value="">All Staff</option>`;


        Array.from(
            staffMap.entries()
        )
        .sort(
            (a, b) =>
                a[1].localeCompare(b[1])
        )
        .forEach(
            ([smc, label]) => {

                const option =
                    document.createElement(
                        "option"
                    );

                option.value =
                    smc;

                option.textContent =
                    label;

                staffSelect.appendChild(
                    option
                );

            }
        );


        outletSelect.innerHTML =
            `<option value="">All Outlets</option>`;


        Array.from(
            outletSet
        )
        .sort()
        .forEach(outlet => {

            const option =
                document.createElement(
                    "option"
                );

            option.value =
                outlet;

            option.textContent =
                outlet;

            outletSelect.appendChild(
                option
            );

        });


        if (
            Array.from(
                staffSelect.options
            )
            .some(
                x =>
                    x.value ===
                    currentStaff
            )
        ) {

            staffSelect.value =
                currentStaff;

        }


        if (
            Array.from(
                outletSelect.options
            )
            .some(
                x =>
                    x.value ===
                    currentOutlet
            )
        ) {

            outletSelect.value =
                currentOutlet;

        }

    }

    function renderSubmissions() {

        const body =
            document.getElementById(
                "submissionsBody"
            );


        const search =
            document.getElementById(
                "submissionSearch"
            ).value
            .trim()
            .toLowerCase();


        const statusFilter =
            document.getElementById(
                "submissionStatus"
            ).value
            .toLowerCase();


        const typeFilter =
            document.getElementById(
                "submissionType"
            ).value
            .toLowerCase();


        const monthFilter =
            document.getElementById(
                "submissionMonth"
            ).value;


        const dayFilter =
            document.getElementById(
                "submissionDay"
            ).value;


        const staffFilter =
            document.getElementById(
                "submissionStaff"
            ).value;


        const outletFilter =
            document.getElementById(
                "submissionOutlet"
            ).value;


        const filtered =
            adminData.submissions
                .filter(item => {

                    const searchable = [

                        getStaffName(item),
                        getSMC(item),
                        getOutlet(item),
                        getPromoName(item),
                        getSubmissionId(item)

                    ]
                    .join(" ")
                    .toLowerCase();


                    const matchesSearch =
                        !search ||
                        searchable.includes(
                            search
                        );


                    const matchesStatus =
                        !statusFilter ||
                        getStatus(item) ===
                        statusFilter;


                    const matchesType =
                        !typeFilter ||
                        getType(item) ===
                        typeFilter;


                    const date =
                        normalizeDate(
                            getSubmissionDate(item)
                        );


                    const matchesMonth =
                        !monthFilter ||
                        date.startsWith(
                            monthFilter
                        );


                    const matchesDay =
                        !dayFilter ||
                        date ===
                        dayFilter;


                    const matchesStaff =
                        !staffFilter ||
                        String(
                            getSMC(item)
                        ) ===
                        String(
                            staffFilter
                        );


                    const matchesOutlet =
                        !outletFilter ||
                        String(
                            getOutlet(item)
                        ) ===
                        String(
                            outletFilter
                        );


                    return (
                        matchesSearch &&
                        matchesStatus &&
                        matchesType &&
                        matchesMonth &&
                        matchesDay &&
                        matchesStaff &&
                        matchesOutlet
                    );

                });


        if (!filtered.length) {

            body.innerHTML = `
                <tr>
                    <td
                        colspan="12"
                        class="empty-row">
                        No submissions found.
                    </td>
                </tr>
            `;

            return;

        }


        body.innerHTML =
            filtered.map(item => {

                const id =
                    Number(
                        getSubmissionId(item)
                    );


                return `

                    <tr>

                        <td>
                            ${escapeHtml(id)}
                        </td>

                        <td>
                            ${escapeHtml(
                                getStaffName(item)

    ) {

                    "editSubmissionId"
                ).value
            );

            Number(
                document.getElementById(
                    "editSubmissionPromo"
                ).value
            );

                ).value
            );

            document.getElementById(

            );

            status === "rejected" &&

            await apiRequest(
                "edit_submission",

                    incentive_id: id,

                    status: status,

                }

                "editSubmissionModal"
            );

                "Submission updated successfully."
            );

    /* =========================================================
       SAVE PROMOTION
    ========================================================== */

    async function savePromotion(
        event
    ) {

        event.preventDefault();


        const editId =
            document.getElementById(
                "editPromotionId"
            ).value;


        const name =
            document.getElementById(
                "promotionName"
            ).value
            .trim();


        const rate =
            Number(
                document.getElementById(
                    "promotionRate"
                ).value
            );


        const type =
            document.getElementById(
                "promotionType"
            ).value;


        const startDate =
            document.getElementById(
                "promotionStartDate"
            ).value;


        const endDate =
            document.getElementById(
                "promotionEndDate"
            ).value;


        if (!name) {

            alert(
                "Promotion name is required."
            );

            return;

        }


        if (
            !Number.isFinite(rate) ||
            rate < 0
        ) {

            alert(
                "Please enter a valid rate."
            );

            return;

        }


        /*
         * IMPORTANT:
         * Bundle items are NOT required individually.
         * At least ONE item is required.
         */

        const bundleItems = {};


        if (
            type === "bundle"
        ) {

            for (
                let i = 1;
                i <= 5;
                i++
            ) {

                const input =
                    document.getElementById(
                        "item_" + i
                    );


                const value =
                    input
                        ? input.value.trim()
                        : "";


                if (value !== "") {

                    bundleItems[
                        "item_" + i
                    ] = value;

                }

            }


            if (
                Object.keys(
                    bundleItems
                ).length === 0
            ) {

                alert(
                    "Please enter at least one bundle item."
                );

                return;

            }

        }


        const payload = {

            promo:
                name,

            Rate:
                rate,

            type:
                type,

            start_date:
                startDate || null,

            end_date:
                endDate || null,

            item_1:
                bundleItems.item_1 || null,

            item_2:
                bundleItems.item_2 || null,

            item_3:
                bundleItems.item_3 || null,

            item_4:
                bundleItems.item_4 || null,

            item_5:
                bundleItems.item_5 || null

        };


        try {

            let action;


            if (editId) {

                payload.promo_id =
                    Number(editId);

                action =
                    "edit_promotion";

            } else {

                action =
                    "add_promotion";

            }


            await apiRequest(
                action,
                payload
            );


            closeModal(
                "promotionModal"
            );


            alert(
                editId
                    ? "Promotion updated successfully."
                    : "Promotion added successfully."
            );


            await refreshAll();

        } catch (error) {

            console.error(error);

        }

    }

    async function saveUser(
        event
    ) {

        event.preventDefault();


        const editSMC =
            document.getElementById(
                "editUserSMC"
            ).value
            .trim();


        const name =
            document.getElementById(
                "userName"
            ).value
            .trim();


        const smc =
            document.getElementById(
                "userSMC"
            ).value
            .trim();


        const outlet =
            document.getElementById(
                "userOutlet"
            ).value
            .trim();


        const role =
            document.getElementById(
                "userRole"
            ).value;


        if (!name) {

            alert(
                "Staff name is required."
            );

            return;

        }


        if (!smc) {

            alert(
                "Sales Man Code is required."
            );

            return;

        }


        if (!outlet) {

            alert(
                "Outlet is required."
            );

            return;

        }


        const payload = {

            staff_name:
                name,

            sales_man_code:
                smc,

            outlet:
                outlet,

            role:
                role

        };


        try {

            const action =
                editSMC
                    ? "edit_user"
                    : "add_user";


            if (editSMC) {

                payload.original_smc =
                    editSMC;

            }


            await apiRequest(
                action,
                payload
            );


            closeModal(
                "userModal"
            );


            alert(
                editSMC
                    ? "Staff updated successfully."
                    : "Staff added successfully."
            );


            await refreshAll();

        } catch (error) {

            console.error(error);

        }

    }

    /* =========================================================
       MONTHLY SUMMARY
    ========================================================== */

    async function loadMonthlySummary() {

        const month =
            document.getElementById(
                "monthlySummaryMonth"
            ).value;


        if (!month) {

            document.getElementById(
                "monthlySummaryBody"
            ).innerHTML = `

                <tr>

                    <td
                        colspan="6"
                        class="loading">

                        Select a month.

                    </td>

                </tr>

            `;

            return;

        }


        const [year, monthNumber] =
            month.split("-");


        const date =
            new Date(
                Number(year),
                Number(monthNumber) - 1,
                1
            );


        const monthName =
            date.toLocaleString(
                "en-US",
                {
                    month: "long"
                }
            );


        document.getElementById(
            "monthlySummaryTitle"
        ).textContent =
            monthName +
            " " +
            year +
            " Monthly Summary";


        document.getElementById(
            "monthlySummaryBody"
        ).innerHTML = `

            <tr>

                <td
                    colspan="6"
                    class="loading">

                    Loading...

                </td>

            </tr>

        `;


        try {

            const result =
                await apiRequest(
                    "monthly_summary",
                    {
                        month:
                            month
                    }
                );


            let rows =
                Array.isArray(
                    result.data
                )
                    ? result.data
                    : [];


            /*
             * Frontend fallback:
             * if admin.php returns raw
             * approved submissions,
             * calculate the monthly
             * totals here.
             */

            if (
                rows.length &&
                (
                    rows[0].incentive_id !== undefined ||
                    rows[0].submission_id !== undefined
                )
            ) {

                rows =
                    buildMonthlySummary(
                        rows,
                        month
                    );

            }


            adminData.monthlySummary =
                rows;


            renderMonthlySummary(
                rows
            );


        } catch (error) {

            console.error(error);


            /*
             * Fallback using currently
             * loaded submissions.
             */

            const rows =
                buildMonthlySummary(
                    adminData.submissions,
                    month
                );


            renderMonthlySummary(
                rows
            );

        }

    }


    /* =========================================================
       BUILD MONTHLY SUMMARY
    ========================================================== */

    function buildMonthlySummary(
        submissions,
        selectedMonth
    ) {

        const map =
            new Map();


        submissions
            .filter(item => {

                const status =
                    getStatus(item);


                const date =
                    normalizeDate(
                        getSubmissionDate(item)
                    );


                return (
                    status === "approved" &&
                    date.startsWith(
                        selectedMonth
                    )
                );

            })
            .forEach(item => {

                const smc =
                    String(
                        getSMC(item)
                    );


                const key =
                    smc;


                if (
                    !map.has(key)
                ) {

                    map.set(
                        key,
                        {
                            staff_name:
                                getStaffName(item),

                            sales_man_code:
                                getSMC(item),

                            outlet:
                                getOutlet(item),

                            individual:
                                0,

                            bundle:
                                0,

                            total:
                                0
                        }
                    );

                }


                const row =
                    map.get(key);


                const amount =
                    getAmount(item);


                if (
                    getType(item) ===
                    "bundle"
                ) {

                    row.bundle +=
                        amount;

                } else {

                    row.individual +=
                        amount;

                }


                row.total =
                    row.individual +
                    row.bundle;

            });


        return Array.from(
            map.values()
        )
        .sort(
            (a, b) =>
                String(
                    a.staff_name
                ).localeCompare(
                    String(
                        b.staff_name
                    )
                )
        );

    }


    /* =========================================================
       RENDER MONTHLY SUMMARY
    ========================================================== */

    function renderMonthlySummary(
        rows
    ) {

        const body =
            document.getElementById(
                "monthlySummaryBody"
            );


        if (
            !rows ||
            !rows.length
        ) {

            body.innerHTML = `

                <tr>

                    <td
                        colspan="6"
                        class="empty-row">

                        No approved incentives for this month.

                    </td>

                </tr>

            `;


            document.getElementById(
                "monthlyGrandTotal"
            ).textContent =
                "$0.00";


            return;

        }


        let grandTotal = 0;


        body.innerHTML =
            rows.map(row => {

                const individual =
                    Number(
                        getValue(
                            row,
                            [
                                "individual",
                                "individual_total"
                            ],
                            0
                        )
                    );


                const bundle =
                    Number(
                        getValue(
                            row,
                            [
                                "bundle",
                                "bundle_total"
                            ],
                            0
                        )
                    );


                let total =
                    Number(
                        getValue(
                            row,
                            [
                                "total",
                                "total_amount",
                                "grand_total"
                            ],
                            individual +
                            bundle
                        )
                    );


                if (
                    !Number.isFinite(total)
                ) {

                    total =
                        individual +
                        bundle;

                }


                grandTotal +=
                    total;


                return `

                    <tr>

                        <td>
                            ${escapeHtml(
                                getValue(
                                    row,
                                    [
                                        "staff_name",
                                        "name"
                                    ],
                                    "-"
                                )
                            )}
                        </td>

                        <td>
                            ${escapeHtml(
                                getValue(
                                    row,
                                    [
                                        "sales_man_code",
                                        "SMC",
                                        "smc"
                                    ],
                                    "-"
                                )
                            )}
                        </td>

                        <td>
                            ${escapeHtml(
                                getValue(
                                    row,
                                    [
                                        "outlet",
                                        "outlet_name"
                                    ],
                                    "-"
                                )
                            )}
                        </td>

                        <td>
                            ${money(individual)}
                        </td>

                        <td>
                            ${money(bundle)}
                        </td>

                        <td>
                            <strong>
                                ${money(total)}
                            </strong>
                        </td>

                    </tr>

                `;

            }).join("");


        document.getElementById(
            "monthlyGrandTotal"
        ).textContent =
            money(grandTotal);

    }

    ========================================================== */

    function buildTop3(
        submissions,
        selectedMonth
    ) {

        const map =
            new Map();


        submissions
            .filter(item => {

                return (
                    getStatus(item) ===
                    "approved" &&

                    normalizeDate(
                        getSubmissionDate(item)
                    )
                    .startsWith(
                        selectedMonth
                    )
                );

            })
            .forEach(item => {

                const smc =
                    String(
                        getSMC(item)
                    );


                if (
                    !map.has(smc)
                ) {

                    map.set(
                        smc,
                        {
                            staff_name:
                                getStaffName(item),

                            sales_man_code:
                                getSMC(item),

                            outlet:
                                getOutlet(item),

                            total:
                                0
                        }
                    );

                }


                map.get(
                    smc
                ).total +=
                    getAmount(item);

            });


        return Array.from(
            map.values()
        )
        .sort(
            (a, b) =>
                Number(b.total) -
                Number(a.total)
        )
        .slice(0, 3);

    }
