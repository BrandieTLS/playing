/* =====================================================
   HELPER: GET FIRST AVAILABLE FIELD
===================================================== */
function getValue(item, fields, fallback = "") {
    if (!item || typeof item !== "object") {
        return fallback;
    }
    for (const field of fields) {
        if (
            Object.prototype.hasOwnProperty.call(item, field) &&
            item[field] !== null &&
            item[field] !== undefined &&
            String(item[field]).trim() !== ""
        ) {
            return item[field];
        }
    }
    return fallback;
}
/* =====================================================
   API
===================================================== */
async function apiRequest(action, data = null) {
    const url =
        "admin.php?action=" +
        encodeURIComponent(action);
    const options = {
        method: "POST",
        headers: {
            "Content-Type":"application/json",
            "Accept":"application/json"
        },
        credentials:"same-origin"
    };
    if (data !== null) {
        options.body = JSON.stringify(data);
    }
    let response;
    try {
        response = await fetch(url, options);
    } catch (error) {
        throw new Error(
            "Unable to connect to admin.php."
        );
    }
    const text = await response.text();
    let result;
    try {
        result = JSON.parse(text);
    } catch (error) {
        console.error("Raw server response:", text);
        throw new Error(
            "admin.php returned invalid JSON."
        );
    }
    if (!result.success) {
        throw new Error(
            result.message ||
            "Request failed."
        );
    }
    return result;
}

function renderSubmissions(){
    const tbody =
        document.getElementById(
            "submissionTableBody"
        );
    const search =
        document.getElementById(
            "submissionSearch"
        ).value
        .toLowerCase()
        .trim();
    const status =
        document.getElementById(
            "submissionStatusFilter"
        ).value
        .toLowerCase()
        .trim();
    const type =
        document.getElementById(
            "submissionTypeFilter"
        ).value
        .toLowerCase()
        .trim();
    let rows =
        adminData.submissions || [];
    rows =
        rows.filter(function(item){
            const text = (
                String(getStaffName(item)) +
                " " +
                String(getSMC(item)) +
                " " +
                String(getOutlet(item)) +
                " " +
                String(getPromotionName(item))
            ).toLowerCase();
            const matchesSearch =
                !search ||
                text.includes(search);
            const matchesStatus =
                !status ||
                getStatus(item) === status;
            const matchesType =
                !type ||
                getType(item) === type;
            return (
                matchesSearch &&
                matchesStatus &&
                matchesType
            );
        });
    if(!rows.length){
        tbody.innerHTML = `
            <tr>
                <td colspan="12" class="empty">
                    No submissions found.
                </td>
            </tr>
        `;
        return;
    }
    tbody.innerHTML =
        rows.map(function(item){
            const id =
                getSubmissionId(item);
            const typeValue =
                getType(item);
            const typeClass =
                typeValue === "outlet_bundle"
                    ? "type-bundle"
                    : "";
            const status =
                getStatus(item);
            const actionButtons =
                status === "pending"
                ? `
                    <button
                        class="btn btn-success btn-small"
                        onclick="approveSubmission(${Number(id)})">
                        Approve
                    </button>
                    <button
                        class="btn btn-danger btn-small"
                        onclick="rejectSubmission(${Number(id)})">
                        Reject
                    </button>
                `
                : "";
            return `
                <tr>
                    <td>
                        ${escapeHtml(id)}
                    </td>
                    <td>
                        ${escapeHtml(
                            getStaffName(item)
                        )}
                    </td>
                    <td>
                        ${escapeHtml(
                            getSMC(item)
                        )}
                    </td>
                    <td>
                        ${escapeHtml(
                            getOutlet(item)
                        )}
                    </td>
                    <td>
                        ${escapeHtml(
                            getPromotionName(item)
                        )}
                    </td>

                                formatType(typeValue)
                            )}
                        </span>
                    </td>

                        ${getQuantity(item)}
                    </td>

                        ${money(getRate(item))}
                    </td>

                    </td>

            }

}
/* =====================================================
   REJECT
===================================================== */
async function rejectSubmission(id){
    const reason =
        prompt(
            "Enter rejection reason:"
        );
    if(reason === null){
        return;
    }
    if(!reason.trim()){
        showMessage(
            "Rejection reason is required.",
            "error"
        );
        return;
    }
    try{
        await apiRequest(
            "reject",
            {
                incentive_id:Number(id),
                rejection_reason:reason.trim()
            }
        );
        showMessage(
            "Submission rejected.",
            "success"
        );
        await loadAdminData();
    }catch(error){
        showMessage(
            error.message,
            "error"
        );
    }
}

async function savePromotion(event){
    event.preventDefault();
    const id =
        document.getElementById(
            "promotionId"
        ).value.trim();
    const promo =
        document.getElementById(
            "promotionName"
        ).value.trim();
    const rate =
        Number(
            document.getElementById(
                "promotionRate"
            ).value
        );
    const type =
        document.getElementById(
            "promotionType"
        ).value;
    const startDate =
        document.getElementById(
            "promotionStartDate"
        ).value;
    const endDate =
        document.getElementById(
            "promotionEndDate"
        ).value;
    if(!promo){
        showMessage(
            "Promotion name is required.",
            "error"
        );
        return;
    }
    if(!Number.isFinite(rate) || rate < 0){
        showMessage(
            "Please enter a valid rate.",
            "error"
        );
        return;
    }
    if(!startDate || !endDate){
        showMessage(
            "Start date and end date are required.",
            "error"
        );
        return;
    }
    if(endDate < startDate){
        showMessage(
            "End date cannot be before start date.",
            "error"
        );
        return;
    }
    const data = {
        promo:promo,
        Rate:rate,
        type:type,
        start_date:startDate,
        end_date:endDate,
        item_1:
            document.getElementById(
                "bundleItem1"
            ).value.trim(),
        item_2:
            document.getElementById(
                "bundleItem2"
            ).value.trim(),
        item_3:
            document.getElementById(
                "bundleItem3"
            ).value.trim(),
        item_4:
            document.getElementById(
                "bundleItem4"
            ).value.trim(),
        item_5:
            document.getElementById(
                "bundleItem5"
            ).value.trim()
    };
    try{
        if(id){
            data.promo_id =
                Number(id);
            await apiRequest(
                "edit_promotion",
                data
            );
            showMessage(
                "Promotion updated.",
                "success"
            );
        }else{
            await apiRequest(
                "add_promotion",
                data
            );
            showMessage(
                "Promotion added.",
                "success"
            );
        }
        closeModal(
            "promotionModal"
        );
        await loadAdminData();
    }catch(error){
        showMessage(
            error.message,
            "error"
        );
    }
}

            money(rate);
        if(
            selectedId !== null &&
            Number(id) ===
            Number(selectedId)
        ){
            option.selected =
                true;
        }
        select.appendChild(
            option
        );
    });
}
/* =====================================================
   MONTH FILTER
===================================================== */
function populateMonthlyFilter(){
    const select =
        document.getElementById(
            "monthlyFilter"
        );
    const oldValue =
        select.value;
    const months =
        new Set();
    (adminData.monthlySummary || [])
    .forEach(function(item){
        const month =
            getValue(
                item,
                [
                    "month",
                    "month_name"
                ],
                ""
            );
        if(month){
            months.add(
                String(month)
            );
        }
    });
    select.innerHTML = `
        <option value="">
            All Months
        </option>
    `;
    Array.from(months)
        .sort()
        .reverse()
        .forEach(function(month){
            const option =
                document.createElement(
                    "option"
                );
            option.value =
                month;
            option.textContent =
                month;
            select.appendChild(
                option
            );
        });
    if(months.has(oldValue)){
        select.value =
            oldValue;
    }
}

/* =====================================================
   LOGOUT
===================================================== */
async function logout(){
    if(!confirm(
        "Are you sure you want to logout?"
    )){
        return;
    }
    try{
        const result =
            await apiRequest(
                "logout"
            );
        if(result.redirect){
            window.location.href =
                result.redirect;
        }else{
            window.location.href =
                "index.html";
        }
    }catch(error){
        window.location.href =
            "index.html";
    }
}

/* =====================================================
   STATUS
===================================================== */
function statusBadge(status){
    const value =
        String(status || "")
        .toLowerCase()
        .trim();
    let className =
        "status-default";
    if(
        value === "approved" ||
        value === "active"
    ){
        className =
            "status-approved";
    }else if(
        value === "pending" ||
        value === "upcoming"
    ){
        className =
            "status-pending";
    }else if(
        value === "rejected" ||
        value === "expired"
    ){
        className =
            "status-rejected";
    }
    return `
        <span class="status ${className}">
            ${escapeHtml(
                status || "-"
            )}
        </span>
    `;
}
function promotionStatusBadge(status){
    return statusBadge(status);
}
/* =====================================================
   DETAIL BOX
===================================================== */
function detailBox(label,value){
    return `
        <div class="detail-box">
            <div class="detail-label">
                ${escapeHtml(label)}
            </div>
            <div class="detail-value">
                ${escapeHtml(
                    value === null ||
                    value === undefined ||
                    value === ""
                        ? "-"
                        : value
                )}
            </div>
        </div>
    `;
}
/* =====================================================
   FORMAT TYPE
===================================================== */
function formatType(type){
    const value =
        String(type || "")
        .toLowerCase()
        .trim();
    if(value === "outlet_bundle"){
        return "Outlet Bundle";
    }
    if(value === "individual"){
        return "Individual";
    }
    return type || "-";
}
/* =====================================================
   MONEY
===================================================== */
function money(value){
    const number =
        Number(value) || 0;
    return "$" +
        number.toLocaleString(
            "en-SG",
            {
                minimumFractionDigits:2,
                maximumFractionDigits:2
            }
        );
}

        document.getElementById(
            "message"
        );
    element.className =
        "message";
}
/* =====================================================
   ESCAPE
===================================================== */
function escapeHtml(value){
    return String(
        value === null ||
        value === undefined
            ? ""
            : value
    )
    .replace(/&/g,"&amp;")
    .replace(/</g,"&lt;")
    .replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;")
    .replace(/'/g,"&#039;");
}
function escapeAttribute(value){
    return escapeHtml(value);
}
