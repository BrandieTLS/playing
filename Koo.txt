/* =====================================================
   API
===================================================== */

async function apiRequest(
    action,
    data = null
) {

    const url =
        "admin.php?action=" +
        encodeURIComponent(action);


    const options = {

        method: "POST",

        headers: {

            "Content-Type":
                "application/json",

            "Accept":
                "application/json"

        },

        credentials:
            "same-origin"

    };


    if (data !== null) {

        options.body =
            JSON.stringify(data);

    }


    let response;


    try {

        response =
            await fetch(
                url,
                options
            );

    } catch (error) {

        throw new Error(
            "Unable to connect to admin.php."
        );

    }


    const text =
        await response.text();


    let result;


    try {

        result =
            JSON.parse(text);

    } catch (error) {

        console.error(
            "Raw server response:",
            text
        );

        throw new Error(
            "admin.php returned invalid JSON."
        );

    }


    if (!result.success) {

        throw new Error(
            result.message ||
            "Request failed."
        );

    }


    return result;

}

   LOAD DATA

}

/* =====================================================

===================================================== */

function renderSubmissions() {

    const tbody =
        document.getElementById(
            "submissionTableBody"
        );


    const search =
        document.getElementById(
            "submissionSearch"
        ).value
        .toLowerCase()
        .trim();


    const status =
        document.getElementById(
            "submissionStatusFilter"
        ).value
        .toLowerCase();


    const type =
        document.getElementById(
            "submissionTypeFilter"
        ).value
        .toLowerCase();


    let rows =
        adminData.submissions || [];


    rows =
        rows.filter(function (item) {

            const text = (

                item.staff_name ||
                "" +
                " " +
                (
                    item.sales_man_code ||
                    item.SMC ||
                    ""
                ) +
                " " +
                (
                    item.outlet ||
                    ""
                ) +
                " " +
                (
                    item.promo ||
                    item.promotion ||
                    ""
                )

            ).toLowerCase();


            const matchesSearch =
                !search ||
                text.includes(search);


            const matchesStatus =
                !status ||
                String(
                    item.status ||
                    ""
                ).toLowerCase() === status;


            const matchesType =
                !type ||
                String(
                    item.type ||
                    item.promotion_type ||
                    ""
                ).toLowerCase() === type;


            return (
                matchesSearch &&
                matchesStatus &&
                matchesType
            );

        });


    if (!rows.length) {

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="12"
                    class="empty"
                >
                    No submissions found.
                </td>
            </tr>
        `;

        return;

    }


    tbody.innerHTML =
        rows.map(function (item) {

            const id =
                item.incentive_id ??
                item.id;


            const quantity =
                Number(
                    item.quantity || 0
                );


            const rate =
                Number(
                    item.Rate ??
                    item.rate ??
                    0
                );


            const amount =
                quantity * rate;


            const typeValue =
                item.type ||
                item.promotion_type ||
                "";

        document.getElementById(
            "promotionTypeFilter"

        adminData.promotions || [];


    rows =
        rows.filter(function (item) {

            const name =
                String(
                    item.promo ||

                ).toLowerCase();

                !search ||
                name.includes(search);


            const matchesStatus =
                !status ||
                item.status === status;

                !type ||
                item.type === type;

                matchesSearch &&
                matchesStatus &&
                matchesType

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="9"

                items.length

                            return `
                                <span class="item-tag">

                                </span>
                            `;

                        }).join("")}

                        ${escapeHtml(

                            item.id ??
                            ""
                        )}
                    </td>

                    <td>
                        <strong>

                                item.promo ||
                                ""
                            )}
                        </strong>

                                item.type ||
                                ""
                            )}
                        </span>

                    </td>

                    <td>
                        ${bundleHtml}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.start_date ||

                        )}

===================================================== */

async function savePromotion(event) {

    event.preventDefault();


    const id =
        document.getElementById(
            "promotionId"
        ).value.trim();


    const promo =
        document.getElementById(
            "promotionName"
        ).value.trim();


    const rate =
        Number(
            document.getElementById(
                "promotionRate"
            ).value
        );


    const type =
        document.getElementById(
            "promotionType"
        ).value;


    const startDate =
        document.getElementById(
            "promotionStartDate"
        ).value;


    const endDate =
        document.getElementById(
            "promotionEndDate"
        ).value;


    if (!promo) {

        showMessage(
            "Promotion name is required.",
            "error"
        );

        return;

    }


    if (
        !Number.isFinite(rate) ||
        rate < 0
    ) {

        showMessage(
            "Please enter a valid rate.",
            "error"
        );

        return;

    }


    if (!startDate ||
        !endDate) {

        showMessage(
            "Start date and end date are required.",
            "error"
        );

        return;

    }


    if (
        endDate < startDate
    ) {

        showMessage(
            "End date cannot be before start date.",
            "error"
        );

        return;

    }


    const data = {

        promo: promo,

        Rate: rate,

        type: type,

        start_date: startDate,

        end_date: endDate,

        item_1:
            document.getElementById(
                "bundleItem1"
            ).value.trim(),

        item_2:
            document.getElementById(
                "bundleItem2"
            ).value.trim(),

        item_3:
            document.getElementById(
                "bundleItem3"
            ).value.trim(),

        item_4:
            document.getElementById(
                "bundleItem4"
            ).value.trim(),

        item_5:
            document.getElementById(
                "bundleItem5"
            ).value.trim()

    };


    try {

        if (id) {

            data.promo_id =
                Number(id);


            await apiRequest(
                "edit_promotion",
                data
            );


            showMessage(
                "Promotion updated.",
                "success"
            );

        } else {

            await apiRequest(
                "add_promotion",
                data
            );


            showMessage(
                "Promotion added.",
                "success"
            );

        }


        closeModal(
            "promotionModal"
        );


        await loadAdminData();


    } catch (error) {

        showMessage(
            error.message,
            "error"
        );

    }

}

function renderMonthlySummary() {

    const tbody =
        document.getElementById(
            "monthlyTableBody"
        );


    const selected =
        document.getElementById(
            "monthlyFilter"
        ).value;


    let rows =
        adminData.monthlySummary || [];


    if (selected) {

        rows =
            rows.filter(function (item) {

                return String(
                    item.month ||
                    ""
                ) === selected;

            });

    }


    let individualTotal = 0;

    let bundleTotal = 0;


    rows.forEach(function (item) {

        individualTotal += Number(
            item.individualTotal ??
            item.individual_total ??
            0
        );


        bundleTotal += Number(
            item.bundleTotal ??
            item.bundle_total ??
            0
        );

    });


    document.getElementById(
        "monthlyIndividualTotal"
    ).textContent =
        money(individualTotal);


    document.getElementById(
        "monthlyBundleTotal"
    ).textContent =
        money(bundleTotal);


    document.getElementById(
        "monthlyOverallTotal"
    ).textContent =
        money(
            individualTotal +
            bundleTotal
        );


    if (!rows.length) {

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="7"
                    class="empty"
                >
                    No monthly summary available.
                </td>
            </tr>
        `;

        return;

    }


    tbody.innerHTML =
        rows.map(function (item) {

            const individual =
                Number(
                    item.individualTotal ??
                    item.individual_total ??
                    0
                );


            const bundle =
                Number(
                    item.bundleTotal ??
                    item.bundle_total ??
                    0
                );


            const total =
                Number(
                    item.total ??
                    item.totalIncentive ??
                    item.total_incentive ??
                    (
                        individual +
                        bundle
                    )
                );


            return `

                <tr>

                    <td>
                        ${escapeHtml(
                            item.month ||
                            ""
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.staff_name ||
                            ""
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.sales_man_code ||
                            item.SMC ||
                            ""
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.outlet ||
                            ""
                        )}
                    </td>

                    <td class="money">
                        ${money(individual)}
                    </td>

                    <td class="money">
                        ${money(bundle)}
                    </td>

                    <td class="money">
                        ${money(total)}
                    </td>

                </tr>

            `;

        }).join("");

        top.top3 ||
        [];


    renderTopCard(
        1,
        rows[0]
    );


    renderTopCard(
        2,
        rows[1]
    );


    renderTopCard(
        3,
        rows[2]
    );


    const tbody =
        document.getElementById(
            "top3TableBody"
        );


    if (!rows.length) {

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="7"
                    class="empty"
                >
                    No Top 3 data available.
                </td>
            </tr>
        `;

        return;

    }


    tbody.innerHTML =
        rows.map(function (item, index) {

            const individual =
                Number(
                    item.individualTotal ??
                    item.individual_total ??
                    0
                );


            const bundle =
                Number(
                    item.bundleShare ??
                    item.bundle_share ??
                    0
                );


            const total =
                Number(
                    item.total ??
                    item.totalIncentive ??
                    item.total_incentive ??
                    individual +
                    bundle
                );


            return `

                <tr>

                    <td>
                        ${index + 1}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.staff_name ||
                            item.name ||
                            ""
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.sales_man_code ||
                            item.SMC ||
                            ""
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.outlet ||
                            ""
                        )}
                    </td>

                    <td class="money">
                        ${money(individual)}
                    </td>

                    <td class="money">
                        ${money(bundle)}
                    </td>

                    <td class="money">
                        ${money(total)}
                    </td>

                </tr>

            `;

        }).join("");

}


function renderTopCard(
    rank,
    item
) {

    const safe =
        item || {};


    const individual =
        Number(
            safe.individualTotal ??
            safe.individual_total ??
            0
        );


    const bundle =
        Number(
            safe.bundleShare ??
            safe.bundle_share ??
            0
        );


    const total =
        Number(
            safe.total ??
            safe.totalIncentive ??
            safe.total_incentive ??
            individual +
            bundle
        );


    document.getElementById(
        `top${rank}Name`
    ).textContent =
        safe.staff_name ||
        safe.name ||
        "-";


    document.getElementById(
        `top${rank}Meta`
    ).textContent =

        safe.sales_man_code ||
        safe.SMC ||

        safe.outlet
            ? (
                (
                    safe.sales_man_code ||
                    safe.SMC ||
                    ""

                (
                    safe.outlet
                        ? " • " +
                          safe.outlet
                        : ""
                )

        `top${rank}Individual`

        `top${rank}Bundle`
    ).textContent =
        money(bundle);

        `top${rank}Total`

}

function populatePromotionDropdown(

        </option>`;


    (adminData.promotions || [])

            promo.promo_id ??

        const option =
            document.createElement(

            );

            );


        if (
            selectedId !== null &&
            Number(id) ===
            Number(selectedId)
        ) {

            option.selected =
                true;

        }


        select.appendChild(
            option
        );

    });

}


/* =====================================================
   MONTH FILTER
===================================================== */

function populateMonthlyFilter() {

    const select =
        document.getElementById(
            "monthlyFilter"
        );


    const oldValue =
        select.value;


    const months =
        new Set();


    (adminData.monthlySummary || [])
    .forEach(function (item) {

        if (item.month) {

            months.add(
                item.month
            );

        }

    });


    select.innerHTML =
        `<option value="">
            All Months
        </option>`;


    Array.from(months)
        .sort()
        .reverse()
        .forEach(function (month) {

            const option =
                document.createElement(
                    "option"
                );


            option.value =
                month;


            option.textContent =
                month;


            select.appendChild(
                option
            );

        });


    if (
        months.has(oldValue)
    ) {

        select.value =
            oldValue;

    }

}

/* =====================================================
   LOGOUT
===================================================== */

async function logout() {

    if (!confirm(
        "Are you sure you want to logout?"
    )) {

        return;

    }


    try {

        const result =
            await apiRequest(
                "logout"
            );


        if (result.redirect) {

            window.location.href =
                result.redirect;

        } else {

            window.location.href =
                "index.html";

        }


    } catch (error) {

        window.location.href =
            "index.html";

    }

}

/* =====================================================
   STATUS
===================================================== */

function statusBadge(status) {

    const value =
        String(
            status || ""
        ).toLowerCase();


    let className =
        "status-default";


    if (
        value === "approved" ||
        value === "active"
    ) {

        className =
            "status-approved";

    } else if (
        value === "pending" ||
        value === "upcoming"
    ) {

        className =
            "status-pending";

    } else if (
        value === "rejected" ||
        value === "expired"
    ) {

        className =
            "status-rejected";

    }


    return `

        <span
            class="status ${className}"
        >
            ${escapeHtml(
                status ||
                ""
            )}
        </span>

    `;

}


function promotionStatusBadge(status) {

    return statusBadge(
        status
    );

}

   DETAIL BOX

function detailBox(
    label,
    value
) {

    return `

        <div class="detail-box">

            <div class="detail-label">
                ${escapeHtml(label)}
            </div>

            <div class="detail-value">
                ${escapeHtml(
                    value === null ||
                    value === undefined ||
                    value === ""
                        ? "-"
                        : value
                )}
            </div>

        </div>

    `;

}


/* =====================================================
   MONEY
===================================================== */

function money(value) {

    const number =
        Number(value) || 0;


    return "$" +
        number.toLocaleString(
            "en-SG",
            {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            }
        );

}

function showMessage(
    message,
    type = "info"
) {

    const element =
        document.getElementById(
            "message"
        );


    element.textContent =
        message;


    element.className =
        "message show " +
        type;


    if (
        type !== "info"
    ) {

        setTimeout(
            function () {

                hideMessage();

            },
            3500
        );

    }

}


function hideMessage() {

    const element =
        document.getElementById(
            "message"
        );


    element.className =
        "message";

}


/* =====================================================
   ESCAPE
===================================================== */

function escapeHtml(value) {

    return String(
        value === null ||
        value === undefined
            ? ""
            : value
    )
    .replace(
        /&/g,
        "&amp;"
    )
    .replace(
        /</g,
        "&lt;"
    )
    .replace(
        />/g,
        "&gt;"
    )
    .replace(
        /"/g,
        "&quot;"
    )
    .replace(
        /'/g,
        "&#039;"
    );

}


function escapeAttribute(value) {

    return escapeHtml(
        value
    );

}
