<script>
    let allSubmissions = [];
    let promotions = [];
    let users = [];

    let selectedMonth = getCurrentMonth();

    let currentFilteredSubmissions = [];

    let currentSubmissionId = null;
    let editingPromotionId = null;


    /* ========================================================= */
    /* HELPERS */
    /* ========================================================= */

    function getCurrentMonth() {
        const d = new Date();

        return d.getFullYear() +
            "-" +
            String(d.getMonth() + 1).padStart(2, "0");
    }


    function money(value) {
        return "S$" + Number(value || 0).toFixed(2);
    }


    function escapeHtml(value) {
        return String(value ?? "")
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;")
            .replace(/'/g, "&#039;");
    }


    function formatDate(value) {

        if (!value) {
            return "-";
        }

        const date = new Date(value);

        if (isNaN(date.getTime())) {
            return value;
        }

        return date.toLocaleDateString("en-SG");
    }


    function promotionStatus(p) {

        const today =
            new Date().toISOString().slice(0, 10);

        if (!p.start_date || !p.end_date) {
            return p.status || "Active";
        }

        if (p.end_date < today) {
            return "Expired";
        }

        if (p.start_date > today) {
            return "Upcoming";
        }

        return "Active";
    }


    function statusBadge(status) {

        const value = String(status || "").toLowerCase();

        let cls = "status-default";

        if (value === "approved") {
            cls = "status-approved";
        }

        if (value === "pending") {
            cls = "status-pending";
        }

        if (value === "rejected") {
            cls = "status-rejected";
        }

        if (value === "active") {
            cls = "status-active";
        }

        if (value === "upcoming") {
            cls = "status-upcoming";
        }

        if (value === "expired") {
            cls = "status-expired";
        }

        return `
            <span class="status ${cls}">
                ${escapeHtml(status || "-")}
            </span>
        `;
    }


    function typeBadge(type) {

        if (type === "outlet_bundle") {

            return `
                <span class="type-badge type-bundle">
                    Outlet Bundle
                </span>
            `;
        }

        return `
            <span class="type-badge">
                Individual
            </span>
        `;
    }


    function closeModal(id) {
        document.getElementById(id).classList.remove("show");
    }


    function showModal(id) {
        document.getElementById(id).classList.add("show");
    }


    function showMessage(message, type = "success") {

        const el = document.getElementById("message");

        el.textContent = message;

        el.className =
            "message show " + type;

        setTimeout(() => {
            el.classList.remove("show");

    }


    /* ========================================================= */
    /* API */
    /* ========================================================= */

    async function api(action, options = {}) {

        let url =
            "admin.php?action=" +
            encodeURIComponent(action);

        if (options.params) {

            const params =
                new URLSearchParams(options.params);

            url += "&" + params.toString();
        }

        const fetchOptions = {
            method: options.method || "GET",
            headers: {
                "Accept": "application/json"
            }
        };

        if (options.method === "POST") {

            fetchOptions.headers["Content-Type"] =
                "application/json";

            fetchOptions.body =
                JSON.stringify(options.body || {});
        }

        const response =
            await fetch(url, fetchOptions);

        const data =
            await response.json();

        if (!response.ok || data.success === false) {

            throw new Error(
                data.message || "Request failed."
            );
        }

        return data;
    }

        } catch (error) {

            showMessage(
                error.message ||
                "Unable to load data.",
                "error"
            );
        }
    }

    function renderDashboard(dashboard) {

        document.getElementById("totalSubmissions").textContent =
            dashboard.totalSubmissions || 0;

        document.getElementById("pendingSubmissions").textContent =
            dashboard.pending || 0;

        document.getElementById("approvedSubmissions").textContent =
            dashboard.approved || 0;

        document.getElementById("rejectedSubmissions").textContent =
            dashboard.rejected || 0;

        document.getElementById("approvedIndividual").textContent =
            money(dashboard.approvedIndividual || 0);

        document.getElementById("bundlePool").textContent =
            money(dashboard.bundlePool || 0);

        document.getElementById("staffCount").textContent =
            dashboard.staffCount || 0;


        const body =
            document.getElementById("dashboardRecentBody");

        const rows =
            allSubmissions
                .filter(s => {
                    return String(s.submission_date || "")
                        .slice(0, 7) === selectedMonth;
                })
                .slice(0, 10);

        if (!rows.length) {

            body.innerHTML = `
                <tr>
                    <td colspan="10" class="empty">
                        No submissions for this month.
                    </td>
                </tr>
            `;

            return;
        }


        body.innerHTML =
            rows.map(s => {

                const total =
                    Number(s.quantity || 0) *
                    Number(s.Rate || 0);

                return `
                    <tr>
                        <td>${escapeHtml(s.incentive_id)}</td>
                        <td>${escapeHtml(s.SMC)}</td>
                        <td>${escapeHtml(s.staff_name)}</td>
                        <td>${escapeHtml(s.outlet)}</td>
                        <td>${escapeHtml(s.promo)}</td>
                        <td>${typeBadge(s.type)}</td>
                        <td>${money(s.Rate)}</td>
                        <td>${escapeHtml(s.quantity)}</td>
                        <td class="money">${money(total)}</td>
                        <td>${statusBadge(s.status)}</td>
                        <td>${formatDate(s.submission_date)}</td>
                    </tr>
                `;

            }).join("");
    }

    function applySubmissionFilters() {

        const search =
            document.getElementById("filterSearch")
                .value
                .trim()
                .toLowerCase();

        const status =
            document.getElementById("filterStatus").value;

        const promotion =
            document.getElementById("filterPromotion").value;

        const smc =
            document.getElementById("filterSMC").value;

        const outlet =
            document.getElementById("filterOutlet").value;

        const from =
            document.getElementById("filterFrom").value;

        const to =
            document.getElementById("filterTo").value;


        currentFilteredSubmissions =
            allSubmissions.filter(s => {

                const total =
                    Number(s.quantity || 0) *
                    Number(s.Rate || 0);

                const searchable =
                    [
                        s.incentive_id,
                        s.SMC,
                        s.staff_name,
                        s.outlet,
                        s.promo,
                        s.type,
                        s.status,
                        total
                    ]
                    .join(" ")
                    .toLowerCase();


                if (
                    search &&
                    !searchable.includes(search)
                ) {
                    return false;
                }


                if (
                    status &&
                    String(s.status).toLowerCase() !==
                    status.toLowerCase()
                ) {
                    return false;
                }


                if (
                    promotion &&
                    String(s.promo_id) !==
                    String(promotion)
                ) {
                    return false;
                }


                if (
                    smc &&
                    String(s.SMC) !==
                    String(smc)
                ) {
                    return false;
                }


                if (
                    outlet &&
                    String(s.outlet) !==
                    String(outlet)
                ) {
                    return false;
                }


                const date =
                    String(s.submission_date || "")
                        .slice(0, 10);


                if (from && date < from) {
                    return false;
                }

                if (to && date > to) {
                    return false;
                }


                return true;

            });


        renderFilteredSubmissions();
    }


    function clearSubmissionFilters() {

        document.getElementById("filterSearch").value = "";
        document.getElementById("filterStatus").value = "";
        document.getElementById("filterPromotion").value = "";
        document.getElementById("filterSMC").value = "";
        document.getElementById("filterOutlet").value = "";
        document.getElementById("filterFrom").value = "";
        document.getElementById("filterTo").value = "";

        applySubmissionFilters();
    }

    async function approveCurrentSubmission() {

        if (!currentSubmissionId) {
            return;
        }


        if (!confirm("Approve this submission?")) {
            return;
        }


        try {

            await api("approve_submission", {
                method: "POST",
                body: {
                    incentive_id: currentSubmissionId
                }
            });


            closeModal("submissionModal");

            showMessage(
                "Submission approved.",
                "success"
            );

            await loadData();

        } catch (error) {

            showMessage(
                error.message,
                "error"
            );
        }
    }

    function renderTop3Breakdown(rows) {

        const body =
            document.getElementById("top3BreakdownBody");


        if (!rows.length) {

            body.innerHTML = `
                <tr>
                    <td colspan="8" class="empty">
                        No approved individual incentive
                        data for this month.
                    </td>
                </tr>
            `;

            return;
        }


        let html = "";

        let currentRank = null;
        let currentTotal = 0;


        rows.forEach((row, index) => {

            const rank =
                Number(row.rank);

            if (
                currentRank !== null &&
                currentRank !== rank
            ) {

                html += `
                    <tr class="breakdown-total">

                        <td colspan="7">
                            ${ordinal(currentRank)}
                            Place Total
                        </td>

                        <td>
                            ${money(currentTotal)}
                        </td>

                    </tr>
                `;

                currentTotal = 0;
            }


            currentRank = rank;

            currentTotal +=
                Number(row.incentive || 0);


            html += `
                <tr>

                    <td>
                        ${ordinal(rank)}
                    </td>

                    <td>
                        ${escapeHtml(row.staff_name)}
                    </td>

                    <td>
                        ${escapeHtml(row.SMC)}
                    </td>

                    <td>
                        ${escapeHtml(row.promo)}
                    </td>

                    <td>
                        ${typeBadge(row.type)}
                    </td>

                    <td>
                        ${money(row.Rate)}
                    </td>

                    <td>
                        ${escapeHtml(row.quantity)}
                    </td>

                    <td class="money">
                        ${money(row.incentive)}
                    </td>

                </tr>
            `;


            if (index === rows.length - 1) {

                html += `
                    <tr class="breakdown-total">

                        <td colspan="7">
                            ${ordinal(currentRank)}
                            Place Total
                        </td>

                        <td>
                            ${money(currentTotal)}
                        </td>

                    </tr>
                `;
            }

        });


        body.innerHTML = html;
    }


    function ordinal(rank) {

        if (rank === 1) {
            return "1st";
        }

        if (rank === 2) {
            return "2nd";
        }

        if (rank === 3) {
            return "3rd";
        }

        return rank + "th";
    }

    function toggleBundleFields() {

        const type =
            document.getElementById("promoType").value;


        const container =
            document.getElementById("bundleItemsContainer");


        if (type === "outlet_bundle") {

            container.classList.remove("hidden");

            for (let i = 1; i <= 5; i++) {

                document.getElementById("item" + i)
                    .disabled = false;
            }

        } else {

            container.classList.add("hidden");

            for (let i = 1; i <= 5; i++) {

                document.getElementById("item" + i)
                    .disabled = true;

                document.getElementById("item" + i)
                    .value = "";
            }
        }
    }


    function updatePromotionStatusPreview() {

        const start =
            document.getElementById("promoStartDate").value;

        const end =
            document.getElementById("promoEndDate").value;


        const status =
            document.getElementById("promoStatus");


        if (!start || !end) {

            status.value = "Active";

            return;
        }


        const today =
            new Date().toISOString().slice(0, 10);


        if (end < today) {

            status.value =
                "Expired";

        } else if (start > today) {

            status.value =
                "Upcoming";

        } else {

            status.value =
                "Active";
        }
    }


    async function savePromotion() {

        const promo =
            document.getElementById("promoName")
                .value
                .trim();

        const rate =
            Number(
                document.getElementById("promoRate")
                    .value
            );

        const type =
            document.getElementById("promoType")
                .value;

        const startDate =
            document.getElementById("promoStartDate")
                .value;

        const endDate =
            document.getElementById("promoEndDate")
                .value;


        if (!promo) {

            showMessage(
                "Promotion name is required.",
                "error"
            );

            return;
        }


        if (
            isNaN(rate) ||
            rate < 0
        ) {

            showMessage(
                "Please enter a valid rate.",
                "error"
            );

            return;
        }


        if (!startDate || !endDate) {

            showMessage(
                "Start date and end date are required.",
                "error"
            );

            return;
        }


        if (endDate < startDate) {

            showMessage(
                "End date cannot be before start date.",
                "error"
            );

            return;
        }


        const body = {

            promo: promo,

            Rate: rate,

            type: type,

            start_date: startDate,

            end_date: endDate,

            item_1:
                document.getElementById("item1").value.trim(),

            item_2:
                document.getElementById("item2").value.trim(),

            item_3:
                document.getElementById("item3").value.trim(),

            item_4:
                document.getElementById("item4").value.trim(),

            item_5:
                document.getElementById("item5").value.trim()
        };


        try {

            if (editingPromotionId) {

                body.promo_id =
                    editingPromotionId;

                await api("edit_promotion", {
                    method: "POST",
                    body: body
                });

                showMessage(
                    "Promotion updated.",
                    "success"
                );

            } else {

                await api("add_promotion", {
                    method: "POST",
                    body: body
                });

                showMessage(
                    "Promotion added.",
                    "success"
                );
            }


            closeModal("promotionModal");

            await loadData();

        } catch (error) {

            showMessage(
                error.message,
                "error"
            );
        }
    }

    function exportExcel() {

        const rows =
            currentFilteredSubmissions;


        if (!rows.length) {

            showMessage(
                "There are no filtered rows to export.",
                "error"
            );

            return;
        }


        let html = `
            <table border="1">

                <tr>
                    <th>ID</th>
                    <th>SMC</th>
                    <th>Staff</th>
                    <th>Outlet</th>
                    <th>Promotion</th>
                    <th>Type</th>
                    <th>Rate</th>
                    <th>Qty</th>
                    <th>Total</th>
                    <th>Status</th>
                    <th>Date</th>
                </tr>
        `;


        rows.forEach(s => {

            const total =
                Number(s.quantity || 0) *
                Number(s.Rate || 0);


            html += `
                <tr>

                    <td>${escapeHtml(s.incentive_id)}</td>

                    <td>${escapeHtml(s.SMC)}</td>

                    <td>${escapeHtml(s.staff_name)}</td>

                    <td>${escapeHtml(s.outlet)}</td>

                    <td>${escapeHtml(s.promo)}</td>

                    <td>${escapeHtml(s.type)}</td>

                    <td>${Number(s.Rate || 0).toFixed(2)}</td>

                    <td>${escapeHtml(s.quantity)}</td>

                    <td>${total.toFixed(2)}</td>

                    <td>${escapeHtml(s.status)}</td>

                    <td>${escapeHtml(s.submission_date)}</td>

                </tr>
            `;
        });


        html += "</table>";


        const blob =
            new Blob(
                [
                    "\ufeff" +
                    html
                ],
                {
                    type:
                        "application/vnd.ms-excel"
                }
            );


        const url =
            URL.createObjectURL(blob);


        const a =
            document.createElement("a");

        a.href = url;

        a.download =
            "incentive_submissions_" +
            selectedMonth +
            ".xls";

        document.body.appendChild(a);

        a.click();

        a.remove();

        URL.revokeObjectURL(url);
    }
