/*
|--------------------------------------------------------------------------
| JSON INPUT
|--------------------------------------------------------------------------
*/

function getJsonInput()
{
    $raw = file_get_contents("php://input");

    if (!$raw) {
        return [];
    }

    $data = json_decode($raw, true);

    return is_array($data)
        ? $data
        : [];
}


/*
|--------------------------------------------------------------------------
| JSON RESPONSE
|--------------------------------------------------------------------------
*/

function response($data, $status = 200)
{
    http_response_code($status);

    echo json_encode(
        $data,
        JSON_UNESCAPED_UNICODE |
        JSON_UNESCAPED_SLASHES
    );

    exit;
}


/*
|--------------------------------------------------------------------------
| REQUIRED FIELD
|--------------------------------------------------------------------------
*/

function requireField($data, $field)
{
    if (
        !isset($data[$field]) ||
        trim((string)$data[$field]) === ""
    ) {

        response([
            "success" => false,
            "message" =>
                "Missing required field: " . $field
        ], 400);
    }

    return trim((string)$data[$field]);
}


/*
|--------------------------------------------------------------------------
| PROMOTION STATUS
|--------------------------------------------------------------------------
*/

function calculatePromotionStatus(
    $startDate,
    $endDate
) {

    $today = date("Y-m-d");

    if (
        !$startDate ||
        !$endDate
    ) {
        return "Active";
    }

    if ($endDate < $today) {
        return "Expired";
    }

    if ($startDate > $today) {
        return "Upcoming";
    }

    return "Active";
}


/*
|--------------------------------------------------------------------------
| SYNC PROMOTION STATUSES
|--------------------------------------------------------------------------
*/

function syncPromotionStatuses($db)
{
    $sql = "
        UPDATE Incentives_information
        SET status =
            CASE
                WHEN end_date < CURDATE()
                    THEN 'Expired'

                WHEN start_date > CURDATE()
                    THEN 'Upcoming'

                ELSE 'Active'
            END

        WHERE start_date IS NOT NULL
          AND end_date IS NOT NULL
    ";

    $db->query($sql);
}


/*
|--------------------------------------------------------------------------
| VALIDATE MONTH
|--------------------------------------------------------------------------
*/

function normalizeMonth($month)
{
    if (
        !$month ||
        !preg_match(
            '/^\d{4}-\d{2}$/',
            $month
        )
    ) {

        return date("Y-m");
    }

    return $month;
}


/*
|--------------------------------------------------------------------------
| MONTH RANGE
|--------------------------------------------------------------------------
*/

function getMonthRange($month)
{
    $start =
        DateTime::createFromFormat(
            "Y-m-d",
            $month . "-01"
        );

    if (!$start) {
        $start = new DateTime(
            date("Y-m-01")
        );
    }

    $end =
        clone $start;

    $end->modify("+1 month");

    return [
        $start->format("Y-m-d"),
        $end->format("Y-m-d")
    ];
}


/*
|--------------------------------------------------------------------------
| FETCH ALL SUBMISSIONS
|--------------------------------------------------------------------------
*/

function fetchSubmissions($db)
{

    $sql = "
        SELECT
            si.incentive_id,
            si.SMC,
            sd.staff_name,
            sd.outlet,

            si.promo_id,

            ii.promo,
            ii.Rate,
            ii.type,
            ii.start_date,
            ii.end_date,
            ii.status AS promo_status,

            si.quantity,
            si.status,
            si.submission_date,
            si.proof,
            si.rejection_reason,

            bi.item_1,
            bi.item_2,
            bi.item_3,
            bi.item_4,
            bi.item_5

        FROM staff_incentives si

        LEFT JOIN staff_details sd
            ON sd.sales_man_code = si.SMC

        LEFT JOIN Incentives_information ii
            ON ii.promo_id = si.promo_id

        LEFT JOIN Bundle_information bi
            ON bi.promo_id = si.promo_id

        ORDER BY
            si.submission_date DESC,
            si.incentive_id DESC
    ";


    $result =
        $db->query($sql);


    if (!$result) {

        response([
            "success" => false,
            "message" =>
                "Failed to fetch submissions: " .
                $db->error
        ], 500);
    }


    $rows = [];


    while (
        $row =
        $result->fetch_assoc()
    ) {

        $row["quantity"] =
            (int)$row["quantity"];

        $row["Rate"] =
            (float)$row["Rate"];

        $rows[] =
            $row;
    }


    return $rows;
}

/*
|--------------------------------------------------------------------------
| MONTH FILTER
|--------------------------------------------------------------------------
*/

function rowsForMonth(
    $rows,
    $month
) {

    return array_values(
        array_filter(
            $rows,
            function ($row) use ($month) {

                return substr(
                    (string)($row["submission_date"] ?? ""),
                    0,
                    7
                ) === $month;
            }
        )
    );
}


/*
|--------------------------------------------------------------------------
| DASHBOARD
|--------------------------------------------------------------------------
*/

function buildDashboard(
    $rows,
    $month,
    $staffCount
) {

    $monthRows =
        rowsForMonth(
            $rows,
            $month
        );


    $total =
        count($monthRows);

    $pending = 0;
    $approved = 0;
    $rejected = 0;

    $approvedIndividual = 0;
    $bundlePool = 0;


    foreach ($monthRows as $row) {

        $status =
            strtolower(
                (string)$row["status"]
            );

        $type =
            strtolower(
                (string)$row["type"]
            );


        $quantity =
            (float)($row["quantity"] ?? 0);

        $rate =
            (float)($row["Rate"] ?? 0);


        if ($status === "pending") {
            $pending++;
        }

        if ($status === "approved") {

            $approved++;

            $amount =
                $quantity * $rate;


            if ($type === "individual") {

                $approvedIndividual +=
                    $amount;

            } elseif (
                $type === "outlet_bundle"
            ) {

                $bundlePool +=
                    $amount;
            }
        }


        if ($status === "rejected") {
            $rejected++;
        }
    }


    return [
        "totalSubmissions" =>
            $total,

        "pending" =>
            $pending,

        "approved" =>
            $approved,

        "rejected" =>
            $rejected,

        "approvedIndividual" =>
            round(
                $approvedIndividual,
                2
            ),

        "bundlePool" =>
            round(
                $bundlePool,
                2
            ),

        "staffCount" =>
            $staffCount
    ];
}


/*
|--------------------------------------------------------------------------
| MONTHLY SUMMARY
|--------------------------------------------------------------------------
*/

function buildMonthlySummary(
    $rows,
    $month
) {

    $monthRows =
        rowsForMonth(
            $rows,
            $month
        );


    $individual = [];
    $bundle = [];


    foreach ($monthRows as $row) {

        if (
            strtolower(
                (string)$row["status"]
            ) !== "approved"
        ) {
            continue;
        }


        $type =
            strtolower(
                (string)$row["type"]
            );


        $quantity =
            (float)($row["quantity"] ?? 0);

        $rate =
            (float)($row["Rate"] ?? 0);

        $amount =
            $quantity * $rate;


        if ($type === "individual") {

            $key =
                (string)$row["SMC"];


            if (!isset($individual[$key])) {

                $individual[$key] = [
                    "SMC" =>
                        $row["SMC"],

                    "staff_name" =>
                        $row["staff_name"],

                    "outlet" =>
                        $row["outlet"],

                    "quantity" =>
                        0,

                    "incentive" =>
                        0
                ];
            }


            $individual[$key]["quantity"] +=
                $quantity;

            $individual[$key]["incentive"] +=
                $amount;
        }


        if ($type === "outlet_bundle") {

            $key =
                (string)$row["outlet"];


            if (!isset($bundle[$key])) {

                $bundle[$key] = [
                    "outlet" =>
                        $row["outlet"],

                    "quantity" =>
                        0,

                    "incentive" =>
                        0
                ];
            }


            $bundle[$key]["quantity"] +=
                $quantity;

            $bundle[$key]["incentive"] +=
                $amount;
        }
    }


    $individual =
        array_values(
            $individual
        );

    $bundle =
        array_values(
            $bundle
        );


    usort(
        $individual,
        function ($a, $b) {

            return
                $b["incentive"] <=>
                $a["incentive"];
        }
    );


    usort(
        $bundle,
        function ($a, $b) {

            return
                $b["incentive"] <=>
                $a["incentive"];
        }
    );


    $totalIndividual = 0;
    $totalBundle = 0;


    foreach ($individual as $row) {

        $totalIndividual +=
            $row["incentive"];
    }


    foreach ($bundle as $row) {

        $totalBundle +=
            $row["incentive"];
    }


    return [

        "individual" =>
            array_map(
                function ($row) {

                    $row["quantity"] =
                        (int)$row["quantity"];

                    $row["incentive"] =
                        round(
                            $row["incentive"],
                            2
                        );

                    return $row;
                },
                $individual
            ),

        "bundle" =>
            array_map(
                function ($row) {

                    $row["quantity"] =
                        (int)$row["quantity"];

                    $row["incentive"] =
                        round(
                            $row["incentive"],
                            2
                        );

                    return $row;
                },
                $bundle
            ),

        "totalIndividual" =>
            round(
                $totalIndividual,
                2
            ),

        "totalBundle" =>
            round(
                $totalBundle,
                2
            ),

        "grandTotal" =>
            round(
                $totalIndividual +
                $totalBundle,
                2
            )
    ];
}


/*
|--------------------------------------------------------------------------
| TOP 3
|--------------------------------------------------------------------------
*/

function buildTop3(
    $rows,
    $month
) {

    $monthRows =
        rowsForMonth(
            $rows,
            $month
        );


    /*
    |--------------------------------------------------------------------------
    | Aggregate approved individual incentive by SMC
    |--------------------------------------------------------------------------
    */

    $staff = [];


    $bundlePool = 0;


    foreach ($monthRows as $row) {

        if (
            strtolower(
                (string)$row["status"]
            ) !== "approved"
        ) {
            continue;
        }


        $type =
            strtolower(
                (string)$row["type"]
            );


        $quantity =
            (float)($row["quantity"] ?? 0);

        $rate =
            (float)($row["Rate"] ?? 0);


        $amount =
            $quantity * $rate;


        if ($type === "outlet_bundle") {

            $bundlePool +=
                $amount;

            continue;
        }


        if ($type !== "individual") {
            continue;
        }


        $smc =
            (string)$row["SMC"];


        if (!isset($staff[$smc])) {

            $staff[$smc] = [
                "SMC" =>
                    $row["SMC"],

                "staff_name" =>
                    $row["staff_name"],

                "outlet" =>
                    $row["outlet"],

                "individual_incentive" =>
                    0
            ];
        }


        $staff[$smc]["individual_incentive"] +=
            $amount;
    }


    /*
    |--------------------------------------------------------------------------
    | Sort
    |--------------------------------------------------------------------------
    */

    usort(
        $staff,
        function ($a, $b) {

            $compare =
                $b["individual_incentive"]
                <=>
                $a["individual_incentive"];


            if ($compare !== 0) {
                return $compare;
            }


            return strcmp(
                (string)$a["staff_name"],
                (string)$b["staff_name"]
            );
        }
    );


    /*
    |--------------------------------------------------------------------------
    | First 3
    |--------------------------------------------------------------------------
    */

    $top3 =
        array_slice(
            $staff,
            0,
            3
        );


    /*
    |--------------------------------------------------------------------------
    | Bundle shares
    |--------------------------------------------------------------------------
    */

    $shareRates = [
        1 => 0.50,
        2 => 0.35,
        3 => 0.15
    ];


    foreach (
        $top3 as $index => &$person
    ) {

        $rank =
            $index + 1;


        $person["rank"] =
            $rank;


        $person["bundle_pool"] =
            round(
                $bundlePool,
                2
            );


        $person["bundle_share"] =
            round(
                $bundlePool *
                $shareRates[$rank],
                2
            );


        $person["total_payout"] =
            round(
                $person["individual_incentive"] +
                $person["bundle_share"],
                2
            );


        $person["individual_incentive"] =
            round(
                $person["individual_incentive"],
                2
            );
    }

    unset($person);


    /*
    |--------------------------------------------------------------------------
    | Promotion-level breakdown
    |--------------------------------------------------------------------------
    */

    $topSMCs = [];


    foreach ($top3 as $person) {

        $topSMCs[
            (string)$person["SMC"]
        ] =
            (int)$person["rank"];
    }


    $breakdown = [];


    foreach ($monthRows as $row) {

        if (
            strtolower(
                (string)$row["status"]
            ) !== "approved"
        ) {
            continue;
        }


        $type =
            strtolower(
                (string)$row["type"]
            );


        if ($type !== "individual") {
            continue;
        }


        $smc =
            (string)$row["SMC"];


        if (
            !isset(
                $topSMCs[$smc]
            )
        ) {
            continue;
        }


        $quantity =
            (float)($row["quantity"] ?? 0);

        $rate =
            (float)($row["Rate"] ?? 0);


        $breakdown[] = [
            "rank" =>
                $topSMCs[$smc],

            "SMC" =>
                $row["SMC"],

            "staff_name" =>
                $row["staff_name"],

            "promo_id" =>
                $row["promo_id"],

            "promo" =>
                $row["promo"],

            "type" =>
                $row["type"],

            "Rate" =>
                $rate,

            "quantity" =>
                $quantity,

            "incentive" =>
                $quantity * $rate
        ];
    }


    /*
    |--------------------------------------------------------------------------
    | Aggregate same staff + same promotion
    |--------------------------------------------------------------------------
    */

    $grouped = [];


    foreach ($breakdown as $row) {

        $key =
            $row["SMC"] .
            "|" .
            $row["promo_id"];


        if (!isset($grouped[$key])) {

            $grouped[$key] =
                $row;

        } else {

            $grouped[$key]["quantity"] +=
                $row["quantity"];

            $grouped[$key]["incentive"] +=
                $row["incentive"];
        }
    }


    $breakdown =
        array_values(
            $grouped
        );


    usort(
        $breakdown,
        function ($a, $b) {

            if (
                $a["rank"] !==
                $b["rank"]
            ) {

                return
                    $a["rank"] <=>
                    $b["rank"];
            }


            return
                $b["incentive"] <=>
                $a["incentive"];
        }
    );


    foreach ($breakdown as &$row) {

        $row["Rate"] =
            round(
                $row["Rate"],
                2
            );

        $row["quantity"] =
            (int)$row["quantity"];

        $row["incentive"] =
            round(
                $row["incentive"],
                2
            );
    }

    unset($row);


    return [

        "top3" =>
            $top3,

        "bundlePool" =>
            round(
                $bundlePool,
                2
            ),

        "breakdown" =>
            $breakdown
    ];
}


/*
|--------------------------------------------------------------------------
| STAFF PROMOTION BREAKDOWN
|--------------------------------------------------------------------------
*/

function getStaffPromotionBreakdown(
    $db,
    $month,
    $smc
) {

    [$startDate, $endDate] =
        getMonthRange($month);


    $sql = "
        SELECT

            si.SMC,

            sd.staff_name,

            sd.outlet,

            si.promo_id,

            ii.promo,

            ii.type,

            ii.Rate,

            SUM(si.quantity) AS quantity,

            SUM(
                si.quantity *
                ii.Rate
            ) AS incentive

        FROM staff_incentives si

        LEFT JOIN staff_details sd
            ON sd.sales_man_code =
               si.SMC

        LEFT JOIN Incentives_information ii
            ON ii.promo_id =
               si.promo_id

        WHERE
            si.status = 'approved'

            AND ii.type = 'individual'

            AND si.submission_date >= ?

            AND si.submission_date < ?

            AND si.SMC = ?

        GROUP BY

            si.SMC,
            sd.staff_name,
            sd.outlet,

            si.promo_id,
            ii.promo,
            ii.type,
            ii.Rate

        ORDER BY
            incentive DESC,
            ii.promo ASC
    ";


    $stmt =
        $db->prepare($sql);


    if (!$stmt) {

        response([
            "success" => false,
            "message" =>
                "Failed to prepare breakdown query: " .
                $db->error
        ], 500);
    }


    $stmt->bind_param(
        "sss",
        $startDate,
        $endDate,
        $smc
    );


    if (!$stmt->execute()) {

        response([
            "success" => false,
            "message" =>
                "Failed to fetch breakdown: " .
                $stmt->error
        ], 500);
    }


    $result =
        $stmt->get_result();


    $rows = [];


    while (
        $row =
        $result->fetch_assoc()
    ) {

        $row["Rate"] =
            (float)$row["Rate"];

        $row["quantity"] =
            (int)$row["quantity"];

        $row["incentive"] =
            (float)$row["incentive"];

        $rows[] =
            $row;
    }


    $stmt->close();


    return $rows;
}

if (

    $data =

        requireField(

        );

        isset($data["Rate"])
            ? (float)$data["Rate"]
            : 0;


    $type =
        requireField(
            $data,
            "type"
        );


    $startDate =
        requireField(
            $data,
            "start_date"
        );


    $endDate =
        requireField(
            $data,
            "end_date"
        );


    if (
        !in_array(
            $type,
            [
                "individual",
                "outlet_bundle"
            ],
            true
        )
    ) {

        response([
            "success" => false,
            "message" =>
                "Invalid promotion type."
        ], 400);
    }


    if ($rate < 0) {

        response([
            "success" => false,
            "message" =>
                "Rate cannot be negative."
        ], 400);
    }


    if ($endDate < $startDate) {

        response([
            "success" => false,
            "message" =>
                "End date cannot be before start date."
        ], 400);
    }


    $status =
        calculatePromotionStatus(
            $startDate,
            $endDate
        );


    $item1 =
        trim(
            (string)($data["item_1"] ?? "")
        );

    $item2 =
        trim(
            (string)($data["item_2"] ?? "")
        );

    $item3 =
        trim(
            (string)($data["item_3"] ?? "")
        );

    $item4 =
        trim(
            (string)($data["item_4"] ?? "")
        );

    $item5 =
        trim(
            (string)($data["item_5"] ?? "")
        );


    $mysqli->begin_transaction();


    try {

        $sql = "
            INSERT INTO Incentives_information
            (
                promo,
                Rate,
                type,
                start_date,
                end_date,
                status
            )

            VALUES
            (
                ?,
                ?,
                ?,
                ?,
                ?,
                ?
            )
        ";


        $stmt =
            $mysqli->prepare($sql);


        if (!$stmt) {
            throw new Exception(
                $mysqli->error
            );
        }


        $stmt->bind_param(
            "sdssss",
            $promo,
            $rate,
            $type,
            $startDate,
            $endDate,
            $status
        );


        if (!$stmt->execute()) {
            throw new Exception(
                $stmt->error
            );
        }


        $promoId =
            $stmt->insert_id;


        $stmt->close();


        /*
        |--------------------------------------------------------------------------
        | Bundle Information
        |--------------------------------------------------------------------------
        */

        if (
            $type ===
            "outlet_bundle"
        ) {

            $sql = "
                INSERT INTO Bundle_information
                (
                    promo_id,
                    item_1,
                    item_2,
                    item_3,
                    item_4,
                    item_5
                )

                VALUES
                (
                    ?,
                    ?,
                    ?,
                    ?,
                    ?,
                    ?
                )
            ";


            $stmt =
                $mysqli->prepare($sql);


            if (!$stmt) {
                throw new Exception(
                    $mysqli->error
                );
            }


            $stmt->bind_param(
                "isssss",
                $promoId,
                $item1,
                $item2,
                $item3,
                $item4,
                $item5
            );


            if (!$stmt->execute()) {
                throw new Exception(
                    $stmt->error
                );
            }


            $stmt->close();
        }


        $mysqli->commit();


        response([
            "success" =>
                true,

            "message" =>
                "Promotion added.",

            "promo_id" =>
                $promoId
        ]);

    } catch (
        Throwable $e
    ) {

        $mysqli->rollback();


        response([
            "success" => false,
            "message" =>
                "Failed to add promotion: " .

        ], 500);
    }
