function escapeHtml(value) {

    if (
        value === null ||
        value === undefined
    ) {
        return "";
    }

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");

}


function money(value) {

    const number = Number(value);

    if (!Number.isFinite(number)) {
        return "S$0.00";
    }

    return "S$" + number.toFixed(2);

}


function numberValue(value) {

    const number = Number(value);

    return Number.isFinite(number)
        ? number
        : 0;

}


function normalise(value) {

    return String(value ?? "")
        .trim()
        .toLowerCase();

}


/* ============================================================
   SUBMISSION ID
   IMPORTANT FIX
============================================================ */

function getSubmissionId(row) {

    if (!row) {
        return 0;
    }

    const raw =
        row.incentive_id ??
        row.incentiveId ??
        row.submission_id ??
        row.submissionId ??
        row.id;

    const id = Number(raw);

    if (
        !Number.isInteger(id) ||
        id <= 0
    ) {
        return 0;
    }

    return id;

}


/* ============================================================
   FIELD HELPERS
============================================================ */

function getStaff(row) {

    return row.staff_name ??
        row.staffName ??
        row.name ??
        "";

}


function getSMC(row) {

    return row.SMC ??
        row.smc ??
        row.sales_man_code ??
        row.salesManCode ??
        "";

}


function getOutlet(row) {

    return row.outlet ??
        row.outlet_name ??
        row.outletName ??
        "";

}


function getPromo(row) {

    return row.promo ??
        row.promotion ??
        row.promo_name ??
        row.promotion_name ??
        "";

}


function getPromoId(row) {

    const raw =
        row.promo_id ??
        row.promotion_id ??
        row.promoId;

    const id = Number(raw);

    return (
        Number.isInteger(id) &&
        id > 0
    )
        ? id
        : 0;

}


function getQuantity(row) {

    return numberValue(
        row.quantity ??
        row.qty
    );

}


function getRate(row) {

    return numberValue(
        row.Rate ??
        row.rate ??
        row.incentive_rate ??
        row.promo_rate
    );

}


function getStatus(row) {

    return row.status ??
        row.submission_status ??
        "";

}


function getDate(row) {

    return row.submission_date ??
        row.submitted_at ??
        row.created_at ??
        row.date ??
        "";

}


/* ============================================================
   CRITICAL TYPE FIX
============================================================ */

/*
 * The submission may not contain `type`.
 *
 * The type belongs to Incentives_information.
 *
 * Therefore:
 *
 * staff_incentives.promo_id
 *          ↓
 * Incentives_information.promo_id
 *          ↓
 * type
 *
 * This prevents Bundle from being treated as Individual.
 */

function getPromotionById(promoId) {

    promoId = Number(promoId);

    if (
        !Number.isInteger(promoId) ||
        promoId <= 0
    ) {
        return null;
    }

    return state.promotions.find(
        promo => {

            const id = Number(
                promo.promo_id ??
                promo.promotion_id ??
                promo.id
            );

            return id === promoId;

        }
    ) || null;

}


function getSubmissionType(row) {

    /*
     * First check whether the submission itself has
     * a meaningful type.
     */

    const directType =
        row.type ??
        row.promotion_type ??
        row.promo_type;


    if (
        directType !== null &&
        directType !== undefined &&
        String(directType).trim() !== ""
    ) {

        return String(directType).trim();

    }


    /*
     * If not, resolve type using promo_id.
     */

    const promo =
        getPromotionById(
            getPromoId(row)
        );


    if (promo) {

        const promoType =
            promo.type ??
            promo.promotion_type ??
            promo.promo_type;


        if (
            promoType !== null &&
            promoType !== undefined &&
            String(promoType).trim() !== ""
        ) {

            return String(
                promoType
            ).trim();

        }

    }


    /*
     * Do NOT assume Individual here.
     *
     * Unknown stays Unknown.
     */
    return "Unknown";

}


function isBundle(row) {

    return normalise(
        getSubmissionType(row)
    ).includes("bundle");

}


function isIndividual(row) {

    return normalise(
        getSubmissionType(row)
    ).includes("individual");

}


function getAmount(row) {

    return (
        getQuantity(row) *
        getRate(row)
    );

}


/* ============================================================
   DATE
============================================================ */

function formatDate(value) {

    if (!value) {
        return "-";
    }

    const text = String(value);

    if (
        /^\d{4}-\d{2}-\d{2}/.test(text)
    ) {
        return text.substring(0, 10);
    }

    const date = new Date(text);

    if (
        Number.isNaN(
            date.getTime()
        )
    ) {
        return text;
    }

    return date.toLocaleDateString(
        "en-SG"
    );

}

function statusBadge(status) {

    const value =
        normalise(status);

    let className =
        "status-default";


    if (value === "approved") {

        className =
            "status-approved";

    } else if (
        value === "pending"
    ) {

        className =
            "status-pending";

    } else if (
        value === "rejected"
    ) {

        className =
            "status-rejected";

    }


    return `
        <span class="status ${className}">
            ${escapeHtml(
                status || "Unknown"
            )}
        </span>
    `;

}


function typeBadge(type) {

    const value =
        normalise(type);

    const bundle =
        value.includes("bundle");


    return `
        <span class="type-badge ${bundle ? "type-bundle" : ""}">
            ${escapeHtml(
                type || "Unknown"
            )}
        </span>
    `;

}

async function apiRequest(
    action,
    data = {}
) {

    const response =
        await fetch(
            "admin.php?action=" +
            encodeURIComponent(action),
            {

                method: "POST",

                credentials: "same-origin",

                headers: {
                    "Content-Type":
                        "application/json",
                    "Accept":
                        "application/json"
                },

                body:
                    JSON.stringify(data)

            }
        );


    const text =
        await response.text();


    let json;


    try {

        json =
            JSON.parse(text);

    } catch (error) {

        throw new Error(
            text
                ? text.substring(
                    0,
                    500
                )
                : `Request failed (${response.status})`
        );

    }


    if (
        !response.ok ||
        json.success === false
    ) {

        throw new Error(
            json.message ||
            json.error ||
            `Request failed (${response.status})`
        );

    }


    return json;

}

/* ============================================================
   REFRESH
============================================================ */

async function refreshAdminData(
    options = {}
) {

    if (
        state.refreshInProgress
    ) {
        return;
    }


    state.refreshInProgress =
        true;


    try {

        const result =
            await apiRequest(
                "bootstrap"
            );


        normaliseBootstrap(
            result
        );


        /*
         * Render immediately.
         *
         * Because getSubmissionType() now resolves
         * the type through promo_id, dashboard totals
         * are also recalculated correctly.
         */
        renderAll();


        /*
         * Get selected month.
         */
        await loadMonthlySummary(
            false
        );


        renderMonthlySummary();


        /*
         * If the user was viewing a submission
         * during refresh, reopen it.
         */
        if (
            options.keepView &&
            state.currentSubmissionId
        ) {

            await showSubmission(
                state.currentSubmissionId,
                true
            );

        }


    } catch (error) {

        console.error(
            "Refresh error:",
            error
        );


        showMessage(
            error.message ||
            "Unable to refresh admin data.",
            "error"
        );


    } finally {

        state.refreshInProgress =
            false;

    }

}

/* ============================================================
   DASHBOARD CALCULATION
============================================================ */

function calculateDashboardTotals() {

    let individual = 0;

    let bundle = 0;

    let pending = 0;

    let approved = 0;

    let rejected = 0;


    state.submissions.forEach(
        row => {

            const status =
                normalise(
                    getStatus(row)
                );


            const amount =
                getAmount(row);


            /*
             * ONLY approved contributes to incentive amount.
             */

            if (
                status ===
                "approved"
            ) {

                approved++;


                /*
                 * THIS IS THE TYPE FIX.
                 *
                 * We use getSubmissionType()
                 * rather than directly using row.type.
                 */

                if (
                    isBundle(row)
                ) {

                    bundle += amount;

                } else if (
                    isIndividual(row)
                ) {

                    individual += amount;

                }

            } else if (
                status ===
                "pending"
            ) {

                pending++;

            } else if (
                status ===
                "rejected"
            ) {

                rejected++;

            }

        }
    );


    return {

        individual,

        bundle,

        total:
            individual + bundle,

        pending,

        approved,

        rejected,

        totalStaff:
            state.users.length

    };

}

/* ============================================================
   FILTER OPTIONS
============================================================ */

function populateSelect(
    elementId,
    values,
    allLabel
) {

    const select =
        document.getElementById(
            elementId
        );


    if (!select) {
        return;
    }


    const oldValue =
        select.value;


    const unique =
        [
            ...new Set(
                values
                    .map(
                        value =>
                            String(
                                value ??
                                ""
                            ).trim()
                    )
                    .filter(Boolean)
            )
        ]
        .sort(
            (a, b) =>
                a.localeCompare(b)
        );


    select.innerHTML = `

        <option value="">
            ${escapeHtml(allLabel)}
        </option>

        ${unique.map(value => `

            <option
                value="${escapeHtml(value)}">

                ${escapeHtml(value)}

            </option>

        `).join("")}

    `;


    if (
        unique.includes(
            oldValue
        )
    ) {

        select.value =
            oldValue;

    }

}

/* ============================================================
   FILTERED SUBMISSIONS
============================================================ */

function getFilteredSubmissions() {

    const search =
        normalise(
            document.getElementById(
                "submissionSearch"
            ).value
        );


    const status =
        normalise(
            document.getElementById(
                "submissionStatusFilter"
            ).value
        );


    const type =
        normalise(
            document.getElementById(
                "submissionTypeFilter"
            ).value
        );


    const staff =
        document.getElementById(
            "submissionStaffFilter"
        ).value;


    const outlet =
        document.getElementById(
            "submissionOutletFilter"
        ).value;


    const promo =
        document.getElementById(
            "submissionPromoFilter"
        ).value;


    return state.submissions.filter(
        row => {

            const staffName =
                getStaff(row);


            const smc =
                getSMC(row);


            const outletName =
                getOutlet(row);


            const promotion =
                getPromo(row);


            const rowStatus =
                normalise(
                    getStatus(row)
                );


            const rowType =
                normalise(
                    getSubmissionType(row)
                );


            if (search) {

                const combined =
                    normalise(
                        `${staffName} ${smc} ${outletName} ${promotion}`
                    );


                if (
                    !combined.includes(
                        search
                    )
                ) {
                    return false;
                }

            }


            if (
                status &&
                rowStatus !== status
            ) {
                return false;
            }


            if (
                type &&
                !rowType.includes(type)
            ) {
                return false;
            }


            if (
                staff &&
                staffName !== staff
            ) {
                return false;
            }


            /*
             * OUTLET FILTER
             */
            if (
                outlet &&
                outletName !== outlet
            ) {
                return false;
            }


            if (
                promo &&
                promotion !== promo
            ) {
                return false;
            }


            return true;

        }
    );

}

/* ============================================================
   APPROVE
============================================================ */

async function approveSubmission(
    id
) {

    id = Number(id);


    if (
        !Number.isInteger(id) ||
        id <= 0
    ) {

        showMessage(
            "Submission ID invalid.",
            "error"
        );

        return;

    }


    if (
        !confirm(
            `Approve submission #${id}?`
        )
    ) {
        return;
    }


    try {

        const result =
            await apiRequest(
                "approve",
                {
                    incentive_id: id
                }
            );


        showMessage(
            result.message ||
            "Submission approved successfully.",
            "success"
        );


        closeModal(
            "viewModal"
        );


        /*
         * Refresh everything.
         *
         * This recalculates:
         * Individual
         * Bundle
         * Total
         * Monthly Summary
         * Top 3
         */
        await refreshAdminData();

    } catch (error) {

        console.error(
            "Approve error:",
            error
        );


        showMessage(
            error.message ||
            "Failed to approve submission.",
            "error"
        );

    }

}


/* ============================================================
   REJECT
============================================================ */

async function rejectSubmission(
    id
) {

    id = Number(id);


    if (
        !Number.isInteger(id) ||
        id <= 0
    ) {

        showMessage(
            "Submission ID invalid.",
            "error"
        );

        return;

    }


    const reason =
        window.prompt(
            `Enter rejection reason for submission #${id}:`
        );


    if (
        reason === null
    ) {
        return;
    }


    const trimmedReason =
        reason.trim();


    if (!trimmedReason) {

        showMessage(
            "Rejection reason is required.",
            "error"
        );

        return;

    }


    if (
        !confirm(
            `Reject submission #${id}?`
        )
    ) {
        return;
    }


    try {

        const result =
            await apiRequest(
                "reject",
                {
                    incentive_id: id,
                    rejection_reason:
                        trimmedReason,
                    reason:
                        trimmedReason
                }
            );


        showMessage(
            result.message ||
            "Submission rejected successfully.",
            "success"
        );


        closeModal(
            "viewModal"
        );


        await refreshAdminData();

    } catch (error) {

        console.error(
            "Reject error:",
            error
        );


        showMessage(
            error.message ||
            "Failed to reject submission.",
            "error"
        );

    }

}


/* ============================================================

function getDefaultMonth() {

        new Date();


    return (
        now.getFullYear() +
        "-" +

        ).padStart(2, "0")
    );

}

}


/* ============================================================
   LOCAL MONTHLY FALLBACK
============================================================ */

function calculateLocalMonthlySummary(
    monthValue
) {

    const groups =
        new Map();


    let individual = 0;

    let bundle = 0;


    state.submissions.forEach(
        row => {

            if (
                normalise(
                    getStatus(row)
                ) !== "approved"
            ) {
                return;
            }


            const date =
                String(
                    getDate(row) || ""
                ).substring(
                    0,
                    7
                );


            if (
                date !==
                monthValue
            ) {
                return;
            }


            const smc =
                getSMC(row) ||
                "-";


            if (
                !groups.has(smc)
            ) {

                groups.set(
                    smc,
                    {

                        staff_name:
                            getStaff(row),

                        SMC:
                            smc,

                        outlet:
                            getOutlet(row),

                        individual:
                            0,

                        bundle:
                            0,

                        total:
                            0

                    }
                );

            }


            const group =
                groups.get(smc);


            const amount =
                getAmount(row);


            /*
             * SAME TYPE FIX USED HERE.
             */
            if (
                isBundle(row)
            ) {

                group.bundle +=
                    amount;

                bundle +=
                    amount;

            } else if (
                isIndividual(row)
            ) {

                group.individual +=
                    amount;

                individual +=
                    amount;

            }


            group.total +=
                amount;

        }
    );


    return {

        rows:
            [...groups.values()]
                .sort(
                    (a, b) =>
                        b.total -
                        a.total
                ),

        individual,

        bundle,

        total:
            individual +
            bundle

    };

}


/* ============================================================
   LOAD MONTHLY SUMMARY
============================================================ */

async function loadMonthlySummary(
    showErrors = true
) {

    if (
        state.monthlyLoading
    ) {
        return;
    }


    state.monthlyLoading =
        true;


    const monthValue =
        getSelectedMonth();


    const parts =
        monthValue.split("-");


    const year =
        Number(parts[0]);


    const month =
        Number(parts[1]);


    try {

        const result =
            await apiRequest(
                "monthly_summary",
                {
                    year,
                    month
                }
            );


        const data =
            result?.data ??
            result?.summary ??
            result;


        if (
            Array.isArray(data)
        ) {

            state.monthlySummary =
                data;

        } else if (
            Array.isArray(
                data?.rows
            )
        ) {

            state.monthlySummary =
                data.rows;

        } else {

            state.monthlySummary =
                [];

        }

    } catch (error) {

        console.warn(
            "Monthly summary API failed:",
            error
        );


        /*
         * Local fallback.
         */
        const local =
            calculateLocalMonthlySummary(
                monthValue
            );


        state.monthlySummary =
            local.rows;


        if (showErrors) {

            showMessage(
                "Using local monthly calculation.",
                "info"
            );

        }

    } finally {

        state.monthlyLoading =
            false;

    }

}


/* ============================================================
   MONTHLY SUMMARY RENDER
============================================================ */

function renderMonthlySummary() {

    const monthValue =
        getSelectedMonth();


    const parts =
        monthValue.split("-");


    const year =
        parts[0];


    const month =
        parts[1];


    document.getElementById(
        "summaryTitle"
    ).textContent =
        `Monthly Staff Summary - ${year}-${month}`;


    let rows =
        state.monthlySummary
            .map(row => {

                const individual =
                    numberValue(
                        row.individual ??
                        row.individual_total ??
                        row.individual_incentive ??
                        row.individual_amount
                    );


                const bundle =
                    numberValue(
                        row.bundle ??
                        row.bundle_total ??
                        row.bundle_incentive ??
                        row.bundle_amount
                    );


                let total =
                    numberValue(
                        row.total ??
                        row.total_incentive ??
                        row.total_amount
                    );


                if (
                    total === 0 &&
                    (
                        individual !== 0 ||
                        bundle !== 0
                    )
                ) {

                    total =
                        individual +
                        bundle;

                }


                return {

                    staff_name:
                        row.staff_name ??
                        row.staffName ??
                        row.name ??
                        "",

                    SMC:
                        row.SMC ??
                        row.smc ??
                        row.sales_man_code ??
                        "",

                    outlet:
                        row.outlet ??
                        row.outlet_name ??
                        "",

                    individual,

                    bundle,

                    total

                };

            });


    /*
     * If backend returns no data, calculate locally.
     */
    if (
        !rows.length
    ) {

        const local =
            calculateLocalMonthlySummary(
                monthValue
            );


        rows =
            local.rows;

    }


    let individualTotal = 0;

    let bundleTotal = 0;


    rows.forEach(
        row => {

            individualTotal +=
                row.individual;

            bundleTotal +=
                row.bundle;

        }
    );


    document.getElementById(
        "summaryIndividual"
    ).textContent =
        money(
            individualTotal
        );


    document.getElementById(
        "summaryBundle"
    ).textContent =
        money(
            bundleTotal
        );


    document.getElementById(
        "summaryTotal"
    ).textContent =
        money(
            individualTotal +
            bundleTotal
        );


    const tbody =
        document.getElementById(
            "summaryTable"
        );


    if (
        !rows.length
    ) {

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="6"
                    class="empty">
                    No approved incentives for this month.
                </td>
            </tr>
        `;

        return;

    }


    tbody.innerHTML =
        rows.map(row => {

            const total =
                row.total ||
                (
                    row.individual +
                    row.bundle
                );


            return `

                <tr>

                    <td>
                        ${escapeHtml(
                            row.staff_name
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            row.SMC
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            row.outlet
                        )}
                    </td>

                    <td class="money">
                        ${money(
                            row.individual
                        )}
                    </td>

                    <td class="money">
                        ${money(
                            row.bundle
                        )}
                    </td>

                    <td class="money">
                        ${money(total)}
                    </td>

                </tr>

            `;

        }).join("");

}


/* ============================================================
   TOP 3
============================================================ */

function calculateTop3() {

    const map =
        new Map();


    state.submissions.forEach(
        row => {

            if (
                normalise(
                    getStatus(row)
                ) !== "approved"
            ) {
                return;
            }


            const smc =
                getSMC(row) ||
                "-";


            if (
                !map.has(smc)
            ) {

                map.set(
                    smc,
                    {

                        staff_name:
                            getStaff(row),

                        SMC:
                            smc,

                        outlet:
                            getOutlet(row),

                        total:
                            0,

                        submissions:
                            0

                    }
                );

            }


            const person =
                map.get(smc);


            person.total +=
                getAmount(row);


            person.submissions++;

        }
    );


    return [
        ...map.values()
    ]
    .sort(
        (a, b) =>
            b.total -
            a.total
    )
    .slice(
        0,
        3
    );

}
