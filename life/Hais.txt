<script>

/* ================================================================

================================================================ */

let adminData = {
    admin: {},

    users: [],
    monthlySummary: [],

/* ================================================================
   HELPERS
================================================================ */

function $(id) {
    return document.getElementById(id);
}


function escapeHtml(value) {

    if (value === null || value === undefined) {
        return "";
    }

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}


function money(value) {

    const n = Number(value);

    if (!Number.isFinite(n)) {
        return "$0.00";
    }

    return "$" + n.toFixed(2);
}


function numberValue(value) {

    const n = Number(value);

    return Number.isFinite(n) ? n : 0;
}


function getId(row) {

    return row?.incentive_id ??
           row?.submission_id ??
           row?.id ??
           row?.ID ??
           "";
}


function getStaff(row) {

    return row?.staff_name ??
           row?.name ??
           row?.staff ??
           "";
}


function getSMC(row) {

    return row?.sales_man_code ??
           row?.SMC ??
           row?.smc ??
           "";
}


function getOutlet(row) {

    return row?.outlet ??
           row?.outlet_name ??
           "";
}


function getPromo(row) {

    return row?.promo ??
           row?.promotion ??
           row?.promo_name ??
           row?.promotion_name ??
           "";
}


function getPromoId(row) {

    return row?.promo_id ??
           row?.promotion_id ??
           "";
}


function getType(row) {

    return String(
        row?.type ??
        row?.promotion_type ??
        row?.promo_type ??
        ""
    ).toLowerCase();
}


function getQuantity(row) {

    return numberValue(
        row?.quantity ??
        row?.qty ??
        0
    );
}


function getRate(row) {

    return numberValue(
        row?.Rate ??
        row?.rate ??
        row?.incentive_rate ??
        row?.promo_rate ??
        0
    );
}


function getStatus(row) {

    return String(
        row?.status ??
        row?.submission_status ??
        ""
    ).toLowerCase();
}


function getDate(row) {

    return row?.submission_date ??
           row?.submitted_at ??
           row?.created_at ??
           row?.date ??
           "";
}


function getAmount(row) {

    if (
        row?.amount !== undefined &&
        row?.amount !== null
    ) {
        return numberValue(row.amount);
    }

    return getQuantity(row) * getRate(row);
}


function formatDate(value) {

    if (!value) {
        return "-";
    }

    const date = new Date(value);

    if (Number.isNaN(date.getTime())) {
        return escapeHtml(value);
    }

    return date.toLocaleDateString();
}


function formatStatus(status) {

    const value = String(status || "").toLowerCase();

    let cls = "status-default";

    if (value === "approved") {
        cls = "status-approved";
    }
    else if (value === "pending") {
        cls = "status-pending";
    }
    else if (value === "rejected") {
        cls = "status-rejected";
    }

    return `
        <span class="status ${cls}">
            ${escapeHtml(value || "-")}
        </span>
    `;
}


function formatType(type) {

    const value = String(type || "").toLowerCase();

    if (value === "bundle") {

        return `
            <span class="type-badge type-bundle">
                Bundle
            </span>
        `;
    }

    return `
        <span class="type-badge">
            Individual
        </span>
    `;
}


function showMessage(message, type = "info") {

    const box = $("message");

    box.textContent = message;

    box.className = "message show " + type;

    clearTimeout(showMessage.timer);

    showMessage.timer = setTimeout(() => {

        box.className = "message";

    }, 3500);
}


/* ================================================================
   API
================================================================ */

async function apiRequest(action, data = {}) {

    const url =
        "admin.php?action=" +
        encodeURIComponent(action);

    let response;

    try {

        response = await fetch(url, {

            method: "POST",

            credentials: "same-origin",

            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },

            body: JSON.stringify(data)

        });

    }
    catch (error) {

        throw new Error(
            "Unable to connect to the server."
        );

    }

    const text = await response.text();

    let result;

    try {

        result = JSON.parse(text);

    }
    catch (error) {

        console.error(
            "Invalid server response:",
            text
        );

        throw new Error(
            "Server returned an invalid response."
        );

    }

    if (!response.ok || result.success === false) {

        throw new Error(
            result.message ||
            result.error ||
            "Request failed."
        );

    }

    return result;
}


/* ================================================================
   LOAD / REFRESH
================================================================ */

async function loadAdminData() {

    try {

        const result =
            await apiRequest("bootstrap");

        const data =
            result.data || result;

        adminData.admin =
            data.admin || {};

        adminData.dashboard =
            data.dashboard || {};

        adminData.submissions =
            Array.isArray(data.submissions)
                ? data.submissions
                : [];

        adminData.promotions =
            Array.isArray(data.promotions)
                ? data.promotions
                : [];

        adminData.users =
            Array.isArray(data.users)
                ? data.users
                : [];

        adminData.monthlySummary =
            Array.isArray(data.monthlySummary)
                ? data.monthlySummary
                : [];

        adminData.top3 =
            Array.isArray(data.top3)
                ? data.top3
                : [];

        renderAll();

        await loadMonthlySummary(false);

    }
    catch (error) {

        console.error(error);

        showMessage(
            error.message || "Failed to load admin data.",
            "error"
        );

    }

}


async function refreshData() {

    const activeTab =
        document.querySelector(".tab.active")?.id ||
        "dashboardTab";

    const monthlyValue =
        $("monthlyFilter")?.value || "";

    showMessage(
        "Refreshing data...",
        "info"
    );

    try {

        await loadAdminData();

        if ($("monthlyFilter")) {
            $("monthlyFilter").value =
                monthlyValue ||
                $("monthlyFilter").value;
        }

        switchTab(activeTab);

        showMessage(
            "Data refreshed successfully.",
            "success"
        );

    }
    catch (error) {

        showMessage(
            error.message || "Refresh failed.",
            "error"
        );

    }

}

/* ================================================================
   DASHBOARD
================================================================ */

function renderDashboard() {

    const submissions =
        adminData.submissions || [];

    const pending =
        submissions.filter(
            row => getStatus(row) === "pending"
        );

    const approved =
        submissions.filter(
            row => getStatus(row) === "approved"
        );

    const rejected =
        submissions.filter(
            row => getStatus(row) === "rejected"
        );

    const individualTotal =
        approved
            .filter(row => getType(row) !== "bundle")
            .reduce(
                (sum, row) => sum + getAmount(row),
                0
            );

    const bundleTotal =
        approved
            .filter(row => getType(row) === "bundle")
            .reduce(
                (sum, row) => sum + getAmount(row),
                0
            );

    $("totalSubmissions").textContent =
        submissions.length;

    $("pendingSubmissions").textContent =
        pending.length;

    $("approvedSubmissions").textContent =
        approved.length;

    $("rejectedSubmissions").textContent =
        rejected.length;

    $("totalIncentive").textContent =
        money(individualTotal);

    $("totalBundle").textContent =
        money(bundleTotal);

    $("totalPayout").textContent =
        money(individualTotal + bundleTotal);


    const recent =
        submissions
            .slice()
            .sort(
                (a, b) =>
                    new Date(getDate(b)) -
                    new Date(getDate(a))
            )
            .slice(0, 10);


    const body =
        $("dashboardRecentBody");

    if (!recent.length) {

        body.innerHTML = `
            <tr>
                <td colspan="7" class="empty">
                    No submissions found.
                </td>
            </tr>
        `;

        return;
    }


    body.innerHTML =
        recent.map(row => {

            return `
                <tr>

                    <td>
                        ${escapeHtml(getId(row))}
                    </td>

                    <td>
                        ${escapeHtml(getStaff(row))}
                    </td>

                    <td>
                        ${escapeHtml(getPromo(row))}
                    </td>

                    <td>
                        ${formatType(getType(row))}
                    </td>

                    <td>
                        ${getQuantity(row)}
                    </td>

                    <td class="money">
                        ${money(getAmount(row))}
                    </td>

                    <td>
                        ${formatStatus(getStatus(row))}
                    </td>

                </tr>
            `;

        }).join("");

}


/* ================================================================
   SUBMISSION FILTER
================================================================ */

function populateSubmissionStaffFilter() {

    const select =
        $("submissionStaffFilter");

    if (!select) {
        return;
    }

    const current =
        select.value;

    const staffMap =
        new Map();

    adminData.submissions.forEach(row => {

        const smc =
            getSMC(row);

        const name =
            getStaff(row);

        if (smc) {
            staffMap.set(
                smc,
                name
                    ? `${name} (${smc})`
                    : smc
            );
        }

    });

    select.innerHTML =
        `<option value="">All Staff</option>` +
        Array.from(staffMap.entries())
            .sort()
            .map(
                ([smc, label]) =>
                    `<option value="${escapeHtml(smc)}">
                        ${escapeHtml(label)}
                    </option>`
            )
            .join("");

    if (
        Array.from(select.options)
            .some(option => option.value === current)
    ) {
        select.value = current;
    }
}


function renderSubmissions() {

    const body =
        $("submissionsBody");

    if (!body) {
        return;
    }

    const search =
        (
            $("submissionSearch")?.value ||
            ""
        ).trim().toLowerCase();

    const status =
        $("submissionStatusFilter")?.value ||
        "";

    const type =
        $("submissionTypeFilter")?.value ||
        "";

    const staff =
        $("submissionStaffFilter")?.value ||
        "";

    const date =
        $("submissionDateFilter")?.value ||
        "";


    const filtered =
        adminData.submissions.filter(row => {

            const text =
                (
                    String(getStaff(row)) +
                    " " +
                    String(getSMC(row)) +
                    " " +
                    String(getOutlet(row)) +
                    " " +
                    String(getPromo(row)) +
                    " " +
                    String(getId(row))
                ).toLowerCase();

            if (
                search &&
                !text.includes(search)
            ) {
                return false;
            }

            if (
                status &&
                getStatus(row) !== status
            ) {
                return false;
            }

            if (
                type &&
                getType(row) !== type
            ) {
                return false;
            }

            if (
                staff &&
                String(getSMC(row)) !==
                String(staff)
            ) {
                return false;
            }

            if (date) {

                const rowDate =
                    String(getDate(row))
                        .substring(0, 10);

                if (rowDate !== date) {
                    return false;
                }

            }

            return true;

        });


    if (!filtered.length) {

        body.innerHTML = `
            <tr>
                <td colspan="12" class="empty">
                    No submissions found.
                </td>
            </tr>
        `;

        return;
    }


    body.innerHTML =
        filtered.map(row => {

            const id =
                getId(row);

            return `
                <tr>

                    <td>
                        ${escapeHtml(id)}
                    </td>

                    <td>
                        ${escapeHtml(getStaff(row))}
                    </td>

                    <td>
                        ${escapeHtml(getSMC(row))}
                    </td>

                    <td>
                        ${escapeHtml(getOutlet(row))}
                    </td>

                    <td>
                        ${escapeHtml(getPromo(row))}
                    </td>

                    <td>
                        ${formatType(getType(row))}
                    </td>

                    <td>
                        ${getQuantity(row)}
                    </td>

                    <td class="money">
                        ${money(getRate(row))}
                    </td>

                    <td class="money">
                        ${money(getAmount(row))}
                    </td>

                    <td>
                        ${formatStatus(getStatus(row))}
                    </td>

                    <td>
                        ${formatDate(getDate(row))}
                    </td>

                    <td class="action-column">

                        <button
                            class="btn btn-primary btn-small"
                            onclick="viewSubmission(${Number(id)})"
                        >
                            View
                        </button>

                    </td>

                </tr>
            `;

        }).join("");
}

                );

                Number(
                    $("editQuantity").value
                );

            const status =
                $("editStatus").value;

            const rejectionReason =
                $("editRejectionReason")
                    .value
                    .trim();


            if (!id || !promoId) {

                );

            ) {

                showMessage(

                );

            ) {

                showMessage(

                );

                    .focus();

                return;
            }


            const button =
                event.target.querySelector(

                );


            if (button) {

                button.disabled = true;

                button.textContent =
                    "Saving...";

            }


            try {

                        incentive_id: id,
                        promo_id: promoId,
                        quantity: quantity,
                        status: status,
                        rejection_reason:

                    }

                    "editSubmissionModal"
                );

                showMessage(
                    error.message ||

                    "error"
                );

            }
            finally {

                if (button) {

                    button.disabled = false;

                    button.textContent =
                        "Save Changes";

                }

            }

        }
    );

/* ================================================================

================================================================ */

function getDefaultMonth() {

    const now =
        new Date();

    return (
        now.getFullYear() +
        "-" +
        String(
            now.getMonth() + 1
        ).padStart(2, "0")
    );

}


async function loadMonthlySummary(showLoading = true) {

    const filter =
        $("monthlyFilter");

    if (!filter) {
        return;
    }


    if (!filter.value) {

        filter.value =
            getDefaultMonth();

    }


    const monthValue =
        filter.value;


    if (!monthValue) {
        return;
    }


    const [
        year,
        month
    ] = monthValue.split("-");


    if (showLoading) {

        $("monthlyBody").innerHTML = `
            <tr>
                <td colspan="6" class="loading">
                    Loading monthly summary...
                </td>
            </tr>
        `;

    }


    try {

        /*
         * Ask the backend for the selected month.
         * This is NOT restricted to the current month.
         */

        const result =
            await apiRequest(
                "monthly_summary",
                {
                    year: Number(year),
                    month: Number(month)
                }
            );


        const data =
            result.data ||
            result.summary ||
            [];


        adminData.monthlySummary =
            Array.isArray(data)
                ? data
                : [];


        renderMonthlySummary();

    }
    catch (error) {

        /*
         * If the backend does not return a separate
         * monthly-summary structure, calculate it from
         * approved submissions locally.
         */

        console.warn(
            "Monthly API unavailable; using local calculation.",
            error
        );


        calculateMonthlySummaryLocally(
            Number(year),
            Number(month)
        );


        renderMonthlySummary();

    }

}


function calculateMonthlySummaryLocally(
    year,
    month
) {

    const staffMap =
        new Map();


    const approved =
        adminData.submissions.filter(row => {

            if (
                getStatus(row) !==
                "approved"
            ) {
                return false;
            }


            const date =
                new Date(
                    getDate(row)
                );


            if (
                Number.isNaN(
                    date.getTime()
                )
            ) {
                return false;
            }


            return (
                date.getFullYear() === year &&
                date.getMonth() + 1 === month
            );

        });


    approved.forEach(row => {

        const smc =
            String(getSMC(row));

        if (!smc) {
            return;
        }


        if (!staffMap.has(smc)) {

            staffMap.set(
                smc,
                {
                    staff_name:
                        getStaff(row),

                    sales_man_code:
                        smc,

                    outlet:
                        getOutlet(row),

                    individual: 0,

                    bundle: 0,

                    total: 0
                }
            );

        }


        const entry =
            staffMap.get(smc);

        const amount =
            getAmount(row);


        if (
            getType(row) ===
            "bundle"
        ) {

            entry.bundle += amount;

        }
        else {

            entry.individual += amount;

        }


        entry.total =
            entry.individual +
            entry.bundle;

    });


    adminData.monthlySummary =
        Array.from(
            staffMap.values()
        );

}


function renderMonthlySummary() {

    const rows =
        adminData.monthlySummary || [];


    let individualTotal = 0;

    let bundleTotal = 0;


    rows.forEach(row => {

        individualTotal +=
            numberValue(
                row.individual ??
                row.individual_total ??
                row.individual_incentive ??
                0
            );

        bundleTotal +=
            numberValue(
                row.bundle ??
                row.bundle_total ??
                row.bundle_incentive ??
                0
            );

    });


    let total =
        individualTotal +
        bundleTotal;


    /*
     * If the backend returns a total row instead of
     * individual staff rows, also support it.
     */

    const backendTotal =
        rows.find(
            row =>
                row.total !== undefined &&
                (
                    row.smc === "TOTAL" ||
                    row.sales_man_code === "TOTAL" ||
                    row.is_total
                )
        );


    if (backendTotal) {

        total =
            numberValue(
                backendTotal.total
            );

    }


    $("monthlyTotal").textContent =
        money(total);

    $("monthlyIndividualTotal").textContent =
        money(individualTotal);

    $("monthlyBundleTotal").textContent =
        money(bundleTotal);


    const body =
        $("monthlyBody");


    if (!rows.length) {

        body.innerHTML = `
            <tr>
                <td colspan="6" class="empty">
                    No approved incentives for the selected month.
                </td>
            </tr>
        `;

        return;
    }


    body.innerHTML =
        rows.map(row => {

            const staff =
                row.staff_name ??
                row.name ??
                "";

            const smc =
                row.sales_man_code ??
                row.SMC ??
                row.smc ??
                "";

            const outlet =
                row.outlet ??
                row.outlet_name ??
                "";

            const individual =
                numberValue(
                    row.individual ??
                    row.individual_total ??
                    row.individual_incentive ??
                    0
                );

            const bundle =
                numberValue(
                    row.bundle ??
                    row.bundle_total ??
                    row.bundle_incentive ??
                    0
                );

            const rowTotal =
                row.total !== undefined
                    ? numberValue(row.total)
                    : individual + bundle;


            return `
                <tr>

                    <td>
                        ${escapeHtml(staff)}
                    </td>

                    <td>
                        ${escapeHtml(smc)}
                    </td>

                    <td>
                        ${escapeHtml(outlet)}
                    </td>

                    <td class="money">
                        ${money(individual)}
                    </td>

                    <td class="money">
                        ${money(bundle)}
                    </td>

                    <td class="money">
                        ${money(rowTotal)}
                    </td>

                </tr>
            `;

        }).join("");

}


/* ================================================================
   TOP 3
================================================================ */

function renderTop3() {

    const grid =
        $("top3Grid");

    if (!grid) {
        return;
    }


    let rows =
        adminData.top3 || [];


    /*
     * Fallback calculation from approved submissions.
     */

    if (!rows.length) {

        const staffMap =
            new Map();


        adminData.submissions
            .filter(
                row =>
                    getStatus(row) ===
                    "approved"
            )
            .forEach(row => {

                const smc =
                    getSMC(row);

                if (!smc) {
                    return;
                }


                if (!staffMap.has(smc)) {

                    staffMap.set(
                        smc,
                        {
                            staff_name:
                                getStaff(row),

                            sales_man_code:
                                smc,

                            outlet:
                                getOutlet(row),

                            total: 0,

                            submissions: 0
                        }
                    );

                }


                const entry =
                    staffMap.get(smc);

                entry.total +=
                    getAmount(row);

                entry.submissions++;

            });


        rows =
            Array.from(
                staffMap.values()
            )
            .sort(
                (a, b) =>
                    numberValue(b.total) -
                    numberValue(a.total)
            )
            .slice(0, 3);

    }


    const cards = [];


    for (
        let index = 0;
        index < 3;
        index++
    ) {

        const row =
            rows[index];


        if (!row) {

            cards.push(`
                <div class="top-card top-empty">

                    <div class="rank-circle">
                        ${index + 1}
                    </div>

                    <div class="place-label">
                        ${index === 0 ? "First" :
                          index === 1 ? "Second" :
                          "Third"}
                    </div>

                    <div class="top-name">
                        No Data
                    </div>

                    <div class="top-meta">
                        -
                    </div>

                    <div class="top-total">

                        <div class="top-total-label">
                            Approved Total
                        </div>

                        <div class="top-total-value">
                            $0.00
                        </div>

                    </div>

                </div>
            `);

            continue;
        }


        const name =
            row.staff_name ??
            row.name ??
            "";

        const smc =
            row.sales_man_code ??
            row.SMC ??
            row.smc ??
            "";

        const outlet =
            row.outlet ??
            row.outlet_name ??
            "";

        const total =
            numberValue(
                row.total ??
                row.amount ??
                row.incentive_total ??
                0
            );

        const submissions =
            numberValue(
                row.submissions ??
                row.submission_count ??
                row.count ??
                0
            );


        const classes = [
            "first",
            "second",
            "third"
        ];

        const labels = [
            "First",
            "Second",
            "Third"
        ];


        cards.push(`

            <div class="top-card ${classes[index]}">

                <div class="rank-circle">
                    ${index + 1}
                </div>

                <div class="place-label">
                    ${labels[index]}
                </div>

                <div class="top-name">
                    ${escapeHtml(name)}
                </div>

                <div class="top-meta">
                    ${escapeHtml(smc)}
                    ${outlet
                        ? " • " +
                          escapeHtml(outlet)
                        : ""}
                </div>

                <div class="top-stat">

                    <span>
                        Approved submissions
                    </span>

                    <span>
                        ${submissions}
                    </span>

                </div>

                <div class="top-total">

                    <div class="top-total-label">
                        Approved Total
                    </div>

                    <div class="top-total-value">
                        ${money(total)}
                    </div>

                </div>

            </div>

        `);

    }


    grid.innerHTML =
        cards.join("");

}

function switchTab(tabId) {

    document
        .querySelectorAll(".tab")
        .forEach(
            tab =>
                tab.classList.remove(
                    "active"
                )
        );


    document
        .querySelectorAll(".tab-btn")
        .forEach(
            button =>
                button.classList.remove(
                    "active"
                )
        );


    const tab =
        $(tabId);

    if (tab) {

        tab.classList.add(
            "active"
        );

    }


    const button =
        document.querySelector(
            `.tab-btn[data-tab="${tabId}"]`
        );


    if (button) {

        button.classList.add(
            "active"
        );

    }


    if (tabId === "monthlyTab") {

        loadMonthlySummary();

    }

}


document
    .querySelectorAll(".tab-btn")
    .forEach(button => {

        button.addEventListener(
            "click",
            () => {

                switchTab(
                    button.dataset.tab
                );

            }
        );

    });
