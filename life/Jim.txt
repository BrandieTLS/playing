async function apiRequest(action, data = {}) {

    const response = await fetch(
        `admin.php?action=${encodeURIComponent(action)}`,
        {
            method: "POST",

            credentials: "same-origin",

            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },

            body: JSON.stringify(data)
        }
    );

    const text = await response.text();

    let json;

    try {

        json = text
            ? JSON.parse(text)
            : {};

    } catch (e) {

        throw new Error(
            text ||
            `Server returned HTTP ${response.status}`
        );

    }

    if (
        !response.ok ||
        json.success === false
    ) {

        throw new Error(
            json.message ||
            json.error ||
            `Request failed (${response.status})`
        );

    }

    return json;
}

function escapeHtml(value) {

    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}


function money(value) {

    const number = Number(value) || 0;

    return `S$${number.toFixed(2)}`;
}


function formatDate(value) {

    if (!value) {
        return "-";
    }

    const str = String(value);

    const match = str.match(
        /^(\d{4}-\d{2}-\d{2})/
    );

    if (match) {
        return match[1];
    }

    return str;
}


function normaliseDateInput(value) {

    if (!value) {
        return "";
    }

    const str = String(value).trim();

    if (
        /^\d{4}-\d{2}-\d{2}$/.test(str)
    ) {
        return str;
    }

    const match = str.match(
        /^(\d{4}-\d{2}-\d{2})/
    );

    return match
        ? match[1]
        : "";
}


function isValidDateInput(value) {

    return /^\d{4}-\d{2}-\d{2}$/.test(
        String(value)
    );

}


function getSubmissionId(x) {

    const raw =
        x?.incentive_id ??
        x?.submission_id ??
        x?.id ??
        x?.incentiveId;

    if (

        raw === undefined ||
        String(raw).trim() === ""
    ) {
        return null;
    }

    const n = Number(raw);

    return Number.isFinite(n) && n > 0
        ? n
        : null;
}

function getType(x) {

    const direct = String(
        x?.type ??
        x?.promotion_type ??
        x?.promo_type ??
        x?.promotionType ??
        ""
    ).trim();

    if (direct) {
        return direct;
    }

    const promoId = getPromoId(x);

    if (promoId !== null) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {

            return String(
                promo?.type ??
                promo?.promotion_type ??
                promo?.promo_type ??
                ""
            ).trim();

        }

    }

    return "";
}


function isBundle(x) {

    return getType(x)
        .toLowerCase()
        .replace(/[\s_-]/g, "") === "bundle";

}


function displayType(x) {

    const type = getType(x);

    if (isBundle(x)) {
        return "Bundle";
    }

    if (type) {
        return type;
    }

    return "Individual";
}


function getPromotionStatus(promo) {

    const directStatus = String(
        promo?.status ??
        promo?.promotion_status ??
        ""
    ).trim();

    if (directStatus) {
        return directStatus;
    }

    const today = new Date();

    today.setHours(
        0,
        0,
        0,
        0
    );

    const start = promo?.start_date
        ? new Date(
            `${normaliseDateInput(promo.start_date)}T00:00:00`
        )
        : null;

    const end = promo?.end_date
        ? new Date(
            `${normaliseDateInput(promo.end_date)}T23:59:59`
        )
        : null;

    if (
        start &&
        !isNaN(start.getTime()) &&
        today < start
    ) {
        return "Upcoming";
    }

    if (
        end &&
        !isNaN(end.getTime()) &&
        today > end
    ) {
        return "Expired";
    }

    return "Active";
}


function promotionStatusClass(status) {

    const value =
        String(status)
            .toLowerCase();

    if (value === "active") {
        return "status status-active";
    }

    if (value === "upcoming") {
        return "status status-upcoming";
    }

    if (
        value === "expired" ||
        value === "inactive"
    ) {
        return "status status-expired";
    }

    return "status status-default";
}

function closeModal(id) {

    const modal =
        document.getElementById(id);

    if (modal) {
        modal.classList.remove("show");
    }

}


/* =====================================================
   TAB SYSTEM
===================================================== */

function activateTab(tabId) {

    state.activeTab = tabId;

    document
        .querySelectorAll(".tab")
        .forEach(tab => {

            tab.classList.toggle(
                "active",
                tab.id === tabId
            );

        });


    document
        .querySelectorAll(".tab-btn")
        .forEach(button => {

            button.classList.toggle(
                "active",
                button.dataset.tab === tabId
            );

        });


    if (tabId === "summaryTab") {
        loadMonthlySummary();
    }

}


document
    .querySelectorAll(".tab-btn")
    .forEach(button => {

        button.addEventListener(
            "click",
            () => {

                activateTab(
                    button.dataset.tab
                );

            }
        );

    });

            ? d.submissions

    state.promotions =
        Array.isArray(d.promotions)
            ? d.promotions
            : [];

        Array.isArray(d.users)
            ? d.users

        Array.isArray(d.monthlySummary)
            ? d.monthlySummary
            : [];

    state.top3 =
        Array.isArray(d.top3)
            ? d.top3
            : [];


    const adminName =
        state.admin?.staff_name ??
        state.admin?.name ??
        state.admin?.sales_man_code ??
        "Administrator";

    document.getElementById(
        "adminName"
    ).textContent = adminName;

}


/* =====================================================
   REFRESH
===================================================== */

async function refreshAdminData() {

    if (refreshInProgress) {
        return;
    }

    refreshInProgress = true;

    try {

        await loadAdminData();

        renderAll();

        populateSubmissionFilters();

        populateUserFilters();

        if (
            state.currentSubmissionId !== null
        ) {

            const exists =
                state.submissions.some(
                    s =>
                        getSubmissionId(s) ===
                        state.currentSubmissionId
                );

            if (!exists) {

                state.currentSubmissionId =
                    null;

                closeModal(
                    "viewModal"
                );

            }

        }

        activateTab(
            state.activeTab
        );

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to refresh data.",
            "error"
        );

    } finally {

        refreshInProgress = false;

    }

}

===================================================== */

function calculateApprovedTotals() {

    let individual = 0;

    let bundle = 0;

    let individualQty = 0;

    let bundleQty = 0;

    let individualCount = 0;

    let bundleCount = 0;

    let pending = 0;

    let approved = 0;

    let rejected = 0;


    state.submissions.forEach(row => {

        const status =
            getStatus(row)
                .toLowerCase()
                .trim();

        const qty =
            getQty(row);

        const rate =
            getRate(row);

        const amount =
            qty * rate;


        if (status === "pending") {
            pending++;
        }

        else if (status === "approved") {

            approved++;

            if (isBundle(row)) {

                bundle += amount;

                bundleQty += qty;

                bundleCount++;

            } else {

                individual += amount;

                individualQty += qty;

                individualCount++;

            }

        }

        else if (status === "rejected") {
            rejected++;
        }

    });


    return {
        individual,
        bundle,
        total: individual + bundle,
        individualQty,
        bundleQty,
        individualCount,
        bundleCount,
        pending,
        approved,
        rejected
    };

}

/* =====================================================
   SUBMISSION FILTERS
===================================================== */

function populateSubmissionFilters() {

    const outletSelect =
        document.getElementById(
            "submissionOutletFilter"
        );

    const staffSelect =
        document.getElementById(
            "submissionStaffFilter"
        );


    const currentOutlet =
        outletSelect.value;

    const currentStaff =
        staffSelect.value;


    const outlets =
        [
            ...new Set(
                state.submissions
                    .map(getOutlet)
                    .filter(Boolean)
            )
        ]
        .sort();


    const staff =
        [
            ...new Map(
                state.submissions
                    .map(row => {

                        const smc =
                            getSMC(row);

                        const name =
                            getStaff(row);

                        return [
                            smc ||
                            name,
                            {
                                smc,
                                name
                            }
                        ];

                    })
                    .filter(([key]) => key)
            ).values()
        ]
        .sort(
            (a, b) =>
                a.name.localeCompare(
                    b.name
                )
        );


    outletSelect.innerHTML = `
        <option value="">
            All Outlets
        </option>

        ${outlets.map(outlet => `
            <option value="${escapeHtml(outlet)}">
                ${escapeHtml(outlet)}
            </option>
        `).join("")}
    `;


    staffSelect.innerHTML = `
        <option value="">
            All Staff
        </option>

        ${staff.map(person => `
            <option value="${escapeHtml(person.smc || person.name)}">
                ${escapeHtml(
                    person.name
                    ? `${person.name}${person.smc ? ` (${person.smc})` : ""}`
                    : person.smc
                )}
            </option>
        `).join("")}
    `;


    if (
        outlets.includes(currentOutlet)
    ) {
        outletSelect.value =
            currentOutlet;
    }

    if (
        staff.some(
            p =>
                (p.smc || p.name) ===
                currentStaff
        )
    ) {
        staffSelect.value =
            currentStaff;
    }

}


function getFilteredSubmissions() {

    const search =
        document.getElementById(
            "submissionSearch"
        ).value
        .trim()
        .toLowerCase();

    const status =
        document.getElementById(
            "submissionStatusFilter"
        ).value
        .toLowerCase();

    const type =
        document.getElementById(
            "submissionTypeFilter"
        ).value
        .toLowerCase();

    const outlet =
        document.getElementById(
            "submissionOutletFilter"
        ).value;

    const staff =
        document.getElementById(
            "submissionStaffFilter"
        ).value;

    const date =
        document.getElementById(
            "submissionDateFilter"
        ).value;


    return state.submissions.filter(row => {

        const id =
            getSubmissionId(row);

        const rowStaff =
            getStaff(row);

        const smc =
            getSMC(row);

        const rowOutlet =
            getOutlet(row);

        const promo =
            getPromo(row);

        const rowStatus =
            getStatus(row)
                .toLowerCase();

        const rowType =
            displayType(row)
                .toLowerCase();

        const rowDate =
            getDate(row);


        if (search) {

            const haystack =
                [
                    id,
                    rowStaff,
                    smc,
                    rowOutlet,
                    promo,
                    rowStatus,
                    rowType
                ]
                .join(" ")
                .toLowerCase();

            if (!haystack.includes(search)) {
                return false;
            }

        }


        if (
            status &&
            rowStatus !== status
        ) {
            return false;
        }


        if (
            type &&
            rowType !== type
        ) {
            return false;
        }


        if (
            outlet &&
            rowOutlet !== outlet
        ) {
            return false;
        }


        if (staff) {

            const staffValue =
                smc || rowStaff;

            if (
                staffValue !== staff
            ) {
                return false;
            }

        }


        if (date) {

            if (
                !rowDate.startsWith(date)
            ) {
                return false;
            }

        }


        return true;

    });

}

/* =====================================================
   VIEW SUBMISSION
===================================================== */

async function showSubmission(id) {

    const numericId =
        Number(id);

    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    state.currentSubmissionId =
        numericId;


    let row =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    try {

        const response =
            await apiRequest(
                "get_submission",
                {
                    incentive_id:
                        numericId,

                    submission_id:
                        numericId
                }
            );


        const serverData =
            response.data ||
            response.submission ||
            response;


        if (
            serverData &&
            !Array.isArray(serverData) &&
            (
                serverData.incentive_id ||
                serverData.submission_id ||
                serverData.id
            )
        ) {
            row = {
                ...row,
                ...serverData
            };
        }

    } catch (err) {

        /*
         * If the backend cannot fetch the single row,
         * still use the row already loaded by bootstrap.
         */

        if (!row) {

            showMessage(
                err.message ||
                "Unable to load submission.",
                "error"
            );

            return;
        }

    }


    if (!row) {

        showMessage(
            "Submission not found.",
            "error"
        );

        return;
    }


    renderSubmissionDetails(row);

    openModal("viewModal");

}

function getSafeFileUrl(value) {

    let url =
        String(value || "").trim();

    if (!url) {
        return "";
    }


    if (
        url.toLowerCase()
            .startsWith("javascript:")
    ) {
        return "";
    }


    if (
        url.startsWith("http://") ||
        url.startsWith("https://") ||
        url.startsWith("/")
    ) {
        return url;
    }


    url =
        url.replace(/^\.?\//, "");

    return url;

}


/* =====================================================
   APPROVE
===================================================== */

async function approveSubmission(id) {

    const numericId =
        Number(id);

    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    const current =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    if (
        current &&
        getStatus(current)
            .toLowerCase() !==
            "pending"
    ) {

        showMessage(
            "Only pending submissions can be approved.",
            "error"
        );

        return;
    }


    if (
        !confirm(
            "Approve this submission?"
        )
    ) {
        return;
    }


    try {

        await apiRequest(
            "approve",
            {
                incentive_id:
                    numericId
            }
        );


        showMessage(
            "Submission approved.",
            "success"
        );


        closeModal(
            "viewModal"
        );


        await refreshAdminData();

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to approve submission.",
            "error"
        );

    }

}


/* =====================================================
   REJECT
===================================================== */

async function rejectSubmission(id) {

    const numericId =
        Number(id);

    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    const current =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    if (
        current &&
        getStatus(current)
            .toLowerCase() !==
            "pending"
    ) {

        showMessage(
            "Only pending submissions can be rejected.",
            "error"
        );

        return;
    }


    const reason =
        window.prompt(
            "Enter rejection reason:"
        );


    if (reason === null) {
        return;
    }


    const trimmed =
        reason.trim();


    if (!trimmed) {

        showMessage(
            "Rejection reason is required.",
            "error"
        );

        return;
    }


    try {

        await apiRequest(
            "reject",
            {
                incentive_id:
                    numericId,

                rejection_reason:
                    trimmed,

                reason:
                    trimmed
            }
        );


        showMessage(
            "Submission rejected.",
            "success"
        );


        closeModal(
            "viewModal"
        );


        await refreshAdminData();

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to reject submission.",
            "error"
        );

    }

}

            x =>
                getSubmissionId(x) ===
                numericId
        );


    if (!row) {

        showMessage(
            "Submission not found.",
            "error"
        );

        return;
    }


    document.getElementById(
        "editSubmissionId"
    ).value = numericId;


    document.getElementById(
        "editSubmissionQuantity"
    ).value =
        getQty(row);


    document.getElementById(
        "editSubmissionStatus"
    ).value =
        getStatus(row)
            .toLowerCase() ||
        "pending";


    document.getElementById(
        "editSubmissionReason"
    ).value =
        row?.rejection_reason ??
        row?.reason ??
        "";


    const select =
        document.getElementById(
            "editSubmissionPromo"
        );


    select.innerHTML =
        state.promotions.map(promo => {

            const promoId =
                getPromoId(promo);

            const selected =
                promoId ===
                getPromoId(row)
                    ? "selected"
                    : "";

            return `
                <option
                    value="${promoId}"
                    ${selected}>

                    ${escapeHtml(
                        getPromo(promo)
                    )}

                    -
                    ${escapeHtml(
                        displayType(promo)
                    )}

                    (${money(
                        getRate(promo)
                    )})

                </option>
            `;

        }).join("");


    closeModal(
        "viewModal"
    );

    openModal(
        "editSubmissionModal"
    );

}

function getFilteredPromotions() {

    const search =
        document.getElementById(
            "promotionSearch"
        ).value
        .trim()
        .toLowerCase();


    const type =
        document.getElementById(
            "promotionTypeFilter"
        ).value
        .toLowerCase();


    const status =
        document.getElementById(
            "promotionStatusFilter"
        ).value
        .toLowerCase();


    return state.promotions.filter(promo => {

        const name =
            getPromo(promo)
                .toLowerCase();

        const promoType =
            displayType(promo)
                .toLowerCase();

        const promoStatus =
            getPromotionStatus(promo)
                .toLowerCase();


        if (
            search &&
            !name.includes(search)
        ) {
            return false;
        }


        if (
            type &&
            promoType !== type
        ) {
            return false;
        }


        if (
            status &&
            promoStatus !== status
        ) {
            return false;
        }


        return true;

    });

}


function renderPromotions() {

    const tbody =
        document.getElementById(
            "promotionsTableBody"
        );


    const rows =
        getFilteredPromotions();


    if (!rows.length) {

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="8"
                    class="empty">
                    No promotions found.
                </td>
            </tr>
        `;

        return;
    }


    tbody.innerHTML =
        rows.map(promo => {

            const promoId =
                getPromoId(promo);

            const name =
                getPromo(promo);

            const type =
                displayType(promo);

            const rate =
                getRate(promo);

            const status =
                getPromotionStatus(promo);

            const startDate =
                promo?.start_date
                    ? formatDate(
                        promo.start_date
                    )
                    : "-";

            const endDate =
                promo?.end_date
                    ? formatDate(
                        promo.end_date
                    )
                    : "-";


            const items = [];


            for (
                let i = 1;
                i <= 5;
                i++
            ) {

                const value =
                    String(
                        promo?.[
                            `item_${i}`
                        ] ??
                        ""
                    ).trim();


                if (value) {

                    items.push(`
                        <span class="item-tag">
                            ${escapeHtml(value)}
                        </span>
                    `);

                }

            }


            const bundle =
                isBundle(promo);


            return `

                <tr>

                    <td>
                        ${promoId ?? "-"}
                    </td>


                    <td>
                        <strong>
                            ${escapeHtml(name)}
                        </strong>
                    </td>


                    <td>

                        <span class="type-badge ${
                            bundle
                                ? "type-bundle"
                                : ""
                        }">

                            ${escapeHtml(type)}

                        </span>

                    </td>


                    <td class="money">
                        ${money(rate)}
                    </td>


                    <td>

                        <span class="${
                            promotionStatusClass(
                                status
                            )
                        }">

                            ${escapeHtml(
                                status
                            )}

                        </span>

                    </td>


                    <td>
                        ${startDate}
                        →
                        ${endDate}
                    </td>


                    <td>

                        ${
                            items.length
                                ? `
                                    <div class="item-tags">
                                        ${items.join("")}
                                    </div>
                                `
                                : "-"
                        }

                    </td>


                    <td class="action-column">

                        ${
                            promoId !== null
                                ? `
                                    <div class="button-row">

                                        <button
                                            type="button"
                                            class="btn btn-primary btn-small edit-promo-btn"
                                            data-id="${promoId}">
                                            Edit
                                        </button>

                                        <button
                                            type="button"
                                            class="btn btn-danger btn-small delete-promo-btn"
                                            data-id="${promoId}">
                                            Delete
                                        </button>

                                    </div>
                                `
                                : `
                                    <span class="status status-rejected">
                                        Invalid ID
                                    </span>
                                `
                        }

                    </td>

                </tr>

            `;

        }).join("");

}

        `).join("")}

    `;


    if (
        outlets.includes(current)
    ) {

    }

}


function getFilteredUsers() {

    const search =
        document.getElementById(
            "userSearch"
        ).value
        .trim()
        .toLowerCase();


    const role =
        document.getElementById(
            "userRoleFilter"
        ).value
        .toLowerCase();


    const outlet =
        document.getElementById(
            "userOutletFilter"
        ).value;


    return state.users.filter(user => {

        const name =
            getStaff(user);

        const smc =
            getSMC(user);

        const rowOutlet =
            getOutlet(user);

        const rowRole =
            String(
                user?.role ??
                ""
            ).toLowerCase();


        if (search) {

            const haystack =
                [
                    name,
                    smc,
                    rowOutlet,
                    rowRole
                ]
                .join(" ")
                .toLowerCase();


            if (
                !haystack.includes(
                    search
                )
            ) {
                return false;
            }

        }


        if (
            role &&
            rowRole !== role
        ) {
            return false;
        }


        if (
            outlet &&
            rowOutlet !== outlet
        ) {
            return false;
        }


        return true;

    });

}

                );

                user?.SMC ??

                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            getSMC(user)
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            getOutlet(user)
                        )}
                    </td>

                    <td>

                        <span class="${
                            role.toLowerCase() ===
                            "admin"
                                ? "status status-approved"
                                : "status status-default"
                        }">

                            ${escapeHtml(
                                role
                            )}

                        </span>

                    </td>

                    <td class="action-column">

                        <div class="button-row">

                            <button
                                type="button"
                                class="btn btn-warning btn-small"
                                onclick='editUser(${JSON.stringify(user)})'>
                                Edit
                            </button>

                            <button
                                type="button"
                                class="btn btn-danger btn-small"
                                onclick='deleteUser("${escapeJs(id)}")'>
                                Delete
                            </button>

                        </div>

                    </td>

                </tr>

            `;

        }).join("");

}


function escapeJs(value) {

    return String(value ?? "")
        .replace(/\\/g, "\\\\")
        .replace(/"/g, '\\"')
        .replace(/'/g, "\\'");
}

===================================================== */

function getDefaultMonth() {

    const now =
        new Date();


    const year =
        now.getFullYear();


    const month =
        String(
            now.getMonth() + 1
        ).padStart(
            2,
            "0"
        );


    return `${year}-${month}`;

}


function calculateMonthlySummary(
    year,
    month
) {

    const prefix =
        `${year}-${String(month).padStart(2, "0")}`;


    const grouped =
        new Map();


    let individual =
        0;

    let bundle =
        0;


    state.submissions.forEach(row => {

        const status =
            getStatus(row)
                .toLowerCase();


        if (
            status !== "approved"
        ) {
            return;
        }


        const date =
            getDate(row);


        if (
            !date.startsWith(prefix)
        ) {
            return;
        }


        const staff =
            getStaff(row);

        const smc =
            getSMC(row);

        const outlet =
            getOutlet(row);

        const promo =
            getPromo(row);

        const type =
            displayType(row);

        const qty =
            getQty(row);

        const rate =
            getRate(row);

        const amount =
            qty * rate;


        const key =
            [
                smc,
                getPromoId(row)
            ].join("|");


        if (!grouped.has(key)) {

            grouped.set(
                key,
                {
                    staff,
                    smc,
                    outlet,
                    promo,
                    type,
                    quantity: 0,
                    rate,
                    amount: 0
                }
            );

        }


        const item =
            grouped.get(key);


        item.quantity +=
            qty;

        item.amount +=
            amount;


        if (isBundle(row)) {
            bundle += amount;
        } else {
            individual += amount;
        }

    });


    return {
        rows:
            [...grouped.values()],

        individual,

        bundle,

        total:
            individual + bundle
    };

}


async function loadMonthlySummary() {

    const input =
        document.getElementById(
            "summaryMonth"
        );


    if (!input.value) {
        input.value =
            getDefaultMonth();
    }


    const [year, month] =
        input.value.split("-");


    if (
        !year ||
        !month
    ) {
        return;
    }


    try {

        const response =
            await apiRequest(
                "monthly_summary",
                {
                    year:
                        Number(year),

                    month:
                        Number(month)
                }
            );


        const d =
            response.data ||
            response;


        if (
            Array.isArray(
                d.rows
            )
        ) {

            state.monthlySummary =
                d.rows;

        }

        else if (
            Array.isArray(
                d.monthlySummary
            )
        ) {

            state.monthlySummary =
                d.monthlySummary;

        }


        renderMonthlySummary(
            d
        );

    } catch (err) {

        /*
         * Local fallback using already loaded submissions.
         */

        const fallback =
            calculateMonthlySummary(
                Number(year),
                Number(month)
            );


        state.monthlySummary =
            fallback.rows;


        renderMonthlySummary(
            fallback
        );

    }

}


function renderMonthlySummary(
    backendData = null
) {

    const input =
        document.getElementById(
            "summaryMonth"
        );


    if (!input.value) {
        input.value =
            getDefaultMonth();
    }


    const [year, month] =
        input.value.split("-");


    let calculated =
        calculateMonthlySummary(
            Number(year),
            Number(month)
        );


    if (
        backendData &&
        !Array.isArray(backendData) &&
        (
            backendData.individual !== undefined ||
            backendData.bundle !== undefined ||
            backendData.total !== undefined
        )
    ) {

        calculated.individual =
            Number(
                backendData.individual ??
                calculated.individual
            );

        calculated.bundle =
            Number(
                backendData.bundle ??
                calculated.bundle
            );

        calculated.total =
            Number(
                backendData.total ??
                (
                    calculated.individual +
                    calculated.bundle
                )
            );

    }


    document.getElementById(
        "summaryIndividual"
    ).textContent =
        money(
            calculated.individual
        );


    document.getElementById(
        "summaryBundle"
    ).textContent =
        money(
            calculated.bundle
        );


    document.getElementById(
        "summaryTotal"
    ).textContent =
        money(
            calculated.total
        );


    const tbody =
        document.getElementById(
            "monthlySummaryBody"
        );


    const rows =
        calculated.rows;


    if (!rows.length) {

        tbody.innerHTML = `
            <tr>
                <td
                    colspan="8"
                    class="empty">
                    No approved incentives for this month.
                </td>
            </tr>
        `;

        return;
    }


    tbody.innerHTML =
        rows.map(row => {

            return `

                <tr>

                    <td>
                        ${escapeHtml(
                            row.staff
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            row.smc
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            row.outlet
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            row.promo
                        )}
                    </td>

                    <td>

                        <span class="type-badge ${
                            String(row.type)
                                .toLowerCase()
                                .replace(/[\s_-]/g, "") ===
                            "bundle"
                                ? "type-bundle"
                                : ""
                        }">

                            ${escapeHtml(
                                row.type
                            )}

                        </span>

                    </td>

                    <td>
                        ${row.quantity}
                    </td>

                    <td class="money">
                        ${money(
                            row.rate
                        )}
                    </td>

                    <td class="money">
                        ${money(
                            row.amount
                        )}
                    </td>

                </tr>

            `;

        }).join("");

}

function calculateRanking() {

    const map =
        new Map();


    state.submissions.forEach(row => {

        if (
            getStatus(row)
                .toLowerCase() !==
            "approved"
        ) {
            return;
        }


        const smc =
            getSMC(row);

        const staff =
            getStaff(row);

        const outlet =
            getOutlet(row);

        const qty =
            getQty(row);

        const amount =
            qty *
            getRate(row);


        const key =
            smc ||
            staff;


        if (!key) {
            return;
        }


        if (!map.has(key)) {

            map.set(
                key,
                {
                    staff,
                    smc,
                    outlet,
                    count: 0,
                    quantity: 0,
                    amount: 0
                }
            );

        }


        const item =
            map.get(key);


        item.count++;

        item.quantity +=
            qty;

        item.amount +=
            amount;

    });


    return [
        ...map.values()
    ].sort(
        (a, b) =>
            b.amount -
            a.amount
    );

}
