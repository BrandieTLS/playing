function escapeHtml(value) {

    if (value === null || value === undefined) {
        return "";
    }

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}


function money(value) {

    const number = Number(value);

    if (!Number.isFinite(number)) {
        return "$0.00";
    }

    return "$" + number.toFixed(2);
}


function getNumber(value) {

    const number = Number(value);

    return Number.isFinite(number) ? number : 0;
}


function todayString() {

    const d = new Date();

    const year = d.getFullYear();

    const month = String(d.getMonth() + 1).padStart(2, "0");

    const day = String(d.getDate()).padStart(2, "0");

    return `${year}-${month}-${day}`;
}


function getDateOnly(value) {

    if (!value) {
        return "";
    }

    const stringValue = String(value);

    return stringValue.substring(0, 10);
}


/* ================================================================
   ID HELPERS
================================================================ */

function getPromoId(x) {

    const raw =
        x?.promo_id ??
        x?.promotion_id ??
        x?.promoId ??
        x?.id;

    const n = Number(raw);

    return Number.isFinite(n) ? n : null;
}


function getSubmissionId(x) {

    const raw =
        x?.incentive_id ??
        x?.submission_id ??
        x?.id ??
        x?.incentiveId;

    if (
        raw === null ||
        raw === undefined ||
        String(raw).trim() === ""
    ) {
        return null;
    }

    const n = Number(raw);

    return Number.isFinite(n) && n > 0 ? n : null;
}


/* ================================================================
   PROMOTION TYPE
================================================================ */

/*
   IMPORTANT:

   Submission rows may not contain "type".

   Therefore we first find the promotion using promo_id.

   Example:

   staff_incentives
       promo_id = 5

   Incentives_information
       promo_id = 5
       type = Bundle

   Therefore submission = Bundle.

   This prevents Bundle / outlet_bundle from being incorrectly
   counted as Individual.
*/

function getType(x) {

    const direct = String(
        x?.type ??
        x?.promotion_type ??
        x?.promo_type ??
        x?.promotionType ??
        ""
    ).trim();

    const promoId = getPromoId(x);

    if (promoId !== null) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {

            const promoType = String(
                promo?.type ??
                promo?.promotion_type ??
                promo?.promo_type ??
                promo?.promotionType ??
                ""
            ).trim();

            if (promoType) {
                return promoType;
            }
        }
    }

    if (direct) {
        return direct;
    }

    return "";
}


function normalizedType(x) {

    return getType(x)
        .toLowerCase()
        .replace(/[\s_-]/g, "");
}


function isBundle(x) {

    const type = normalizedType(x);

    /*
       Both of these are treated as Bundle:

       Bundle
       outlet_bundle
       outlet-bundle
       outlet bundle
    */

    return (
        type === "bundle" ||
        type === "outletbundle"
    );
}


function displayType(x) {

    if (isBundle(x)) {
        return "Bundle";
    }

    const type = getType(x);

    return type || "Individual";
}


/* ================================================================
   PROMOTION STATUS
================================================================ */

function getPromotionStatus(p) {

    const direct = String(
        p?.status ??
        p?.promotion_status ??
        p?.promo_status ??
        ""
    ).trim();

    if (direct) {

        const n = direct.toLowerCase();

        if (n === "active") {
            return "Active";
        }

        if (n === "upcoming") {
            return "Upcoming";
        }

        if (
            n === "expired" ||
            n === "inactive"
        ) {
            return "Expired";
        }
    }

    const start = getDateOnly(
        p?.start_date ??
        p?.startDate
    );

    const end = getDateOnly(
        p?.end_date ??
        p?.endDate
    );

    const today = todayString();

    if (start && today < start) {
        return "Upcoming";
    }

    if (end && today > end) {
        return "Expired";
    }

    return "Active";
}


function statusBadge(status) {

    const s = String(status || "").toLowerCase();

    let className = "status-default";

    if (
        s === "approved" ||
        s === "active"
    ) {
        className = "status-approved";
    }
    else if (
        s === "pending" ||
        s === "upcoming"
    ) {
        className = "status-pending";
    }
    else if (
        s === "rejected" ||
        s === "expired"
    ) {
        className = "status-rejected";
    }

    return `
        <span class="status ${className}">
            ${escapeHtml(status || "Unknown")}
        </span>
    `;
}

function getSubmissionAmount(x) {

    const quantity = getQuantity(x);

    let rate = getRate(x);

    const promoId = getPromoId(x);

    if (
        rate === 0 &&
        promoId !== null
    ) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {
            rate = getRate(promo);
        }
    }

    return quantity * rate;
}

async function apiRequest(action, data = {}) {

    const response = await fetch(
        `admin.php?action=${encodeURIComponent(action)}`,
        {
            method: "POST",
            credentials: "same-origin",
            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },
            body: JSON.stringify(data)
        }
    );

    const text = await response.text();

    let json;

    try {

        json = text
            ? JSON.parse(text)
            : {};

    }
    catch (e) {

        throw new Error(
            text ||
            `Server returned HTTP ${response.status}`
        );
    }

    if (
        !response.ok ||
        json.success === false
    ) {

        throw new Error(
            json.message ||
            json.error ||
            `Request failed (${response.status})`
        );
    }

    return json;
}

}

    if (refreshInProgress) {
        return;
    }

    refreshInProgress = true;

    try {

        await loadAdminData();

        renderAll();

        if (
            state.currentSubmissionId !== null
        ) {

            const exists =
                state.submissions.some(
                    s =>
                        getSubmissionId(s) ===
                        state.currentSubmissionId
                );

            if (!exists) {

                state.currentSubmissionId = null;

                closeModal("viewModal");
            }
        }

    }
    catch (err) {

        showMessage(
            err.message ||
            "Failed to refresh data.",
            "error"
        );

    }
    finally {

        refreshInProgress = false;
    }
}

function calculateApprovedTotals() {

    let individual = 0;

    let bundle = 0;

    let individualQty = 0;

    let bundleQty = 0;

    let approvedCount = 0;

    let pendingCount = 0;

    let rejectedCount = 0;


    for (const submission of state.submissions) {

        const status =
            getStatus(submission)
                .toLowerCase();

        const amount =
            getSubmissionAmount(submission);

        const quantity =
            getQuantity(submission);


        if (status === "approved") {

            approvedCount++;

            if (isBundle(submission)) {

                bundle += amount;

                bundleQty += quantity;

            }
            else {

                individual += amount;

                individualQty += quantity;

            }

        }
        else if (status === "pending") {

            pendingCount++;

        }
        else if (status === "rejected") {

            rejectedCount++;

        }
    }


    return {
        individual,
        bundle,
        total: individual + bundle,

        individualQty,
        bundleQty,

        approvedCount,
        pendingCount,
        rejectedCount
    };
}


function renderDashboard() {

    const totals =
        calculateApprovedTotals();


    document.getElementById(
        "dashboardTotal"
    ).textContent =
        money(totals.total);


    document.getElementById(
        "dashboardIndividual"

function populateOutletFilter() {

    const select =
        document.getElementById(
            "submissionOutlet"
        );

    const previous =
        select.value;


    const outlets =
        [...new Set(
            state.submissions
                .map(getOutlet)
                .filter(Boolean)
        )]
        .sort(
            (a, b) =>
                String(a).localeCompare(
                    String(b)
                )
        );


    select.innerHTML =
        `<option value="">All Outlets</option>`;


    outlets.forEach(outlet => {

        const option =
            document.createElement("option");

        option.value = outlet;

        option.textContent = outlet;

        select.appendChild(option);

    });


    if (
        outlets.includes(previous)
    ) {

        select.value = previous;

    }
}


function getFilteredSubmissions() {

    const search =
        document.getElementById(
            "submissionSearch"
        ).value
        .trim()
        .toLowerCase();


    const outlet =
        document.getElementById(
            "submissionOutlet"
        ).value;


    const status =
        document.getElementById(
            "submissionStatus"
        ).value
        .toLowerCase();


    return state.submissions.filter(
        submission => {

            const staff =
                getStaffName(submission)
                    .toLowerCase();

            const smc =
                String(
                    getSMC(submission)
                ).toLowerCase();

            const submissionOutlet =
                String(
                    getOutlet(submission)
                );

            const promotion =
                String(
                    getPromoName(submission)
                ).toLowerCase();

            const submissionStatus =
                getStatus(submission)
                    .toLowerCase();


            const matchesSearch =
                !search ||
                staff.includes(search) ||
                smc.includes(search) ||
                promotion.includes(search) ||
                submissionOutlet
                    .toLowerCase()
                    .includes(search);


            const matchesOutlet =
                !outlet ||
                submissionOutlet === outlet;


            const matchesStatus =
                !status ||
                submissionStatus === status;


            return (
                matchesSearch &&
                matchesOutlet &&
                matchesStatus
            );
        }
    );
}

function getRateForSubmission(submission) {

    let rate =
        getRate(submission);


    if (rate !== 0) {
        return rate;
    }


    const promoId =
        getPromoId(submission);


    if (promoId === null) {
        return 0;
    }


    const promotion =
        state.promotions.find(
            p =>
                getPromoId(p) ===
                promoId
        );


    return promotion
        ? getRate(promotion)
        : 0;
}

async function showSubmission(id) {

    const numericId =
        Number(id);


    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    state.currentSubmissionId =
        numericId;


    openModal("viewModal");


    document.getElementById(
        "submissionDetails"
    ).innerHTML = `
        <div class="loading">
            Loading submission...
        </div>
    `;


    document.getElementById(
        "submissionActions"
    ).innerHTML = "";


    const localSubmission =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    let submission =
        localSubmission;


    try {

        const response =
            await apiRequest(
                "get_submission",
                {
                    incentive_id:
                        numericId,

                    submission_id:
                        numericId
                }
            );


        const remote =
            response.data ||
            response.submission ||
            response;


        if (
            remote &&
            typeof remote === "object"
        ) {

            submission =
                remote.submission ||
                remote.data ||
                remote ||
                submission;

        }

    }
    catch (err) {

        /*
           If backend does not support get_submission
           or returns invalid ID, use the row already
           loaded from bootstrap.
        */

        if (!submission) {

            document.getElementById(
                "submissionDetails"
            ).innerHTML = `
                <div class="empty">
                    ${escapeHtml(
                        err.message ||
                        "Failed to load submission."
                    )}
                </div>
            `;

            return;
        }
    }


    if (!submission) {

        document.getElementById(
            "submissionDetails"
        ).innerHTML = `
            <div class="empty">
                Submission not found.
            </div>
        `;

        return;
    }


    renderSubmissionDetails(
        submission
    );
}

/* ================================================================
   REJECT
================================================================ */

async function rejectSubmission(id) {

    const numericId =
        Number(id);


    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    const current =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    if (
        current &&
        getStatus(current).toLowerCase() !==
        "pending"
    ) {

        showMessage(
            "Only pending submissions can be rejected.",
            "error"
        );

        return;
    }


    const reason =
        window.prompt(
            "Enter rejection reason:"
        );


    if (reason === null) {
        return;
    }


    const trimmed =
        reason.trim();


    if (!trimmed) {

        showMessage(
            "Rejection reason is required.",
            "error"
        );

        return;
    }


    try {

        await apiRequest(
            "reject",
            {
                incentive_id:
                    numericId,

                rejection_reason:
                    trimmed,

                reason:
                    trimmed
            }
        );


        showMessage(
            "Submission rejected.",
            "success"
        );


        closeModal(
            "viewModal"
        );


        await refreshAdminData();

    }
    catch (err) {

        showMessage(
            err.message ||
            "Failed to reject submission.",
            "error"
        );

    }
}

function getSelectedMonth() {

    const input =
        document.getElementById(
            "summaryMonth"
        );


    if (input.value) {
        return input.value;
    }


    const d =
        new Date();


    return (
        d.getFullYear() +
        "-" +
        String(
            d.getMonth() + 1
        ).padStart(2, "0")
    );
}


function calculateMonthlySummary(month) {

    let individual =
        0;

    let bundle =
        0;

    let individualQty =
        0;

    let bundleQty =
        0;


    const approved =
        state.submissions.filter(
            submission => {

                if (
                    getStatus(
                        submission
                    ).toLowerCase() !==
                    "approved"
                ) {
                    return false;
                }


                const date =
                    getDateOnly(
                        getSubmissionDate(
                            submission
                        )
                    );


                return date.startsWith(
                    month
                );
            }
        );


    for (const submission of approved) {

        const quantity =
            getQuantity(
                submission
            );


        const amount =
            getSubmissionAmount(
                submission
            );


        if (
            isBundle(
                submission
            )
        ) {

            bundle += amount;

            bundleQty += quantity;

        }
        else {

            individual += amount;

            individualQty += quantity;

        }

    }


    return {
        individual,
        bundle,
        total:
            individual + bundle,

        individualQty,
        bundleQty,

        count:
            approved.length
    };
}


function renderMonthlySummary() {

    const month =
        getSelectedMonth();


    const totals =
        calculateMonthlySummary(
            month
        );


    document.getElementById(

    tbody.innerHTML = `

            </td>

            </td>

                    totals.bundle
                )}
            </td>

            <td class="money">

                )}
            </td>

        </tr>

    `;
}


/* ================================================================
   TOP 3
================================================================ */

function calculateTop3() {

    const map =
        new Map();


    for (
        const submission
        of state.submissions
    ) {

        if (
            getStatus(
                submission
            ).toLowerCase() !==
            "approved"
        ) {
            continue;
        }


        const smc =
            String(
                getSMC(submission)
            ).trim();


        if (!smc) {
            continue;
        }


        const name =
            getStaffName(
                submission
            ) ||
            smc;


        const outlet =
            getOutlet(
                submission
            );


        const amount =
            getSubmissionAmount(
                submission
            );


        const existing =
            map.get(smc) ||
            {
                smc,
                name,
                outlet,
                total: 0,
                quantity: 0,
                individual: 0,
                bundle: 0
            };


        existing.total +=
            amount;


        existing.quantity +=
            getQuantity(
                submission
            );


        if (
            isBundle(
                submission
            )
        ) {

            existing.bundle +=
                amount;

        }
        else {

            existing.individual +=
                amount;

        }


        map.set(
            smc,
            existing
        );
    }


    return Array.from(
        map.values()
    )
    .sort(
        (a, b) =>
            b.total -
            a.total
    )
    .slice(
        0,
        3
    );
}

        }


        cards.push(`

/* ================================================================
   USER EDIT / DELETE
================================================================ */

document.getElementById(
    "usersTableBody"
).addEventListener(
    "click",
    async function(e) {

        const editBtn =
            e.target.closest(
                ".user-edit-btn"
            );


        if (editBtn) {

            const smc =
                editBtn.dataset.id;


            const user =
                state.users.find(
                    x =>
                        String(
                            x?.sales_man_code ??
                            x?.SMC ??
                            ""
                        ) ===
                        String(smc)
                );


            if (!user) {

                showMessage(
                    "User not found.",
                    "error"
                );

                return;
            }


            const name =
                window.prompt(
                    "Staff name:",
                    getStaffName(user)
                );


            if (name === null) {
                return;
            }


            const outlet =
                window.prompt(
                    "Outlet:",
                    getOutlet(user)
                );


            if (outlet === null) {
                return;
            }


            const role =
                window.prompt(
                    "Role (staff/admin):",
                    user?.role ||
                    "staff"
                );


            if (role === null) {
                return;
            }


            try {

                await apiRequest(
                    "update_user",
                    {
                        sales_man_code:
                            smc,

                        SMC:
                            smc,

                        staff_name:
                            name.trim(),

                        outlet:
                            outlet.trim(),

                        role:
                            role.trim()
                    }
                );


                showMessage(
                    "User updated successfully.",
                    "success"
                );


                await refreshAdminData();

            }
            catch (err) {

                showMessage(
                    err.message ||
                    "Failed to update user.",
                    "error"
                );

            }


            return;
        }


        const deleteBtn =
            e.target.closest(
                ".user-delete-btn"
            );


        if (deleteBtn) {

            const smc =
                deleteBtn.dataset.id;


            if (
                !window.confirm(
                    `Delete user ${smc}?`
                )
            ) {
                return;
            }


            try {

                await apiRequest(
                    "delete_user",
                    {
                        sales_man_code:
                            smc,

                        SMC:
                            smc
                    }
                );


                showMessage(
                    "User deleted.",
                    "success"
                );


                await refreshAdminData();

            }
            catch (err) {

                showMessage(
                    err.message ||
                    "Failed to delete user.",
                    "error"
                );

            }

        }

    }
);
