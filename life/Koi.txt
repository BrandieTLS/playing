function escapeHtml(value) {

    if (value === null || value === undefined) {
        return "";
    }

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}


function money(value) {

    const n = Number(value);

    if (!Number.isFinite(n)) {
        return "$0.00";
    }

    return "$" + n.toFixed(2);
}


function numberValue(value) {

    const n = Number(value);

    return Number.isFinite(n) ? n : 0;
}

function formatStatus(status) {

    const value = String(status || "")
        .toLowerCase();

    const label =
        value.charAt(0).toUpperCase() +
        value.slice(1);

    let cls = "status-default";

    if (value === "approved") {
        cls = "status-approved";
    }
    else if (value === "pending") {
        cls = "status-pending";
    }
    else if (value === "rejected") {
        cls = "status-rejected";
    }
    else if (value === "active") {
        cls = "status-active";
    }
    else if (value === "upcoming") {
        cls = "status-upcoming";
    }
    else if (value === "expired") {
        cls = "status-expired";
    }

    return `<span class="status ${cls}">
                ${escapeHtml(label || "Unknown")}
            </span>`;
}


function formatType(type) {

    const value = String(type || "")
        .toLowerCase();

    if (value.includes("bundle")) {

        return `<span class="type-badge type-bundle">
                    Bundle
                </span>`;

    }

    return `<span class="type-badge">
                Individual
            </span>`;
}


function formatDate(value) {

    if (!value) {
        return "-";
    }

    const str = String(value);

    if (str.includes(" ")) {
        return str.split(" ")[0];
    }

    return str;
}

async function apiRequest(action, data = {}) {

    const url =
        "admin.php?action=" +
        encodeURIComponent(action);

    try {

        const response = await fetch(url, {

            method: "POST",

            credentials: "same-origin",

            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },

            body: JSON.stringify(data)

        });


        const raw = await response.text();

        let result;

        try {

            result = JSON.parse(raw);

        } catch (error) {

            throw new Error(
                raw
                    ? raw.substring(0, 500)
                    : "Server returned an empty response."
            );

        }


        if (!response.ok) {

            throw new Error(
                result.message ||
                result.error ||
                "HTTP error " + response.status
            );

        }


        if (
            result.success === false ||
            result.ok === false
        ) {

            throw new Error(
                result.message ||
                result.error ||
                "Request failed."
            );

        }


        return result;

    } catch (error) {

        console.error("API Error:", action, error);

        throw error;
    }
}

/* =============================================================
   DASHBOARD
============================================================= */

function calculateDashboard() {

    const submissions =
        Array.isArray(adminData.submissions)
            ? adminData.submissions
            : [];


    let total = 0;
    let pending = 0;
    let approved = 0;
    let rejected = 0;

    let individualTotal = 0;
    let bundleTotal = 0;


    submissions.forEach(row => {

        total++;

        const status =
            getStatus(row);

        const type =
            getType(row);

        const amount =
            getAmount(row);


        if (status === "pending") {
            pending++;
        }

        else if (status === "approved") {

            approved++;

            if (
                type.includes("bundle")
            ) {
                bundleTotal += amount;
            }

            else {
                individualTotal += amount;
            }

        }

        else if (status === "rejected") {
            rejected++;
        }

    });


    return {
        total,
        pending,
        approved,
        rejected,
        individualTotal,
        bundleTotal,
        overallTotal:
            individualTotal +
            bundleTotal
    };
}


function renderDashboard() {

    const stats =
        calculateDashboard();


    document.getElementById(
        "totalSubmissions"
    ).textContent = stats.total;


    document.getElementById(
        "pendingSubmissions"
    ).textContent = stats.pending;


    document.getElementById(
        "approvedSubmissions"
    ).textContent = stats.approved;


    document.getElementById(
        "rejectedSubmissions"
    ).textContent = stats.rejected;


    document.getElementById(
        "individualTotal"
    ).textContent =
        money(stats.individualTotal);


    document.getElementById(
        "bundleTotal"
    ).textContent =
        money(stats.bundleTotal);


    document.getElementById(
        "overallTotal"
    ).textContent =
        money(stats.overallTotal);


    const individual =
        adminData.submissions.filter(row => {

            return (
                !getType(row).includes("bundle")
            );

        });


    const bundles =
        adminData.submissions.filter(row => {

            return getType(row).includes("bundle");

        });


    function breakdown(rows) {

        let a = 0;
        let p = 0;
        let r = 0;
        let amount = 0;

        rows.forEach(row => {

            const status =
                getStatus(row);

            if (status === "approved") {

                a++;
                amount += getAmount(row);

            }
            else if (status === "pending") {
                p++;
            }
            else if (status === "rejected") {
                r++;
            }

        });

        return {
            a,
            p,
            r,
            amount
        };
    }


    const i = breakdown(individual);
    const b = breakdown(bundles);


    document.getElementById(
        "dashboardBreakdown"
    ).innerHTML = `

        <tr>
            <td>Individual</td>
            <td>${i.a}</td>
            <td>${i.p}</td>
            <td>${i.r}</td>
            <td class="money">${money(i.amount)}</td>
        </tr>

        <tr>
            <td>Bundle</td>
            <td>${b.a}</td>
            <td>${b.p}</td>
            <td>${b.r}</td>
            <td class="money">${money(b.amount)}</td>
        </tr>

        <tr class="breakdown-total">
            <td>Total</td>
            <td>${i.a + b.a}</td>
            <td>${i.p + b.p}</td>
            <td>${i.r + b.r}</td>
            <td class="money">
                ${money(i.amount + b.amount)}
            </td>
        </tr>
    `;
}

function populateSubmissionStaffFilter() {

    const select =
        document.getElementById(
            "submissionStaffFilter"
        );


    const current =
        select.value;


    const map =
        new Map();


    adminData.submissions.forEach(row => {

        const smc = getSMC(row);

        const name = getStaffName(row);

        if (smc || name) {

            const key =
                smc || name;

            map.set(
                key,
                name
                    ? `${name} (${smc || "-"})`
                    : key
            );

        }

    });


    select.innerHTML =
        `<option value="">All Staff</option>`;


    [...map.entries()]
        .sort((a, b) =>
            a[1].localeCompare(b[1])
        )
        .forEach(([value, label]) => {

            const option =
                document.createElement("option");

            option.value = value;

            option.textContent = label;

            select.appendChild(option);

        });


    if (
        [...select.options]
            .some(o => o.value === current)
    ) {
        select.value = current;
    }
}

function renderSubmissions() {

    const body =
        document.getElementById(
            "submissionsBody"
        );


    const search =
        document.getElementById(
            "submissionSearch"
        ).value
        .trim()
        .toLowerCase();


    const statusFilter =
        document.getElementById(
            "submissionStatusFilter"
        ).value
        .toLowerCase();


    const typeFilter =
        document.getElementById(
            "submissionTypeFilter"
        ).value
        .toLowerCase();


    const staffFilter =
        document.getElementById(
            "submissionStaffFilter"
        ).value;


    const filtered =
        adminData.submissions.filter(row => {

            const searchable = (

                String(getId(row)) +
                " " +
                getStaffName(row) +
                " " +
                getSMC(row) +
                " " +
                getOutlet(row) +
                " " +
                getPromoName(row) +
                " " +
                getType(row)

            ).toLowerCase();


            if (
                search &&
                !searchable.includes(search)
            ) {
                return false;
            }


            if (
                statusFilter &&
                getStatus(row) !== statusFilter
            ) {
                return false;
            }


            if (
                typeFilter
            ) {

                const type =
                    getType(row);

                if (
                    typeFilter === "bundle" &&
                    !type.includes("bundle")
                ) {
                    return false;
                }

                if (
                    typeFilter === "individual" &&
                    type.includes("bundle")
                ) {
                    return false;
                }

            }


            if (staffFilter) {

                const smc =
                    getSMC(row);

                if (
                    smc !== staffFilter
                ) {
                    return false;
                }

            }


            return true;

        });


    if (!filtered.length) {

        body.innerHTML = `
            <tr>
                <td colspan="12" class="empty">
                    No submissions found.
                </td>
            </tr>
        `;

        return;
    }


    body.innerHTML =
        filtered.map(row => {

            const id =
                getId(row);

            return `

                <tr>

                    <td>
                        ${escapeHtml(id)}
                    </td>

                    <td>
                        ${escapeHtml(
                            getStaffName(row) || "-"
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            getSMC(row) || "-"
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            getOutlet(row) || "-"
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            getPromoName(row) || "-"
                        )}
                    </td>

                    <td>
                        ${formatType(
                            getType(row)
                        )}
                    </td>

                    <td>
                        ${getQuantity(row)}
                    </td>

                    <td class="money">
                        ${money(getRate(row))}
                    </td>

                    <td class="money">
                        ${money(getAmount(row))}
                    </td>

                    <td>
                        ${formatStatus(
                            getStatus(row)
                        )}
                    </td>

                    <td>
                        ${escapeHtml(
                            formatDate(getDate(row))
                        )}
                    </td>

                    <td class="action-column">

                        <button
                            class="btn btn-primary btn-small"
                            onclick="viewSubmission(${Number(id)})">

                            View

                        </button>

                    </td>

                </tr>

            `;

        }).join("");
}

function getBundleItems(row) {

    const items = [];


    for (
        let i = 1;
        i <= 5;
        i++
    ) {

        const value =
            row?.[`item_${i}`];


        if (
            value !== null &&
            value !== undefined &&
            String(value).trim() !== ""
        ) {

            items.push(
                String(value).trim()
            );

        }

    }


    return items;
}

function normaliseDateInput(value) {

    if (!value) {
        return "";
    }


    const str =
        String(value);


    if (
        /^\d{4}-\d{2}-\d{2}$/.test(str)
    ) {
        return str;
    }


    if (
        str.includes(" ")
    ) {
        return str.split(" ")[0];
    }


    return str.substring(
        0,
        10
    );

}

function renderUsers() {

    const body =
        document.getElementById(
            "usersBody"
        );


    const search =
        document.getElementById(
            "userSearch"
        ).value
        .trim()
        .toLowerCase();


    const roleFilter =
        document.getElementById(
            "userRoleFilter"
        ).value
        .toLowerCase();


    const filtered =
        adminData.users.filter(row => {

            const name =
                String(
                    row.staff_name ??
                    row.name ??
                    ""
                );


            const smc =
                String(
                    row.sales_man_code ??
                    row.SMC ??
                    ""
                );


            const outlet =
                String(
                    row.outlet ??
                    ""
                );


            const role =
                String(
                    row.role ??
                    ""
                ).toLowerCase();


            const searchable =
                (
                    name +
                    " " +
                    smc +
                    " " +
                    outlet
                ).toLowerCase();


            if (
                search &&
                !searchable.includes(search)
            ) {
                return false;
            }


            if (
                roleFilter &&
                role !== roleFilter
            ) {
                return false;
            }


            return true;

        });


    if (!filtered.length) {

        body.innerHTML = `

            <tr>
                <td colspan="5" class="empty">
                    No users found.
                </td>
            </tr>

        `;

        return;
    }


    body.innerHTML =
        filtered.map(row => {

            const name =
                row.staff_name ??
                row.name ??
                "-";


            const smc =
                row.sales_man_code ??
                row.SMC ??
                "-";


            const outlet =
                row.outlet ??
                "-";


            const role =
                String(
                    row.role ??
                    "staff"
                ).toLowerCase();


            const userId =
                row.id ??
                row.user_id ??
                row.sales_man_code ??
                row.SMC ??
                "";


            return `

                <tr>

                    <td>
                        ${escapeHtml(name)}
                    </td>

                    <td>
                        ${escapeHtml(smc)}
                    </td>

                    <td>
                        ${escapeHtml(outlet)}
                    </td>

                    <td>
                        ${formatStatus(role)}
                    </td>

                    <td>

                        <button
                            class="btn btn-danger btn-small"
                            onclick="deleteUser('${escapeHtml(userId)}')">

                            Delete

                        </button>

                    </td>

                </tr>

            `;

        }).join("");
}

function getDefaultMonth() {

    const now =
        new Date();


    const year =
        now.getFullYear();


    const month =
        String(
            now.getMonth() + 1
        ).padStart(2, "0");


    return `${year}-${month}`;

}


function ensureMonthSelected() {

    const input =
        document.getElementById(
            "summaryMonth"
        );


    if (!input.value) {

        input.value =
            getDefaultMonth();

    }

}

    fromBootstrap = false
) {

    ensureMonthSelected();


    const month =
        document.getElementById(
            "summaryMonth"
        ).value;


    if (!month) {
        return;
    }


    const [year, monthNumber] =
        month.split("-");


    try {

        const result =

                "monthly_summary",
                {
                    year:
                        Number(year),

                    month:

                    selected_month:

                }
            );

        adminData.monthlySummary =

                Array.isArray(data)
                    ? data
                    : []
            );


        renderMonthlySummary(

function renderMonthlySummary(
    serverData
) {

    const rows =
        adminData.monthlySummary;


    if (
        rows.length

function renderMonthlyRows(rows) {

    const body =
        document.getElementById(
            "monthlyBody"
        );


    const grouped =
        new Map();


    rows.forEach(row => {

        const smc =
            getSMC(row) ||
            row.smc ||
            "-";


        const name =
            getStaffName(row) ||
            "-";


        const outlet =
            getOutlet(row) ||
            "-";


        const individual =
            numberValue(
                row.individual_total ??
                row.individual ??
                row.individual_amount ??
                0
            );


        const bundle =
            numberValue(
                row.bundle_total ??
                row.bundle ??
                row.bundle_amount ??
                0
            );


        const total =
            numberValue(
                row.total ??
                row.total_amount ??
                (
                    individual +
                    bundle
                )
            );


        const key =
            smc;


        if (!grouped.has(key)) {

            grouped.set(
                key,
                {
                    name,
                    smc,
                    outlet,
                    individual: 0,
                    bundle: 0,
                    total: 0
                }
            );

        }


        const item =
            grouped.get(key);


        item.individual +=
            individual;


        item.bundle +=
            bundle;


        item.total +=
            total;

    });


    if (!grouped.size) {

        body.innerHTML = `

            <tr>
                <td colspan="6" class="empty">
                    No approved incentives for this month.
                </td>
            </tr>

        `;

        return;
    }


    body.innerHTML =
        [...grouped.values()]
            .sort(
                (a, b) =>
                    b.total - a.total
            )
            .map(item => `

                <tr>

                    <td>
                        ${escapeHtml(item.name)}
                    </td>

                    <td>
                        ${escapeHtml(item.smc)}
                    </td>

                    <td>
                        ${escapeHtml(item.outlet)}
                    </td>

                    <td class="money">
                        ${money(item.individual)}
                    </td>

                    <td class="money">
                        ${money(item.bundle)}
                    </td>

                    <td class="money">
                        ${money(item.total)}
                    </td>

                </tr>

            `)
            .join("");

}


function calculateMonthlyCardsFromRows(
    rows,
    serverData
) {

    let individual = 0;
    let bundle = 0;


    rows.forEach(row => {

        individual +=
            numberValue(
                row.individual_total ??
                row.individual ??
                row.individual_amount ??
                0
            );


        bundle +=
            numberValue(
                row.bundle_total ??
                row.bundle ??
                row.bundle_amount ??
                0
            );

    });


    /*
     * If the server gives explicit totals,
     * prefer them.
     */

    const serverIndividual =
        numberValue(
            serverData?.individual_total ??
            serverData?.individualTotal
        );


    const serverBundle =
        numberValue(
            serverData?.bundle_total ??
            serverData?.bundleTotal
        );


    if (
        serverIndividual ||
        serverBundle
    ) {

        individual =
            serverIndividual;

        bundle =
            serverBundle;

    }


    document.getElementById(
        "monthlyIndividualTotal"
    ).textContent =
        money(individual);


    document.getElementById(
        "monthlyBundleTotal"
    ).textContent =
        money(bundle);


    document.getElementById(
        "monthlyOverallTotal"
    ).textContent =
        money(
            individual +
            bundle
        );

}

function renderLocalMonthlySummary(
    year,
    month
) {

    const approved =
        adminData.submissions.filter(row => {

            if (
                getStatus(row) !==
                "approved"
            ) {
                return false;
            }


            const date =
                getDate(row);


            if (!date) {
                return false;
            }


            const normal =
                String(date)
                    .substring(0, 7);


            const wanted =
                `${year}-${String(month).padStart(2, "0")}`;


            return normal === wanted;

        });


    let individual = 0;
    let bundle = 0;


    const staffMap =
        new Map();


    const promoMap =
        new Map();


    approved.forEach(row => {

        const amount =
            getAmount(row);


        const type =
            getType(row);


        if (
            type.includes("bundle")
        ) {

            bundle += amount;

        } else {

            individual += amount;

        }


        const smc =
            getSMC(row) || "-";


        if (!staffMap.has(smc)) {

            staffMap.set(
                smc,
                {
                    name:
                        getStaffName(row) || "-",

                    smc,

                    outlet:
                        getOutlet(row) || "-",

                    individual: 0,

                    bundle: 0,

                    total: 0
                }
            );

        }


        const staff =
            staffMap.get(smc);


        if (
            type.includes("bundle")
        ) {

            staff.bundle +=
                amount;

        } else {

            staff.individual +=
                amount;

        }


        staff.total +=
            amount;


        const promoId =
            getPromoId(row);


        const promoKey =
            promoId ||
            getPromoName(row);


        if (!promoMap.has(promoKey)) {

            promoMap.set(
                promoKey,
                {
                    promo:
                        getPromoName(row) || "-",

                    type:
                        getType(row),

                    quantity: 0,

                    rate:
                        getRate(row),

                    total: 0
                }
            );

        }


        const promotion =
            promoMap.get(
                promoKey
            );


        promotion.quantity +=
            getQuantity(row);


        promotion.total +=
            amount;

    });


    document.getElementById(
        "monthlyIndividualTotal"
    ).textContent =
        money(individual);


    document.getElementById(
        "monthlyBundleTotal"
    ).textContent =
        money(bundle);


    document.getElementById(
        "monthlyOverallTotal"
    ).textContent =
        money(
            individual +
            bundle
        );


    const monthlyBody =
        document.getElementById(
            "monthlyBody"
        );


    if (!staffMap.size) {

        monthlyBody.innerHTML = `

            <tr>
                <td colspan="6" class="empty">
                    No approved incentives for this month.
                </td>
            </tr>

        `;

    } else {

        monthlyBody.innerHTML =
            [...staffMap.values()]
                .sort(
                    (a, b) =>
                        b.total - a.total
                )
                .map(item => `

                    <tr>

                        <td>
                            ${escapeHtml(item.name)}
                        </td>

                        <td>
                            ${escapeHtml(item.smc)}
                        </td>

                        <td>
                            ${escapeHtml(item.outlet)}
                        </td>

                        <td class="money">
                            ${money(item.individual)}
                        </td>

                        <td class="money">
                            ${money(item.bundle)}
                        </td>

                        <td class="money">
                            ${money(item.total)}
                        </td>

                    </tr>

                `)
                .join("");

    }


    const promoBody =
        document.getElementById(
            "monthlyPromotionBody"
        );


    if (!promoMap.size) {

        promoBody.innerHTML = `

            <tr>
                <td colspan="5" class="empty">
                    No approved promotion data.
                </td>
            </tr>

        `;

    } else {

        promoBody.innerHTML =
            [...promoMap.values()]
                .sort(
                    (a, b) =>
                        b.total - a.total
                )
                .map(item => `

                    <tr>

                        <td>
                            ${escapeHtml(item.promo)}
                        </td>

                        <td>
                            ${formatType(item.type)}
                        </td>

                        <td>
                            ${item.quantity}
                        </td>

                        <td class="money">
                            ${money(item.rate)}
                        </td>

                        <td class="money">
                            ${money(item.total)}
                        </td>

                    </tr>

                `)
                .join("");

    }

}

async function loadTop3(
    fromBootstrap = false
) {

    try {

        const result =
            await apiRequest(
                "top3"
            );


        const data =
            result?.data ||
            result;


        adminData.top3 =
            firstArray(
                data?.top3,
                data?.top_3,
                data?.rows,
                Array.isArray(data)
                    ? data
                    : []
            );


        if (
            adminData.top3.length
        ) {

            renderTop3(
                adminData.top3
            );

        } else {

            renderLocalTop3();

        }

    } catch (error) {

        console.warn(
            "Top 3 API failed:",
            error
        );


        renderLocalTop3();

    }

}


function renderTop3(rows) {

    const normalised =
        rows.map(row => {

            const smc =
                getSMC(row) ||
                "-";


            const name =
                getStaffName(row) ||
                "-";


            const outlet =
                getOutlet(row) ||
                "-";


            const individual =
                numberValue(
                    row.individual_total ??
                    row.individual ??
                    0
                );


            const bundle =
                numberValue(
                    row.bundle_total ??
                    row.bundle ??
                    0
                );


            const total =
                numberValue(
                    row.total ??
                    row.total_amount ??
                    individual +
                    bundle
                );


            return {
                name,
                smc,
                outlet,
                individual,
                bundle,
                total
            };

        });


    normalised.sort(
        (a, b) =>
            b.total - a.total
    );


    renderTop3Cards(
        normalised.slice(0, 3)
    );

}


function renderLocalTop3() {

    const map =
        new Map();


    adminData.submissions
        .filter(row =>
            getStatus(row) ===
            "approved"
        )
        .forEach(row => {

            const smc =
                getSMC(row) || "-";


            if (!map.has(smc)) {

                map.set(
                    smc,
                    {
                        name:
                            getStaffName(row) || "-",

                        smc,

                        outlet:
                            getOutlet(row) || "-",

                        individual: 0,

                        bundle: 0,

                        total: 0
                    }
                );

            }


            const item =
                map.get(smc);


            const amount =
                getAmount(row);


            if (
                getType(row)
                    .includes("bundle")
            ) {

                item.bundle +=
                    amount;

            } else {

                item.individual +=
                    amount;

            }


            item.total +=
                amount;

        });


    const rows =
        [...map.values()]
            .sort(
                (a, b) =>
                    b.total - a.total
            )
            .slice(0, 3);


    renderTop3Cards(
        rows
    );

}


function renderTop3Cards(rows) {

    const grid =
        document.getElementById(
            "top3Grid"
        );


    const positions = [
        {
            className: "first",
            rank: 1,
            label: "First"
        },
        {
            className: "second",
            rank: 2,
            label: "Second"
        },
        {
            className: "third",
            rank: 3,
            label: "Third"
        }
    ];


    grid.innerHTML =
        positions.map(
            (position, index) => {

                const row =
                    rows[index];


                if (!row) {

                    return `

                        <div class="top-card top-empty">

                            <div class="rank-circle">
                                ${position.rank}
                            </div>

                            <div class="place-label">
                                ${position.label}
                            </div>

                            <div class="top-name">
                                —
                            </div>

                            <div class="top-meta">
                                No data
                            </div>

                        </div>

                    `;

                }


                return `

                    <div class="top-card ${position.className}">

                        <div class="rank-circle">
                            ${position.rank}
                        </div>

                        <div class="place-label">
                            ${position.label}
                        </div>

                        <div class="top-name">
                            ${escapeHtml(row.name)}
                        </div>

                        <div class="top-meta">
                            SMC: ${escapeHtml(row.smc)}
                            <br>
                            ${escapeHtml(row.outlet)}
                        </div>


                        <div class="top-stat">

                            <span>
                                Individual
                            </span>

                            <span>
                                ${money(row.individual)}
                            </span>

                        </div>


                        <div class="top-stat">

                            <span>
                                Bundle
                            </span>

                            <span>
                                ${money(row.bundle)}
                            </span>

                        </div>


                        <div class="top-total">

                            <div class="top-total-label">
                                Approved Total
                            </div>

                            <div class="top-total-value">
                                ${money(row.total)}
                            </div>

                        </div>

                    </div>

                `;

            }
        ).join("");

}

function openModal(id) {

    const modal =
        document.getElementById(id);


    if (modal) {

        modal.classList.add(
            "show"
        );

    }

}


function closeModal(id) {

    const modal =
        document.getElementById(id);


    if (modal) {

        modal.classList.remove(
            "show"
        );

    }

}


document.querySelectorAll(
    ".modal-overlay"
).forEach(modal => {

    modal.addEventListener(
        "click",
        function(event) {

            if (
                event.target === modal
            ) {

                modal.classList.remove(
                    "show"
                );

            }

        }
    );

});


document.addEventListener(
    "keydown",
    function(event) {

        if (
            event.key === "Escape"
        ) {

            document.querySelectorAll(
                ".modal-overlay.show"
            ).forEach(modal => {

                modal.classList.remove(
                    "show"
                );

            });

        }

    }
);

document.querySelectorAll(
    ".tab-btn"
).forEach(button => {

    button.addEventListener(
        "click",
        function() {

            const target =
                this.dataset.tab;


            document.querySelectorAll(
                ".tab-btn"
            ).forEach(btn => {

                btn.classList.remove(
                    "active"
                );

            });


            document.querySelectorAll(
                ".tab"
            ).forEach(tab => {

                tab.classList.remove(
                    "active"
                );

            });


            this.classList.add(
                "active"
            );


            document.getElementById(
                target
            ).classList.add(
                "active"
            );


            /*
             * Refresh the selected section.
             */

            if (
                target ===
                "dashboardTab"
            ) {

                renderDashboard();

            }

            else if (
                target ===
                "submissionsTab"
            ) {

                loadSubmissions();

            }

            else if (
                target ===
                "promotionsTab"
            ) {

                loadPromotions();

            }

            else if (
                target ===
                "usersTab"
            ) {

                loadUsers();

            }

            else if (
                target ===
                "monthlyTab"
            ) {

                loadMonthlySummary();

            }

            else if (
                target ===
                "top3Tab"
            ) {

                loadTop3();

            }

        }
    );

});
