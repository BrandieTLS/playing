async function apiRequest(action, data = {}) {

    const response = await fetch(
        `admin.php?action=${encodeURIComponent(action)}`,
        {
            method: "POST",

            credentials: "same-origin",

            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },

            body: JSON.stringify(data)
        }
    );


    const text = await response.text();

    let json;

    try {

        json = text ? JSON.parse(text) : {};

    } catch (error) {

        throw new Error(
            text ||
            `Server returned HTTP ${response.status}`
        );

    }


    if (
        !response.ok ||
        json.success === false
    ) {

        throw new Error(
            json.message ||
            json.error ||
            `Request failed (${response.status})`
        );

    }


    return json;
}


/* ================================================================

function escapeHtml(value) {

    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");

}


function money(value) {

    const number = Number(value) || 0;

    return `S$${number.toFixed(2)}`;

}


function getSubmissionId(x) {

    const raw =
        x?.incentive_id ??
        x?.submission_id ??
        x?.id ??
        x?.incentiveId;

    if (
        raw === null ||
        raw === undefined ||
        String(raw).trim() === ""
    ) {

        return null;

    }


    const n = Number(raw);

    return Number.isFinite(n) && n > 0
        ? n
        : null;

}


function getPromoId(x) {

    const raw =
        x?.promo_id ??
        x?.promotion_id ??
        x?.promoId;

    const n = Number(raw);

    return Number.isFinite(n)
        ? n
        : null;

}

/* ================================================================

function getPromotionType(x) {

    const direct = String(
        x?.type ??
        x?.promotion_type ??
        x?.promo_type ??
        x?.promotionType ??
        ""
    ).trim();


    if (direct) {

        return direct;

    }


    const promoId = getPromoId(x);


    if (promoId !== null) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );


        if (promo) {

            return String(
                promo.type ??
                promo.promotion_type ??
                promo.promo_type ??
                ""
            ).trim();

        }

    }


    return "";

}


function isBundle(x) {

    return getPromotionType(x)
        .toLowerCase()
        .replace(/[\s_-]/g, "") === "bundle";

}


function displayType(x) {

    const type = getPromotionType(x);

    if (isBundle(x)) {

        return "Bundle";

    }


    if (type) {

        return type;

    }


    return "Individual";

}

function getPromotionStartDate(promo) {

    return String(
        promo?.start_date ??
        promo?.startDate ??
        promo?.date_start ??
        ""
    ).substring(0, 10);

}


function getPromotionEndDate(promo) {

    return String(
        promo?.end_date ??
        promo?.endDate ??
        promo?.date_end ??
        ""
    ).substring(0, 10);

}


function getPromotionStatus(promo) {

    const backendStatus = String(
        promo?.status ??
        promo?.promotion_status ??
        ""
    ).trim();


    if (backendStatus) {

        return backendStatus;

    }


    const start = getPromotionStartDate(promo);
    const end = getPromotionEndDate(promo);


    if (!start && !end) {

        return "Active";

    }


    const today = new Date();

    const todayString =
        today.getFullYear() +
        "-" +
        String(today.getMonth() + 1).padStart(2, "0") +
        "-" +
        String(today.getDate()).padStart(2, "0");


    if (start && todayString < start) {

        return "Upcoming";

    }


    if (end && todayString > end) {

        return "Expired";

    }


    return "Active";

}


function statusClass(status) {

    const value = String(status)
        .toLowerCase()
        .replace(/[\s_-]/g, "");


    if (
        value === "approved" ||
        value === "active"
    ) {

        return "status-approved";

    }


    if (
        value === "pending" ||
        value === "upcoming"
    ) {

        return "status-pending";

    }


    if (
        value === "rejected" ||
        value === "expired"
    ) {

        return "status-rejected";

    }


    return "status-default";

}


function statusBadge(status) {

    return `
        <span class="status ${statusClass(status)}">
            ${escapeHtml(status || "Unknown")}
        </span>
    `;

}

function formatDate(value) {

    const str = String(value || "");

    if (!str) {

        return "-";

    }


    if (/^\d{4}-\d{2}-\d{2}/.test(str)) {

        return str.substring(0, 10);

    }


    return str;

}


/* ================================================================
   PROOF URL
================================================================ */

function getProofUrl(proof) {

    const value = String(proof || "").trim();

    if (!value) {

        return "";

    }


    if (
        value.startsWith("http://") ||
        value.startsWith("https://")
    ) {

        return value;

    }


    if (value.startsWith("javascript:")) {

        return "";

    }


    if (value.startsWith("/")) {

        return value;

    }


    return value;

}

        return;

    refreshInProgress = true;


    try {

        await loadAdminData();

        renderAll();


        if (
            state.currentSubmissionId !== null

            const exists =

                closeModal("viewModal");

            }

    } catch (error) {

        console.error(error);

        showMessage(
            error.message ||
            "Failed to refresh data.",
            "error"
        );

    } finally {

        refreshInProgress = false;

    }

}


/* ================================================================
   DASHBOARD CALCULATION
================================================================ */

function calculateDashboard() {

    let individual = 0;

    let bundle = 0;

    let pending = 0;

    let approved = 0;

    let rejected = 0;


    state.submissions.forEach(submission => {

        const status =
            getStatus(submission)
                .toLowerCase();


        const quantity =
            getQty(submission);


        const rate =
            getRate(submission);


        const amount =
            quantity * rate;


        if (status === "approved") {

            approved++;


            if (isBundle(submission)) {

                bundle += amount;

            } else {

                individual += amount;

            }

        }


        else if (status === "pending") {

            pending++;

        }


        else if (status === "rejected") {

            rejected++;

        }

    });


    const totalStaff =
        new Set(
            state.users
                .map(getSMC)
                .filter(Boolean)
        ).size;


    return {

        individual,

        bundle,

        total: individual + bundle,

        pending,

        approved,

        rejected,

        totalStaff:
            totalStaff ||
            state.users.length

    };

}

/* ================================================================
   SUBMISSION FILTER
================================================================ */

function getFilteredSubmissions() {

    const search =
        document.getElementById(
            "submissionSearch"
        ).value
        .trim()
        .toLowerCase();


    const status =
        document.getElementById(
            "submissionStatusFilter"
        ).value
        .toLowerCase();


    const type =
        document.getElementById(
            "submissionTypeFilter"
        ).value
        .toLowerCase();


    const outlet =
        document.getElementById(
            "submissionOutletFilter"
        ).value;


    const from =
        document.getElementById(
            "submissionDateFrom"
        ).value;


    const to =
        document.getElementById(
            "submissionDateTo"
        ).value;


    return state.submissions.filter(row => {

        const staff =
            getStaff(row).toLowerCase();


        const smc =
            getSMC(row).toLowerCase();


        const promo =
            getPromo(row).toLowerCase();


        const rowStatus =
            getStatus(row).toLowerCase();


        const rowType =
            displayType(row).toLowerCase();


        const rowOutlet =
            getOutlet(row);


        const rowDate =
            formatDate(getDate(row));


        if (
            search &&
            !staff.includes(search) &&
            !smc.includes(search) &&
            !promo.includes(search)
        ) {

            return false;

        }


        if (
            status &&
            rowStatus !== status
        ) {

            return false;

        }


        if (
            type &&
            rowType !== type
        ) {

            return false;

        }


        if (
            outlet &&
            rowOutlet !== outlet
        ) {

            return false;

        }


        if (
            from &&
            rowDate &&
            rowDate < from
        ) {

            return false;

        }


        if (
            to &&
            rowDate &&
            rowDate > to
        ) {

            return false;

        }


        return true;

    });

}

function populateOutletFilter() {

    const select =
        document.getElementById(
            "submissionOutletFilter"
        );


    const previous =
        select.value;


    const outlets =
        new Set();


    state.submissions.forEach(row => {

        const outlet =
            getOutlet(row).trim();


        if (outlet) {

            outlets.add(outlet);

        }

    });


    state.users.forEach(row => {

        const outlet =
            getOutlet(row).trim();


        if (outlet) {

            outlets.add(outlet);

        }

    });


    select.innerHTML =
        `<option value="">All Outlets</option>`;


    [...outlets]
        .sort()
        .forEach(outlet => {

            const option =
                document.createElement("option");


            option.value = outlet;

            option.textContent = outlet;


            select.appendChild(option);

        });


    if (
        [...outlets].includes(previous)
    ) {

        select.value = previous;

    }

}

/* ================================================================
   SUBMISSION VIEW
================================================================ */

async function showSubmission(id) {

    const numericId =
        Number(id);


    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;

    }


    state.currentSubmissionId =
        numericId;


    const localSubmission =
        state.submissions.find(
            row =>
                getSubmissionId(row) ===
                numericId
        );


    openModal("viewModal");


    document.getElementById(
        "submissionDetails"
    ).innerHTML = `
        <div class="loading">
            Loading submission...
        </div>
    `;


    document.getElementById(
        "viewActions"
    ).innerHTML = "";


    let row =
        localSubmission ||
        null;


    try {

        const response =
            await apiRequest(
                "get_submission",
                {
                    incentive_id: numericId,
                    submission_id: numericId
                }
            );


        const data =
            response.data ||
            response.submission ||
            response;


        if (data && typeof data === "object") {

            row =
                data.submission ||
                data;

        }

    } catch (error) {

        /*
         * If the backend get_submission endpoint
         * rejects the ID but the row exists in the
         * bootstrap response, continue using the
         * local row.
         */

        if (!row) {

            document.getElementById(
                "submissionDetails"
            ).innerHTML = `
                <div class="empty">
                    ${escapeHtml(
                        error.message ||
                        "Unable to load submission."
                    )}
                </div>
            `;

            return;

        }

    }


    if (!row) {

        document.getElementById(
            "submissionDetails"
        ).innerHTML = `
            <div class="empty">
                Submission not found.
            </div>
        `;

        return;

    }


    renderSubmissionDetails(row);

}

================================================================ */

async function rejectSubmission(id) {

    const numericId =
        Number(id);


    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        );

        await refreshAdminData();

    } catch (error) {

        );

    }

/* ================================================================
   SAVE PROMOTION
================================================================ */

document.getElementById(
    "promotionForm"
).addEventListener(
    "submit",
    async function(event) {

        event.preventDefault();


        const editId =
            Number(
                document.getElementById(
                    "promotionEditId"
                ).value
            );


        const isEdit =
            Number.isFinite(editId) &&
            editId > 0;


        const promo =
            document.getElementById(
                "promotionName"
            ).value.trim();


        const rate =
            Number(
                document.getElementById(
                    "promotionRate"
                ).value
            );


        const type =
            document.getElementById(
                "promotionType"
            ).value;


        const startDate =
            document.getElementById(
                "promotionStartDate"
            ).value;


        const endDate =
            document.getElementById(
                "promotionEndDate"
            ).value;


        if (!promo) {

            showMessage(
                "Promotion name is required.",
                "error"
            );

            return;

        }


        if (
            !Number.isFinite(rate) ||
            rate < 0
        ) {

            showMessage(
                "Please enter a valid rate.",
                "error"
            );

            return;

        }


        if (
            startDate &&
            !/^\d{4}-\d{2}-\d{2}$/.test(
                startDate
            )
        ) {

            showMessage(
                "Invalid start date.",
                "error"
            );

            return;

        }


        if (
            endDate &&
            !/^\d{4}-\d{2}-\d{2}$/.test(
                endDate
            )
        ) {

            showMessage(
                "Invalid end date.",
                "error"
            );

            return;

        }


        if (
            startDate &&
            endDate &&
            endDate < startDate
        ) {

            showMessage(
                "End date cannot be before start date.",
                "error"
            );

            return;

        }


        const payload = {

            promo: promo,

            Rate: rate,

            type: type

        };


        /*
         * Only send valid dates.
         * This prevents the previous
         * "incorrect date value 2026"
         * problem.
         */

        if (startDate) {

            payload.start_date =
                startDate;

        }


        if (endDate) {

            payload.end_date =
                endDate;

        }


        try {

            if (isEdit) {

                /*
                 * Send both names so the PHP backend
                 * can use either naming convention.
                 */

                payload.promo_id =
                    editId;

                payload.promotion_id =
                    editId;


                await apiRequest(
                    "update_promotion",
                    payload
                );


                showMessage(
                    "Promotion updated successfully.",
                    "success"
                );

            } else {

                await apiRequest(
                    "add_promotion",
                    payload
                );


                showMessage(
                    "Promotion added successfully.",
                    "success"
                );

            }


            closeModal("promotionModal");


            document.getElementById(
                "promotionForm"
            ).reset();


            await refreshAdminData();

        } catch (error) {

            showMessage(
                error.message ||
                "Failed to save promotion.",
                "error"
            );

        }

    }
);

/* ================================================================
   MONTHLY SUMMARY
================================================================ */

function calculateMonthlySummary() {

    const month =
        state.selectedMonth;


    const staffMap =
        new Map();


    const promoMap =
        new Map();


    let individual =
        0;


    let bundle =
        0;


    state.submissions.forEach(row => {

        if (
            getStatus(row).toLowerCase() !==
            "approved"
        ) {

            return;

        }


        const date =
            formatDate(
                getDate(row)
            );


        if (
            !date.startsWith(month)
        ) {

            return;

        }


        const smc =
            getSMC(row);


        const staff =
            getStaff(row);


        const outlet =
            getOutlet(row);


        const quantity =
            getQty(row);


        const amount =
            quantity *
            getRate(row);


        const bundleRow =
            isBundle(row);


        if (bundleRow) {

            bundle += amount;

        } else {

            individual += amount;

        }


        if (!staffMap.has(smc)) {

            staffMap.set(
                smc,
                {
                    staff,
                    smc,
                    outlet,
                    individual: 0,
                    bundle: 0
                }
            );

        }


        const staffData =
            staffMap.get(smc);


        if (bundleRow) {

            staffData.bundle += amount;

        } else {

            staffData.individual += amount;

        }


        const promoId =
            getPromoId(row);


        const promoKey =
            `${promoId ?? ""}|${getPromo(row)}`;


        if (!promoMap.has(promoKey)) {

            promoMap.set(
                promoKey,
                {
                    promo:
                        getPromo(row),

                    type:
                        displayType(row),

                    quantity:
                        0,

                    total:
                        0
                }
            );

        }


        const promoData =
            promoMap.get(promoKey);


        promoData.quantity +=
            quantity;


        promoData.total +=
            amount;

    });


    return {

        individual,

        bundle,

        total:
            individual + bundle,

        staff:
            [...staffMap.values()],

        promotions:
            [...promoMap.values()]

    };

}


function renderMonthlySummary() {

    const summary =
        calculateMonthlySummary();


    document.getElementById(

    ).textContent =

    document.getElementById(

            </tr>
        `;

    } else {

        staffBody.innerHTML =

                    (a, b) =>
                        (
                            b.individual +
                            b.bundle
                        ) -
                        (
                            a.individual +
                            a.bundle
                        )
                )
                .map(row => {

                    const total =
                        row.individual +
                        row.bundle;


                    return `
                        <tr>

                            <td>
                                ${escapeHtml(
                                    row.staff || "-"
                                )}
                            </td>

                            <td>
                                ${escapeHtml(
                                    row.smc || "-"
                                )}
                            </td>

                            <td>
                                ${escapeHtml(
                                    row.outlet || "-"
                                )}
                            </td>

                            <td class="money">
                                ${money(
                                    row.individual
                                )}
                            </td>

                            <td class="money">
                                ${money(
                                    row.bundle
                                )}
                            </td>

                            <td class="money">
                                ${money(total)}
                            </td>

                        </tr>
                    `;

                })
                .join("");

    }


    const promoBody =
        document.getElementById(
            "summaryPromotionBody"
        );


    if (!summary.promotions.length) {

        promoBody.innerHTML = `
            <tr>
                <td colspan="4" class="empty">
                    No approved incentives for this month.
                </td>
            </tr>
        `;

    } else {

        promoBody.innerHTML =
            summary.promotions
                .sort(
                    (a, b) =>
                        b.total -
                        a.total
                )
                .map(row => {

                    return `
                        <tr>

                            <td>
                                ${escapeHtml(
                                    row.promo || "-"
                                )}
                            </td>

                            <td>
                                <span class="type-badge ${
                                    row.type
                                        .toLowerCase() ===
                                        "bundle"
                                            ? "type-bundle"
                                            : ""
                                }">
                                    ${escapeHtml(
                                        row.type
                                    )}
                                </span>
                            </td>

                            <td>
                                ${row.quantity}
                            </td>

                            <td class="money">
                                ${money(
                                    row.total
                                )}
                            </td>

                        </tr>
                    `;

                })
                .join("");

    }

}


/* ================================================================
   TOP 3
================================================================ */

function calculateTop3() {

    const map =
        new Map();


    state.submissions.forEach(row => {

        if (
            getStatus(row).toLowerCase() !==
            "approved"
        ) {

            return;

        }


        const smc =
            getSMC(row);


        if (!smc) {

            return;

        }


        const amount =
            getQty(row) *
            getRate(row);


        if (!map.has(smc)) {

            map.set(
                smc,
                {
                    staff:
                        getStaff(row),

                    smc:
                        smc,

                    outlet:
                        getOutlet(row),

                    amount:
                        0,

                    count:
                        0
                }
            );

        }


        const data =
            map.get(smc);


        data.amount +=
            amount;


        data.count++;

    });


    return [...map.values()]
        .sort(
            (a, b) =>
                b.amount -
                a.amount
        )
        .slice(0, 3);

}

        let i = 0;
        i < 3;
        i++
    ) {

        const person =
            top[i];


        if (!person) {

            html += `
                <div
                    class="top-card ${classes[i]} top-empty"
                >

                    <div class="rank-circle">
                        ${rank[i]}
                    </div>

                    <div class="place-label">
                        ${labels[i]}
                    </div>

                    <div class="top-name">
                        -
                    </div>

                    <div class="top-meta">
                        No data
                    </div>

                    <div class="top-total">
                        <div class="top-total-label">
                            Approved Incentive
                        </div>

                        <div class="top-total-value">
                            S$0.00
                        </div>
                    </div>

                </div>
            `;

            continue;

        }


        html += `
            <div
                class="top-card ${classes[i]}"
            >

                <div class="rank-circle">
                    ${rank[i]}
                </div>

                <div class="place-label">
                    ${labels[i]}
                </div>

                <div class="top-name">
                    ${escapeHtml(
                        person.staff || "-"
                    )}
                </div>

                <div class="top-meta">
                    SMC:
                    ${escapeHtml(
                        person.smc || "-"
                    )}
                    <br>
                    ${escapeHtml(
                        person.outlet || "-"
                    )}
                </div>

                <div class="top-stat">
                    <span>Submissions</span>
                    <span>
                        ${person.count}
                    </span>
                </div>

                <div class="top-total">

                    <div class="top-total-label">
                        Approved Incentive
                    </div>

                    <div class="top-total-value">
                        ${money(
                            person.amount
                        )}
                    </div>

                </div>

            </div>
        `;

    }


    grid.innerHTML = html;

}

/*
 * Submission View button.
 *
 * This fixes the "Submission ID invalid" issue by reading
 * data-id from the actual button and converting it to a number.
 */

document.getElementById(
    "submissionsTableBody"
).addEventListener(
    "click",
    function(event) {

        const button =
            event.target.closest(
                ".view-btn"
            );


        if (!button) {

            return;

        }


        const id =
            Number(
                button.dataset.id
            );


        if (
            !Number.isFinite(id) ||
            id <= 0
        ) {

            showMessage(
                "Invalid submission ID.",
                "error"
            );

            return;

        }


        showSubmission(id);

    }
);

/* ================================================================
   MONTH CHANGE
================================================================ */

document.getElementById(
    "summaryMonth"
).addEventListener(
    "change",
    async function() {

        state.selectedMonth =
            this.value;


        if (!state.selectedMonth) {

            return;

        }


        /*
         * Try backend monthly_summary first.
         * The local calculation remains available
         * so the page still displays correct data.
         */

        try {

            const parts =
                state.selectedMonth.split("-");


            const year =
                Number(parts[0]);


            const month =
                Number(parts[1]);


            const response =
                await apiRequest(
                    "monthly_summary",
                    {
                        year,
                        month
                    }
                );


            const data =
                response.data ||
                response;


            if (
                Array.isArray(data)
            ) {

                state.monthlySummary =
                    data;

            }

        } catch (error) {

            console.warn(
                "Backend monthly summary unavailable; using local calculation.",
                error
            );

        }


        renderMonthlySummary();

    }
);
