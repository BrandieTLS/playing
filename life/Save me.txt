/*

| RESPONSE HELPERS
|--------------------------------------------------------------------------
*/

function response($data, $statusCode = 200)
{
    http_response_code($statusCode);

    echo json_encode(
        $data,
        JSON_UNESCAPED_UNICODE |
        JSON_UNESCAPED_SLASHES
    );

    exit;
}


function errorResponse($message, $statusCode = 400)
{
    response([
        "success" => false,
        "error" => $message,
        "message" => $message
    ], $statusCode);
}

/*
|--------------------------------------------------------------------------
| SHUTDOWN HANDLER
|--------------------------------------------------------------------------
*/

register_shutdown_function(function () {

    $error = error_get_last();

    if ($error !== null) {

        $fatalTypes = [
            E_ERROR,
            E_PARSE,
            E_CORE_ERROR,
            E_COMPILE_ERROR
        ];

        if (in_array($error['type'], $fatalTypes, true)) {

            if (!headers_sent()) {
                http_response_code(500);
                header("Content-Type: application/json; charset=UTF-8");
            }

            echo json_encode([
                "success" => false,
                "error" => "Fatal server error.",
                "message" => $error['message']
            ]);
        }
    }
});

/*
|--------------------------------------------------------------------------
| JSON INPUT
|--------------------------------------------------------------------------
*/

function getJsonInput()
{
    static $input = null;

    if ($input !== null) {
        return $input;
    }

    $raw = file_get_contents("php://input");

    if (!empty($raw)) {

        $decoded = json_decode($raw, true);

        if (is_array($decoded)) {
            $input = $decoded;
            return $input;
        }
    }

    $input = $_POST;

    return $input;
}


/*
|--------------------------------------------------------------------------
| CLEAN VALUE
|--------------------------------------------------------------------------
*/

function cleanValue($value)
{
    if ($value === null) {
        return null;
    }

    if (is_string($value)) {
        return trim($value);
    }

    return $value;
}


/*
|--------------------------------------------------------------------------
| NORMALIZE PROMOTION TYPE
|--------------------------------------------------------------------------
*/

function normalizePromotionType($type)
{
    $type = strtolower(trim((string)$type));

    $type = str_replace(
        ['-', ' ', '__'],
        ['_', '_', '_'],
        $type
    );

    $map = [

        'individual' => 'individual',

        'outlet_bundle' => 'outlet_bundle',
        'outletbundle' => 'outlet_bundle',
        'bundle' => 'outlet_bundle',

        'outlet_bundle_incentive' => 'outlet_bundle',
        'outletbundleincentive' => 'outlet_bundle'
    ];

    return $map[$type] ?? $type;
}


/*
|--------------------------------------------------------------------------
| PROMOTION STATUS
|--------------------------------------------------------------------------
*/

function getCalculatedPromotionStatus($startDate, $endDate)
{
    if (empty($startDate) || empty($endDate)) {
        return 'active';
    }

    $today = date('Y-m-d');

    if ($today < $startDate) {
        return 'upcoming';
    }

    if ($today > $endDate) {
        return 'expired';
    }

    return 'active';
}


/*
|--------------------------------------------------------------------------
| SYNC PROMOTION STATUSES
|--------------------------------------------------------------------------
*/

function syncPromotionStatuses($mysqli)
{
    $sql = "
        SELECT
            promo_id,
            start_date,
            end_date,
            status
        FROM Incentives_information
    ";

    $result = $mysqli->query($sql);

    if (!$result) {
        return;
    }

    while ($row = $result->fetch_assoc()) {

        $calculatedStatus = getCalculatedPromotionStatus(
            $row['start_date'] ?? '',
            $row['end_date'] ?? ''
        );

        if (
            !empty($row['status']) &&
            strtolower($row['status']) === 'deleted'
        ) {
            continue;
        }

        if ($row['status'] !== $calculatedStatus) {

            $stmt = $mysqli->prepare("
                UPDATE Incentives_information
                SET status = ?
                WHERE promo_id = ?
            ");

            if ($stmt) {

                $stmt->bind_param(
                    "si",
                    $calculatedStatus,
                    $row['promo_id']
                );

                $stmt->execute();
                $stmt->close();
            }
        }
    }

    $result->free();
}

/*
|--------------------------------------------------------------------------
| DASHBOARD
|--------------------------------------------------------------------------
*/

function buildDashboard($submissions, $users)
{
    $total = count($submissions);

    $pending = 0;
    $approved = 0;
    $rejected = 0;

    $individualTotal = 0;
    $bundleTotal = 0;


    foreach ($submissions as $row) {

        $status = strtolower(
            trim($row['status'] ?? '')
        );


        if ($status === 'pending') {
            $pending++;
        }

        if ($status === 'approved') {

            $approved++;

            $amount = (float)(
                $row['amount'] ?? 0
            );

            $type = normalizePromotionType(
                $row['type'] ?? ''
            );


            if ($type === 'individual') {

                $individualTotal += $amount;

            } elseif ($type === 'outlet_bundle') {

                $bundleTotal += $amount;
            }
        }


        if ($status === 'rejected') {
            $rejected++;
        }
    }


    return [

        "totalSubmissions" =>
            $total,

        "total_submissions" =>
            $total,

        "pending" =>
            $pending,

        "approved" =>
            $approved,

        "rejected" =>
            $rejected,


        "individual" =>
            $individualTotal,

        "individualTotal" =>
            $individualTotal,

        "totalIncentive" =>
            $individualTotal,


        "bundle" =>
            $bundleTotal,

        "bundleTotal" =>
            $bundleTotal,


        "staffCount" =>
            count($users)
    ];
}


/*
|--------------------------------------------------------------------------
| MONTHLY SUMMARY
|--------------------------------------------------------------------------
*/

function buildMonthlySummary($submissions)
{
    $summary = [];


    foreach ($submissions as $row) {

        if (
            strtolower(
                trim($row['status'] ?? '')
            ) !== 'approved'
        ) {
            continue;
        }


        $date = $row['submission_date'] ?? '';

        if (empty($date)) {
            continue;
        }


        $timestamp = strtotime($date);

        if ($timestamp === false) {
            continue;
        }


        $month = date(
            'Y-m',
            $timestamp
        );


        $smc = $row['sales_man_code'] ?? '';

        $staff = $row['staff_name'] ?? '';

        $outlet = $row['outlet'] ?? '';

        $type = normalizePromotionType(
            $row['type'] ?? ''
        );

        $amount = (float)(
            $row['amount'] ?? 0
        );

        $quantity = (float)(
            $row['quantity'] ?? 0
        );


        $key =
            $month . '|' .
            $smc . '|' .
            $type;


        if (!isset($summary[$key])) {

            $summary[$key] = [

                "month" =>
                    $month,

                "sales_man_code" =>
                    $smc,

                "SMC" =>
                    $smc,

                "staff_name" =>
                    $staff,

                "outlet" =>
                    $outlet,

                "type" =>
                    $type,

                "quantity" =>
                    0,

                "amount" =>
                    0
            ];
        }


        $summary[$key]['quantity'] +=
            $quantity;

        $summary[$key]['amount'] +=
            $amount;
    }


    return array_values($summary);
}


/*
|--------------------------------------------------------------------------
| TOP 3
|--------------------------------------------------------------------------
*/

function buildTop3($submissions)
{
    $individual = [];
    $bundlePool = 0;


    foreach ($submissions as $row) {

        if (
            strtolower(
                trim($row['status'] ?? '')
            ) !== 'approved'
        ) {
            continue;
        }


        $amount = (float)(
            $row['amount'] ?? 0
        );


        $type = normalizePromotionType(
            $row['type'] ?? ''
        );


        $smc = $row['sales_man_code'] ?? '';

        $staff = $row['staff_name'] ?? '';

        $outlet = $row['outlet'] ?? '';


        if ($type === 'individual') {

            if (!isset($individual[$smc])) {

                $individual[$smc] = [

                    "sales_man_code" =>
                        $smc,

                    "SMC" =>
                        $smc,

                    "staff_name" =>
                        $staff,

                    "outlet" =>
                        $outlet,

                    "amount" =>
                        0
                ];
            }


            $individual[$smc]['amount'] +=
                $amount;


        } elseif ($type === 'outlet_bundle') {

            $bundlePool += $amount;
        }
    }


    usort(
        $individual,
        function ($a, $b) {

            return $b['amount']
                <=> $a['amount'];
        }
    );


    $top3 = array_slice(
        $individual,
        0,
        3
    );


    $shares = [
        0.50,
        0.35,
        0.15
    ];


    foreach ($top3 as $index => &$row) {

        $share =
            $shares[$index]
            ?? 0;

        $row['rank'] =
            $index + 1;

        $row['percentage'] =
            $share * 100;

        $row['payout'] =
            $bundlePool * $share;
    }

    unset($row);


    return [

        "top3" =>
            $top3,

        "individual" =>
            $top3,

        "bundlePool" =>
            $bundlePool,

        "bundle_pool" =>
            $bundlePool,

        "shares" => [

            [
                "rank" => 1,
                "percentage" => 50
            ],

            [
                "rank" => 2,
                "percentage" => 35
            ],

            [
                "rank" => 3,
                "percentage" => 15
            ]
        ]
    ];
}


/*
|--------------------------------------------------------------------------
| STAFF PROMOTION BREAKDOWN
|--------------------------------------------------------------------------
*/

function getStaffPromotionBreakdown(
    $mysqli,
    $sales_man_code
) {

    $sql = "
        SELECT
            si.promo_id,
            ii.promo,
            ii.Rate,
            ii.type,
            SUM(si.quantity) AS quantity,
            SUM(
                si.quantity * ii.Rate
            ) AS amount

        FROM staff_incentives si

        LEFT JOIN Incentives_information ii
            ON ii.promo_id = si.promo_id

        WHERE
            si.sales_man_code = ?

            AND LOWER(
                TRIM(si.status)
            ) = 'approved'

        GROUP BY
            si.promo_id,
            ii.promo,
            ii.Rate,
            ii.type

        ORDER BY
            ii.promo ASC
    ";


    $stmt = $mysqli->prepare($sql);

    if (!$stmt) {

        throw new Exception(
            "Unable to prepare staff breakdown query."
        );
    }


    $stmt->bind_param(
        "s",
        $sales_man_code
    );


    $stmt->execute();


    $result = $stmt->get_result();

    $rows = [];


    while ($row = $result->fetch_assoc()) {

        $quantity = (float)(
            $row['quantity'] ?? 0
        );

        $rate = (float)(
            $row['Rate'] ?? 0
        );

        $amount = (float)(
            $row['amount'] ?? 0
        );


        $type = normalizePromotionType(
            $row['type'] ?? ''
        );


        $rows[] = [

            "promo_id" =>
                (int)$row['promo_id'],

            "promo" =>
                $row['promo'] ?? '',

            "promotion" =>
                $row['promo'] ?? '',

            "Rate" =>
                $rate,

            "rate" =>
                $rate,

            "type" =>
                $type,

            "quantity" =>
                $quantity,

            "amount" =>
                $amount
        ];
    }


    $stmt->close();

    return $rows;
}

/*
|--------------------------------------------------------------------------
| ADD PROMOTION
|--------------------------------------------------------------------------
*/

function addPromotion(
    $mysqli,
    $data
) {

    $promo = cleanValue(
        $data['promo']
        ?? $data['promotion']
        ?? ''
    );


    $rate = (float)(
        $data['Rate']
        ?? $data['rate']
        ?? 0
    );


    $type = normalizePromotionType(
        $data['type']
        ?? $data['promo_type']
        ?? $data['promotion_type']
        ?? ''
    );


    $startDate = cleanValue(
        $data['start_date'] ?? ''
    );


    $endDate = cleanValue(
        $data['end_date'] ?? ''
    );


    if ($promo === '') {
        throw new Exception(
            "Promotion name is required."
        );
    }


    if ($rate < 0) {
        throw new Exception(
            "Rate cannot be negative."
        );
    }


    if (
        $type !== 'individual' &&
        $type !== 'outlet_bundle'
    ) {

        throw new Exception(
            "Invalid promotion type."
        );
    }


    if (
        !empty($startDate) &&
        !empty($endDate) &&
        $startDate > $endDate
    ) {

        throw new Exception(
            "Start date cannot be after end date."
        );
    }


    $status = getCalculatedPromotionStatus(
        $startDate,
        $endDate
    );


    $stmt = $mysqli->prepare("
        INSERT INTO Incentives_information
        (
            promo,
            Rate,
            type,
            start_date,
            end_date,
            status
        )
        VALUES
        (?, ?, ?, ?, ?, ?)
    ");


    if (!$stmt) {

        throw new Exception(
            "Unable to prepare promotion insert."
        );
    }


    $stmt->bind_param(
        "sdssss",
        $promo,
        $rate,
        $type,
        $startDate,
        $endDate,
        $status
    );


    if (!$stmt->execute()) {

        $error = $stmt->error;

        $stmt->close();

        throw new Exception(
            "Unable to add promotion: " .
            $error
        );
    }


    $promoId =
        $mysqli->insert_id;


    $stmt->close();


    /*
    |--------------------------------------------------------------------------
    | BUNDLE ITEMS
    |--------------------------------------------------------------------------
    */

    if ($type === 'outlet_bundle') {

        $items = [];

        for ($i = 1; $i <= 5; $i++) {

            $item = cleanValue(
                $data["item_$i"] ?? ''
            );

            $items[$i] = $item;
        }


        $hasItem = false;

        foreach ($items as $item) {

            if ($item !== '') {
                $hasItem = true;
                break;
            }
        }


        if ($hasItem) {

            $stmt = $mysqli->prepare("
                INSERT INTO Bundle_information
                (
                    promo_id,
                    item_1,
                    item_2,
                    item_3,
                    item_4,
                    item_5
                )
                VALUES
                (?, ?, ?, ?, ?, ?)
            ");


            if (!$stmt) {

                throw new Exception(
                    "Unable to prepare bundle insert."
                );
            }


            $stmt->bind_param(
                "isssss",
                $promoId,
                $items[1],
                $items[2],
                $items[3],
                $items[4],
                $items[5]
            );


            if (!$stmt->execute()) {

                $error = $stmt->error;

                $stmt->close();

                throw new Exception(
                    "Unable to add bundle information: " .
                    $error
                );
            }


            $stmt->close();
        }
    }


    return $promoId;
}

        WHERE
            promo_id = ?

    $stmt->bind_param(
        "sdssssi",

        $rate,
        $type,

    );


    if (!$stmt->execute()) {

        $error = $stmt->error;

            "Unable to edit promotion: " .
            $error
        );
    }


    $stmt->close();

    |--------------------------------------------------------------------------
    */

                $data["item_$i"] ?? ''
            );

        $stmt = $mysqli->prepare("

            FROM Bundle_information
            WHERE promo_id = ?
            LIMIT 1

                "Unable to check bundle information."

                    item_2 = ?,
                    item_3 = ?,
                    item_4 = ?,

                    item_2,
                    item_3,
                    item_4,

                (?, ?, ?, ?, ?, ?)

                throw new Exception(
                    "Unable to prepare bundle insert."

            $stmt->bind_param(
                "isssss",
                $promoId,
                $items[1],
                $items[2],
                $items[3],
                $items[4],
                $items[5]
            );


            $stmt->execute();

            $stmt->close();
        }


    } else {

        /*
        |--------------------------------------------------------------------------
        | INDIVIDUAL PROMOTION
        |--------------------------------------------------------------------------

        | from bundle to individual.
        */

        $stmt = $mysqli->prepare("
            DELETE FROM Bundle_information
            WHERE promo_id = ?
        ");


        if ($stmt) {

            $stmt->bind_param(
                "i",
                $promoId
            );

            $stmt->execute();

            $stmt->close();
        }
    }


    return true;

                "Quantity must be greater than zero."
            );
        }


        editSubmission(
            $mysqli,
            $incentiveId,
            $promoId,
            $quantity
        );


        response([

            "success" => true,

            "message" =>
                "Submission updated successfully."
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | DELETE SUBMISSION
    |--------------------------------------------------------------------------
    */

    case 'delete_submission':

        $data = getJsonInput();


        $incentiveId = (int)(
            $data['incentive_id']
            ?? $data['id']
            ?? $_GET['incentive_id']
            ?? $_GET['id']
            ?? 0
        );


        if ($incentiveId <= 0) {

            errorResponse(
                "Invalid submission ID."
            );
        }


        deleteSubmission(
            $mysqli,
            $incentiveId
        );


        response([

            "success" => true,

            "message" =>
                "Submission deleted successfully."
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | GET PROMOTIONS
    |--------------------------------------------------------------------------
    */

    case 'get_promotions':
    case 'promotions':

        $promotions =
            fetchPromotions($mysqli);


        response([

            "success" => true,

            "data" =>
                $promotions,

            "promotions" =>
                $promotions
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | ADD PROMOTION
    |--------------------------------------------------------------------------
    */

    case 'add_promotion':

        $data =
            getJsonInput();


        $promoId =
            addPromotion(
                $mysqli,
                $data
            );


        response([

            "success" => true,

            "message" =>
                "Promotion added successfully.",

            "promo_id" =>
                $promoId
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | EDIT PROMOTION
    |--------------------------------------------------------------------------
    */

    case 'edit_promotion':

        $data =
            getJsonInput();


        editPromotion(
            $mysqli,
            $data
        );


        response([

            "success" => true,

            "message" =>
                "Promotion updated successfully."
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | DELETE PROMOTION
    |--------------------------------------------------------------------------
    */

    case 'delete_promotion':

        $data =
            getJsonInput();


        $promoId = (int)(
            $data['promo_id']
            ?? $data['id']
            ?? $_GET['promo_id']
            ?? $_GET['id']
            ?? 0
        );


        if ($promoId <= 0) {

            errorResponse(
                "Invalid promotion ID."
            );
        }


        deletePromotion(
            $mysqli,
            $promoId
        );


        response([

            "success" => true,

            "message" =>
                "Promotion deleted successfully."
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | GET USERS
    |--------------------------------------------------------------------------
    */

    case 'get_users':
    case 'users':

        $users =
            fetchUsers($mysqli);


        response([

            "success" => true,

            "data" =>
                $users,

            "users" =>
                $users
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | ADD USER
    |--------------------------------------------------------------------------
    */

    case 'add_user':

        $data =
            getJsonInput();


        addUser(
            $mysqli,
            $data
        );


        response([

            "success" => true,

            "message" =>
                "User added successfully."
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | DELETE USER
    |--------------------------------------------------------------------------
    */

    case 'delete_user':

        $data =
            getJsonInput();


        $smc = cleanValue(
            $data['sales_man_code']
            ?? $data['SMC']
            ?? $data['smc']
            ?? $_GET['sales_man_code']
            ?? $_GET['SMC']
            ?? ''
        );


        if ($smc === '') {

            errorResponse(
                "Sales Man Code is required."
            );
        }


        deleteUser(
            $mysqli,
            $smc
        );


        response([

            "success" => true,

            "message" =>
                "User deleted successfully."
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | MONTHLY SUMMARY
    |--------------------------------------------------------------------------
    */

    case 'monthly_summary':
    case 'get_monthly_summary':

        $submissions =
            fetchSubmissions($mysqli);


        $summary =
            buildMonthlySummary(
                $submissions
            );


        response([

            "success" => true,

            "data" =>
                $summary,

            "monthlySummary" =>
                $summary,

            "monthly_summary" =>
                $summary
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | TOP 3
    |--------------------------------------------------------------------------
    */

    case 'top3':
    case 'get_top3':

        $submissions =
            fetchSubmissions($mysqli);


        $top3 =
            buildTop3(
                $submissions
            );


        response([

            "success" => true,

            "data" =>
                $top3,

            "top3" =>
                $top3
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | CURRENT ADMIN
    |--------------------------------------------------------------------------
    */

    case 'get_current_admin':

        $admin =
            getCurrentAdmin();


        /*
        |--------------------------------------------------------------------------
        | IMPORTANT
        |--------------------------------------------------------------------------
        | Return the admin information in BOTH:
        |
        | result.data
        | result.admin
        |
        | AND flat at the root.
        |
        | This makes it compatible with the current admin.html:
        |
        | const data =
        |     result && result.data !== undefined
        |         ? result.data
        |         : result;
        |
        */

        response([

            "success" => true,

            "data" =>
                $admin,

            "admin" =>
                $admin,


            "staff_name" =>
                $admin['staff_name'],

            "name" =>
                $admin['name'],

            "admin_name" =>
                $admin['admin_name'],


            "sales_man_code" =>
                $admin['sales_man_code'],

            "SMC" =>
                $admin['SMC'],

            "smc" =>
                $admin['smc'],


            "outlet" =>
                $admin['outlet'],

            "role" =>
                $admin['role']
        ]);

        break;


    /*
    |--------------------------------------------------------------------------
    | DEFAULT
    |--------------------------------------------------------------------------
    */

    default:

        errorResponse(
            "Invalid or missing action.",
            400
        );
}


$mysqli->close();
