async function apiRequest(action, data = {}) {

    const response = await fetch(
        `${ADMIN_API_FILE}?action=${encodeURIComponent(action)}`,
        {
            method: "POST",
            credentials: "same-origin",
            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },
            body: JSON.stringify(data)
        }
    );

    const text = await response.text();

    let json;

    try {

        json = text ? JSON.parse(text) : {};

    } catch (e) {

        throw new Error(
            text ||
            `Server returned HTTP ${response.status}`
        );

    }

    if (!response.ok || json.success === false) {

        throw new Error(
            json.message ||
            json.error ||
            `Request failed (${response.status})`
        );

    }

    return json;

}


/* =========================================================

function escapeHtml(value) {

    if (value === null || value === undefined) {
        return "";
    }

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");

}


function money(value) {

    const number = Number(value || 0);

    return "$" + number.toFixed(2);

}


function getPromoId(x) {

    const raw =
        x?.promo_id ??
        x?.promotion_id ??
        x?.promoId ??
        x?.id;

    const n = Number(raw);

    return Number.isFinite(n) ? n : null;

}


function getSubmissionId(x) {

    const raw =
        x?.incentive_id ??
        x?.submission_id ??
        x?.id ??
        x?.incentiveId;

    if (
        raw === null ||
        raw === undefined ||
        String(raw).trim() === ""
    ) {
        return null;
    }

    const n = Number(raw);

    return Number.isFinite(n) && n > 0
        ? n
        : null;

}

function getDateOnly(value) {

    if (!value) {
        return "";
    }

    const stringValue = String(value).trim();

    const match = stringValue.match(
        /^(\d{4}-\d{2}-\d{2})/
    );

    if (match) {
        return match[1];
    }

    const date = new Date(stringValue);

    if (Number.isNaN(date.getTime())) {
        return "";
    }

    return [
        date.getFullYear(),
        String(date.getMonth() + 1).padStart(2, "0"),
        String(date.getDate()).padStart(2, "0")
    ].join("-");

}


function todayString() {

    const date = new Date();

    return [
        date.getFullYear(),
        String(date.getMonth() + 1).padStart(2, "0"),
        String(date.getDate()).padStart(2, "0")
    ].join("-");

}


function currentMonthString() {

    const date = new Date();

    return [
        date.getFullYear(),
        String(date.getMonth() + 1).padStart(2, "0")
    ].join("-");

}

function getPromoTypeValue(type) {

    const value = String(type || "")
        .trim()
        .toLowerCase()
        .replace(/[\s_-]/g, "");

    if (
        value === "bundle" ||
        value === "outletbundle"
    ) {
        return "Bundle";
    }

    if (value === "individual") {
        return "Individual";
    }

    return "";

}


function getType(x) {

    const direct = String(
        x?.type ??
        x?.promotion_type ??
        x?.promo_type ??
        x?.promotionType ??
        ""
    ).trim();

    const promoId = getPromoId(x);

    if (promoId !== null) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {

            const promoType = String(
                promo?.type ??
                promo?.promotion_type ??
                promo?.promo_type ??
                promo?.promotionType ??
                ""
            ).trim();

            if (promoType) {
                return promoType;
            }

        }

    }

    if (direct) {
        return direct;
    }

    return "";

}


function normalizedType(x) {

    return getType(x)
        .toLowerCase()
        .replace(/[\s_-]/g, "");

}


function isBundle(x) {

    const type = normalizedType(x);

    return (
        type === "bundle" ||
        type === "outletbundle"
    );

}


function displayType(x) {

    if (isBundle(x)) {
        return "Bundle";
    }

    const type = getType(x);

    return type || "Individual";

}

   PROMOTION STATUS
========================================================= */

function getPromotionStatus(p) {

    const direct = String(
        p?.status ??
        p?.promotion_status ??
        p?.promo_status ??
        ""
    ).trim();

    if (direct) {

        const n = direct.toLowerCase();

        if (n === "active") {
            return "Active";
        }

        if (n === "upcoming") {
            return "Upcoming";
        }

        if (
            n === "expired" ||
            n === "inactive"
        ) {
            return "Expired";
        }

    }

    const start = getDateOnly(
        p?.start_date ??
        p?.startDate
    );

    const end = getDateOnly(
        p?.end_date ??
        p?.endDate
    );

    const today = todayString();

    if (start && today < start) {
        return "Upcoming";
    }

    if (end && today > end) {
        return "Expired";
    }

    return "Active";

}


function statusClass(status) {

    const s = String(status || "")
        .toLowerCase();

    if (s === "approved") {
        return "status-approved";
    }

    if (s === "active") {
        return "status-active";
    }

    if (
        s === "pending" ||
        s === "upcoming"
    ) {
        return "status-pending";
    }

    if (
        s === "rejected" ||
        s === "expired"
    ) {
        return "status-rejected";
    }

    return "status-default";

}


function statusBadge(status) {

    return `

}

            : [];

            ? d.top3
            : [];

}


/* =========================================================
   REFRESH
========================================================= */

async function refreshAdminData() {

    if (refreshInProgress) {
        return;
    }

    refreshInProgress = true;

    try {

        await loadAdminData();

        renderAll();

        if (
            state.currentSubmissionId !== null
        ) {

            const exists =
                state.submissions.some(
                    s =>
                        getSubmissionId(s) ===
                        state.currentSubmissionId
                );

            if (!exists) {

                state.currentSubmissionId = null;

                closeModal("viewModal");

            }

        }

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to refresh data.",
            "error"
        );

    } finally {

        refreshInProgress = false;

    }

}

function calculateApprovedTotals() {

    let individual = 0;
    let bundle = 0;

    let pending = 0;
    let approved = 0;
    let rejected = 0;

    let individualQty = 0;
    let bundleQty = 0;

    let individualCount = 0;
    let bundleCount = 0;

    state.submissions.forEach(submission => {

        const status =
            getStatus(submission)
                .toLowerCase();

        const quantity =
            Number(
                submission.quantity ??
                0
            );

        const rate =
            Number(
                submission.Rate ??
                submission.rate ??
                0
            );

        const amount =
            quantity * rate;

        if (status === "pending") {
            pending++;
        }

        if (status === "approved") {

            approved++;

            if (isBundle(submission)) {

                bundle += amount;

                bundleQty += quantity;

                bundleCount++;

            } else {

                individual += amount;

                individualQty += quantity;

                individualCount++;

            }

        }

        if (status === "rejected") {
            rejected++;
        }

    });

    return {
        individual,
        bundle,
        total: individual + bundle,
        pending,
        approved,
        rejected,
        individualQty,
        bundleQty,
        individualCount,
        bundleCount
    };

}

function findStaffBySMC(smc) {

    return state.users.find(user => {

        const code = String(
            user.sales_man_code ??
            user.SMC ??
            user.smc ??
            ""
        ).trim();

/* =========================================================
   SUBMISSION FILTER OPTIONS
========================================================= */

function populateSubmissionFilters() {

    const staffSelect =
        document.getElementById(
            "submissionStaffFilter"
        );

    const outletSelect =
        document.getElementById(
            "submissionOutletFilter"
        );

    const previousStaff =
        staffSelect.value;

    const previousOutlet =
        outletSelect.value;


    const staffMap = new Map();

    state.submissions.forEach(submission => {

        const smc =
            getStaffCode(submission);

        const staff =
            findStaffBySMC(smc);

        const name =
            getStaffName(submission) ||
            getStaffName(staff) ||
            smc;

        if (smc) {
            staffMap.set(smc, {
                smc,
                name
            });
        }

    });


    const staffOptions =
        Array.from(staffMap.values())
            .sort((a, b) =>
                a.name.localeCompare(b.name)
            );


    staffSelect.innerHTML =
        `<option value="">All Staff</option>` +
        staffOptions.map(staff => `
            <option value="${escapeHtml(staff.smc)}">
                ${escapeHtml(staff.name)}
                (${escapeHtml(staff.smc)})
            </option>
        `).join("");


    const outlets =
        Array.from(
            new Set(
                state.submissions
                    .map(getOutlet)
                    .filter(Boolean)
            )
        ).sort();


    outletSelect.innerHTML =
        `<option value="">All Outlets</option>` +
        outlets.map(outlet => `
            <option value="${escapeHtml(outlet)}">
                ${escapeHtml(outlet)}
            </option>
        `).join("");


    if (
        Array.from(staffSelect.options)
            .some(option =>
                option.value === previousStaff
            )
    ) {
        staffSelect.value = previousStaff;
    }


    if (
        Array.from(outletSelect.options)
            .some(option =>
                option.value === previousOutlet
            )
    ) {
        outletSelect.value = previousOutlet;
    }

}


/* =========================================================
   SUBMISSION FILTER
========================================================= */

function getFilteredSubmissions() {

    const search =
        document
            .getElementById("submissionSearch")
            .value
            .trim()
            .toLowerCase();

    const staff =
        document
            .getElementById("submissionStaffFilter")
            .value
            .trim();

    const outlet =
        document
            .getElementById("submissionOutletFilter")
            .value
            .trim();

    const status =
        document
            .getElementById("submissionStatusFilter")
            .value
            .trim()
            .toLowerCase();

    const date =
        document
            .getElementById("submissionDateFilter")
            .value
            .trim();

    const type =
        document
            .getElementById("submissionTypeFilter")
            .value
            .trim();


    return state.submissions.filter(submission => {

        const id =
            String(
                getSubmissionId(submission) ?? ""
            );

        const smc =
            getStaffCode(submission);

        const staffRecord =
            findStaffBySMC(smc);

        const staffName =
            getStaffName(submission) ||
            getStaffName(staffRecord);

        const promotion =
            String(
                submission.promo ??
                submission.promotion ??
                submission.promo_name ??
                submission.promotion_name ??
                ""
            );

        const submissionOutlet =
            getOutlet(submission);

        const submissionStatus =
            getStatus(submission)
                .toLowerCase();

        const submissionDate =
            getDateOnly(
                getSubmissionDate(submission)
            );

        const submissionType =
            displayType(submission);


        if (search) {

            const searchable = [

                id,
                smc,
                staffName,
                submissionOutlet,
                promotion,
                submissionType,
                submissionStatus

            ]
                .join(" ")
                .toLowerCase();

            if (!searchable.includes(search)) {
                return false;
            }

        }


        if (
            staff &&
            smc !== staff
        ) {
            return false;
        }


        if (
            outlet &&
            submissionOutlet !== outlet
        ) {
            return false;
        }


        if (
            status &&
            submissionStatus !== status
        ) {
            return false;
        }


        if (
            date &&
            submissionDate !== date
        ) {
            return false;
        }


        if (
            type &&
            submissionType !== type
        ) {
            return false;
        }


        return true;

    });

}

async function showSubmission(id) {

    const numericId =
        Number(id);

    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    state.currentSubmissionId =
        numericId;


    let submission =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    try {

        const response =
            await apiRequest(
                "get_submission",
                {
                    incentive_id: numericId,
                    submission_id: numericId
                }
            );

        const serverSubmission =
            response.data?.submission ||
            response.submission ||
            response.data;

        if (serverSubmission) {
            submission = serverSubmission;
        }

    } catch (err) {

        if (!submission) {

            showMessage(
                err.message ||
                "Failed to load submission.",
                "error"
            );

            return;
        }

    }


    if (!submission) {

        showMessage(
            "Submission not found.",
            "error"
        );

        return;
    }


    renderSubmissionDetails(submission);

    openModal("viewModal");

}

/* =========================================================
   REJECT
========================================================= */

async function rejectSubmission(id) {

    const numericId =
        Number(id);

    if (
        !Number.isFinite(numericId) ||
        numericId <= 0
    ) {

        showMessage(
            "Invalid submission ID.",
            "error"
        );

        return;
    }


    const current =
        state.submissions.find(
            x =>
                getSubmissionId(x) ===
                numericId
        );


    if (
        current &&
        getStatus(current).toLowerCase() !==
        "pending"
    ) {

        showMessage(
            "Only pending submissions can be rejected.",
            "error"
        );

        return;
    }


    const reason =
        window.prompt(
            "Enter rejection reason:"
        );


    if (reason === null) {
        return;
    }


    const trimmed =
        reason.trim();


    if (!trimmed) {

        showMessage(
            "Rejection reason is required.",
            "error"
        );

        return;
    }


    try {

        await apiRequest(
            "reject",
            {
                incentive_id: numericId,
                rejection_reason: trimmed,
                reason: trimmed
            }
        );

        showMessage(
            "Submission rejected.",
            "success"
        );

        closeModal("viewModal");

        await refreshAdminData();

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to reject submission.",
            "error"
        );

    }

}

function getMonthlySummaryRows(month) {

    const map = {};


    state.submissions.forEach(submission => {

        if (
            getStatus(submission)
                .toLowerCase() !==
            "approved"
        ) {
            return;
        }


        const date =
            getDateOnly(
                getSubmissionDate(submission)
            );


        if (
            !date ||
            !date.startsWith(month)
        ) {
            return;
        }


        const smc =
            getStaffCode(submission);


        if (!smc) {
            return;
        }


        const staff =
            findStaffBySMC(smc);


        const name =
            getStaffName(submission) ||
            getStaffName(staff) ||
            smc;


        const outlet =
            getOutlet(submission) ||
            getOutlet(staff) ||
            "";


        const quantity =
            Number(
                submission.quantity ?? 0
            );


        const rate =
            Number(
                submission.Rate ??
                submission.rate ??
                0
            );


        const amount =
            quantity * rate;


        if (!map[smc]) {

            map[smc] = {

                name,

                smc,

                outlet,

                individualQty: 0,

                individualAmount: 0,

                bundleQty: 0,

                bundleAmount: 0

            };

        }


        if (isBundle(submission)) {

            map[smc].bundleQty +=
                quantity;

            map[smc].bundleAmount +=
                amount;

        } else {

            map[smc].individualQty +=
                quantity;

            map[smc].individualAmount +=
                amount;

        }

    });


    return Object.values(map);

}

        `;

    body.innerHTML =

            const total =

/* =========================================================
   TOP 3
========================================================= */

function getAllApprovedBundleSales(month) {

    const totals = {};


    state.submissions.forEach(submission => {

        if (
            getStatus(submission)
                .toLowerCase() !==
            "approved"
        ) {
            return;
        }


        if (!isBundle(submission)) {
            return;
        }


        const date =
            getDateOnly(
                getSubmissionDate(submission)
            );


        if (
            !date ||
            !date.startsWith(month)
        ) {
            return;
        }


        const smc =
            getStaffCode(submission);


        if (!smc) {
            return;
        }


        const quantity =
            Number(
                submission.quantity ?? 0
            );


        const rate =
            Number(
                submission.Rate ??
                submission.rate ??
                0
            );


        const amount =
            quantity * rate;


        totals[smc] =
            (totals[smc] || 0) +
            amount;

    });


    return totals;

}


function getTop3BundleSales(month) {

    const staffTotals = {};


    state.submissions.forEach(submission => {

        const status =
            getStatus(submission)
                .toLowerCase();


        if (status !== "approved") {
            return;
        }


        if (!isBundle(submission)) {
            return;
        }


        const date =
            getDateOnly(
                getSubmissionDate(submission)
            );


        if (
            !date ||
            !date.startsWith(month)
        ) {
            return;
        }


        const smc =
            getStaffCode(submission);


        if (!smc) {
            return;
        }


        const staff =
            findStaffBySMC(smc);


        const staffName =
            getStaffName(submission) ||
            getStaffName(staff) ||
            smc;


        const outlet =
            getOutlet(submission) ||
            getOutlet(staff) ||
            "";


        const quantity =
            Number(
                submission.quantity ?? 0
            );


        const rate =
            Number(
                submission.Rate ??
                submission.rate ??
                0
            );


        const amount =
            quantity * rate;


        if (!staffTotals[smc]) {

            staffTotals[smc] = {

                smc: smc,

                name: staffName,

                outlet: outlet,

                sales: 0

            };

        }


        staffTotals[smc].sales +=
            amount;

    });


    return Object.values(staffTotals)
        .sort(
            (a, b) =>
                b.sales - a.sales
        )
        .slice(0, 3);

}


function calculateTop3Payouts(month) {

    const top3 =
        getTop3BundleSales(month);


    const allSales =
        getAllApprovedBundleSales(month);


    const totalBundleSales =
        Object.values(allSales)
            .reduce(
                (total, value) =>
                    total + Number(value || 0),
                0
            );


    const percentages = [
        0.50,
        0.35,
        0.15
    ];


    return {

        totalBundleSales,

        top3: top3.map(
            (person, index) => ({

                rank:
                    index + 1,

                name:
                    person.name,

                smc:
                    person.smc,

                outlet:
                    person.outlet,

                sales:
                    person.sales,

                percentage:
                    percentages[index] ||
                    0,

                payout:
                    totalBundleSales *
                    (
                        percentages[index] ||
                        0
                    )

            })
        )

    };

}


function renderTop3() {

    const month =
        document.getElementById(
            "top3Month"
        ).value ||
        currentMonthString();


    if (!document.getElementById(
        "top3Month"
    ).value) {

        document.getElementById(
            "top3Month"
        ).value =
            month;

    }


    const result =
        calculateTop3Payouts(month);


    document.getElementById(
        "top3TotalSales"
    ).textContent =
        money(
            result.totalBundleSales
        );


    renderTop3Podium(
        result.top3
    );


    const body =
        document.getElementById(
            "top3TableBody"
        );


    if (!result.top3.length) {

        body.innerHTML = `
            <tr>
                <td colspan="7" class="empty">
                    No approved outlet bundle sales for ${escapeHtml(month)}.
                </td>
            </tr>
        `;

        return;
    }


    body.innerHTML =
        result.top3.map(person => {

            return `

                <tr>

                    <td>
                        ${person.rank}
                    </td>

                    <td>
                        ${escapeHtml(person.name)}
                    </td>

                    <td>
                        ${escapeHtml(person.smc)}
                    </td>

                    <td>
                        ${escapeHtml(person.outlet)}
                    </td>

                    <td class="money">
                        ${money(person.sales)}
                    </td>

                    <td>
                        ${(person.percentage * 100).toFixed(0)}%
                    </td>

                    <td class="money">
                        ${money(person.payout)}
                    </td>

                </tr>

            `;

        }).join("");

}
