/* ================================================================
   BASIC HELPERS
================================================================ */

function escapeHtml(value) {

    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");

}


function money(value) {

    const n = Number(value) || 0;

    return "S$" + n.toFixed(2);

}


function getSubmissionId(x) {

    const raw =
        x?.incentive_id ??
        x?.submission_id ??
        x?.id ??
        x?.incentiveId;

    if (
        raw === null ||
        raw === undefined ||
        String(raw).trim() === ""
    ) {
        return null;
    }

    const n = Number(raw);

    return Number.isFinite(n) && n > 0 ? n : null;

}


function getPromoId(x) {

    const raw =
        x?.promo_id ??
        x?.promotion_id ??
        x?.promoId;

    const n = Number(raw);

    return Number.isFinite(n) ? n : null;

}


function getStaff(x) {

    return String(
        x?.staff_name ??
        x?.name ??
        x?.staff ??
        ""
    );

}


function getSMC(x) {

    return String(
        x?.sales_man_code ??
        x?.SMC ??
        x?.smc ??
        ""
    );

}


function getOutlet(x) {

    return String(
        x?.outlet ??
        x?.outlet_name ??
        ""
    );

}


function getPromo(x) {

    return String(
        x?.promo ??
        x?.promotion ??
        x?.promo_name ??
        x?.promotion_name ??
        ""
    );

}


function getQty(x) {

    return Number(
        x?.quantity ??
        x?.qty ??
        0
    ) || 0;

}


function getRate(x) {

    return Number(
        x?.Rate ??
        x?.rate ??
        x?.incentive_rate ??
        x?.promo_rate ??
        0
    ) || 0;

}


function getStatus(x) {

    return String(
        x?.status ??
        x?.submission_status ??
        ""
    ).trim();

}


function getDate(x) {

    return String(
        x?.submission_date ??
        x?.submitted_at ??
        x?.created_at ??
        x?.date ??
        ""
    );

}


/* ================================================================
   PROMOTION TYPE
   IMPORTANT:
   If submission does not contain type, resolve it using promo_id.
================================================================ */

function getType(x) {

    const direct = String(
        x?.type ??
        x?.promotion_type ??
        x?.promo_type ??
        x?.promotionType ??
        ""
    ).trim();

    if (direct) {
        return direct;
    }

    const promoId = getPromoId(x);

    if (promoId !== null) {

        const promo = state.promotions.find(
            p => getPromoId(p) === promoId
        );

        if (promo) {

            return String(
                promo.type ??
                promo.promotion_type ??
                promo.promo_type ??
                ""
            ).trim();

        }

    }

    return "";

}


function isBundle(x) {

    return getType(x)
        .toLowerCase()
        .replace(/[\s_-]/g, "") === "bundle";

}


/* ================================================================
   PROMOTION STATUS
================================================================ */

function normalizeDate(value) {

    if (!value) {
        return "";
    }

    const str = String(value).trim();

    /*
     * Only accept a proper SQL/HTML date.
     * This prevents values such as "2026" from being
     * accidentally sent/displayed as a complete date.
     */

    if (/^\d{4}-\d{2}-\d{2}/.test(str)) {
        return str.substring(0, 10);
    }

    return "";

}


function getPromotionStart(promo) {

    return normalizeDate(
        promo?.start_date ??
        promo?.startDate ??
        promo?.date_start ??
        ""
    );

}


function getPromotionEnd(promo) {

    return normalizeDate(
        promo?.end_date ??
        promo?.endDate ??
        promo?.date_end ??
        ""
    );

}


function getPromotionStatus(promo) {

    const start = getPromotionStart(promo);
    const end = getPromotionEnd(promo);

    /*
     * No dates = Active
     */

    if (!start && !end) {
        return "active";
    }

    const today = new Date();

    const todayString =
        today.getFullYear() +
        "-" +
        String(today.getMonth() + 1).padStart(2, "0") +
        "-" +
        String(today.getDate()).padStart(2, "0");

    if (start && todayString < start) {
        return "upcoming";
    }

    if (end && todayString > end) {
        return "expired";
    }

    return "active";

}


function promotionStatusBadge(promo) {

    const status = getPromotionStatus(promo);

    let cls = "status-default";

    if (status === "active") {
        cls = "status-active";
    }

    if (status === "upcoming") {
        cls = "status-upcoming";
    }

    if (status === "expired") {
        cls = "status-expired";
    }

    return `
        <span class="status ${cls}">
            ${escapeHtml(
                status.charAt(0).toUpperCase() + status.slice(1)
            )}
        </span>
    `;

}


/* ================================================================
   API
================================================================ */

async function apiRequest(action, data = {}) {

    const response = await fetch(
        `admin.php?action=${encodeURIComponent(action)}`,
        {
            method: "POST",
            credentials: "same-origin",
            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },
            body: JSON.stringify(data)
        }
    );

    const text = await response.text();

    let json;

    try {

        json = text ? JSON.parse(text) : {};

    } catch (error) {

        throw new Error(
            text ||
            `Server returned HTTP ${response.status}`
        );

    }

    if (!response.ok || json.success === false) {

        throw new Error(
            json.message ||
            json.error ||
            `Request failed (${response.status})`
        );

    }

    return json;

}

async function loadAdminData() {

    const response = await apiRequest("bootstrap");

    const d = response.data || response;

    state.admin = d.admin || {};

    state.dashboard =
        Array.isArray(d.dashboard)
            ? {}
            : (d.dashboard || {});

    state.submissions =
        Array.isArray(d.submissions)
            ? d.submissions
            : [];

    state.promotions =
        Array.isArray(d.promotions)
            ? d.promotions
            : [];

    state.users =
        Array.isArray(d.users)
            ? d.users
            : [];

    state.monthlySummary =
        Array.isArray(d.monthlySummary)
            ? d.monthlySummary
            : [];

    state.top3 =
        Array.isArray(d.top3)
            ? d.top3
            : [];

    if (state.admin?.name) {
        document.getElementById("adminName").textContent =
            state.admin.name;
    }

}


/* ================================================================
   REFRESH
================================================================ */

async function refreshAdminData() {

    if (refreshInProgress) {
        return;
    }

    refreshInProgress = true;

    try {

        await loadAdminData();

        renderAll();

        populateSubmissionFilters();

        if (state.currentSubmissionId !== null) {

            const exists =
                state.submissions.some(
                    s =>
                        getSubmissionId(s) ===
                        state.currentSubmissionId
                );

            if (!exists) {

                state.currentSubmissionId = null;

                closeModal("viewModal");

            }

        }

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to refresh data.",
            "error"
        );

    } finally {

        refreshInProgress = false;

    }

}

/* ================================================================
   DASHBOARD CALCULATIONS
================================================================ */

function calculateApprovedTotals() {

    let individual = 0;
    let bundle = 0;

    let pending = 0;
    let approved = 0;
    let rejected = 0;

    for (const submission of state.submissions) {

        const status =
            getStatus(submission)
                .toLowerCase();

        const amount =
            getQty(submission) *
            getRate(submission);

        if (status === "approved") {

            if (isBundle(submission)) {
                bundle += amount;
            } else {
                individual += amount;
            }

        }

        if (status === "pending") {
            pending++;
        }

        if (status === "approved") {
            approved++;
        }

        if (status === "rejected") {
            rejected++;
        }

    }

    return {
        individual,
        bundle,
        total: individual + bundle,
        pending,
        approved,
        rejected
    };

}

    document.getElementById("rejectedCount").textContent =

            uniqueSMC.add(smc);

        const smc = getSMC(submission);

        if (smc) {
            uniqueSMC.add(smc);
        }

        uniqueSMC.size;

    const recent =
        [...state.submissions]
            .sort((a, b) =>
                getSubmissionId(b) -
                getSubmissionId(a)

            .slice(0, 10);

    const body =

    if (!recent.length) {

        body.innerHTML = `
            <tr>
                <td colspan="6" class="empty">
                    No submissions found.
                </td>
            </tr>
        `;

        return;
    }

    body.innerHTML = recent.map(row => {

        const id = getSubmissionId(row);

        const status =
            getStatus(row).toLowerCase();

        const statusClass =
            status === "approved"
                ? "status-approved"
                : status === "pending"
                    ? "status-pending"
                    : status === "rejected"
                        ? "status-rejected"
                        : "status-default";

        const type =
            isBundle(row)
                ? "Bundle"
                : getType(row) || "Individual";

        outletSelect.value = oldOutlet;

        document.getElementById(
            "submissionPromoFilter"
        ).value;

    const rows =
        state.submissions.filter(row => {

            const staff =
                getStaff(row).toLowerCase();

            const smc =

                getPromo(row).toLowerCase();

                getOutlet(row);

                getStatus(row).toLowerCase();

                (
                    isBundle(row)
                        ? "bundle"
                        : getType(row).toLowerCase() ||
                          "individual"

                !search ||
                staff.includes(search) ||
                smc.includes(search) ||
                promo.includes(search);

/* ================================================================
   SAVE PROMOTION
================================================================ */

async function savePromotionEdit(event) {

    event.preventDefault();

    const id =
        Number(
            document.getElementById(
                "editPromotionId"
            ).value
        );

    if (
        !Number.isFinite(id) ||
        id <= 0
    ) {

        showMessage(
            "Invalid promotion ID.",
            "error"
        );

        return;

    }

    const promo =
        document.getElementById(
            "editPromotionName"
        ).value.trim();

    const rate =
        Number(
            document.getElementById(
                "editPromotionRate"
            ).value
        );

    const type =
        document.getElementById(
            "editPromotionType"
        ).value;

    const startDate =
        document.getElementById(
            "editPromotionStartDate"
        ).value;

    const endDate =
        document.getElementById(
            "editPromotionEndDate"
        ).value;

    if (!promo) {

        showMessage(
            "Promotion name is required.",
            "error"
        );

        return;

    }

    if (
        !Number.isFinite(rate) ||
        rate < 0
    ) {

        showMessage(
            "Please enter a valid rate.",
            "error"
        );

        return;

    }

    if (
        startDate &&
        endDate &&
        endDate < startDate
    ) {

        showMessage(
            "End date cannot be before start date.",
            "error"
        );

        return;

    }

    const payload = {

        incentive_id: id,

        promo_id: id,

        promo: promo,

        Rate: rate,

        rate: rate,

        type: type,

        start_date: startDate || null,

        end_date: endDate || null

    };

    if (type === "Bundle") {

        const item1 =
            document.getElementById(
                "editBundleItem1"
            ).value.trim();

        if (!item1) {

            showMessage(
                "Item 1 is required.",
                "error"
            );

            return;

        }

        payload.item_1 = item1;

        payload.item_2 =
            document.getElementById(
                "editBundleItem2"
            ).value.trim();

        payload.item_3 =
            document.getElementById(
                "editBundleItem3"
            ).value.trim();

        payload.item_4 =
            document.getElementById(
                "editBundleItem4"
            ).value.trim();

        payload.item_5 =
            document.getElementById(
                "editBundleItem5"
            ).value.trim();

    }

    try {

        /*
         * First try the standard update_promotion endpoint.
         * The payload contains bundle fields too, so a backend
         * that updates bundle_information can save everything.
         */

        try {

            await apiRequest(
                "update_promotion",
                payload
            );

        } catch (firstError) {

            /*
             * If the backend separates bundle updates,
             * try update_bundle.
             */

            if (type === "Bundle") {

                await apiRequest(
                    "update_bundle",
                    payload
                );

            } else {

                throw firstError;

            }

        }

        showMessage(
            "Promotion updated successfully.",
            "success"
        );

        closeModal("editPromotionModal");

        await refreshAdminData();

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to update promotion.",
            "error"
        );

    }

}

/* ================================================================
   MONTHLY SUMMARY
================================================================ */

function getSelectedMonth() {

    const input =
        document.getElementById(
            "summaryMonth"
        );

    if (input.value) {
        return input.value;
    }

    const now = new Date();

    return (
        now.getFullYear() +
        "-" +
        String(
            now.getMonth() + 1
        ).padStart(2, "0")
    );

}


function calculateMonthlySummary() {

    const month =
        getSelectedMonth();

    const grouped = new Map();

    let individual = 0;
    let bundle = 0;

    state.submissions.forEach(row => {

        if (
            getStatus(row).toLowerCase() !==
            "approved"
        ) {
            return;
        }

        const date =
            getDate(row);

        if (!date.startsWith(month)) {
            return;
        }

        const amount =
            getQty(row) *
            getRate(row);

        if (isBundle(row)) {
            bundle += amount;
        } else {
            individual += amount;
        }

        const key = [
            getSMC(row),
            getPromoId(row)
        ].join("|");

        if (!grouped.has(key)) {

            grouped.set(key, {

                staff: getStaff(row),

                smc: getSMC(row),

                outlet: getOutlet(row),

                promo: getPromo(row),

                type:
                    isBundle(row)
                        ? "Bundle"
                        : getType(row) ||
                          "Individual",

                quantity: 0,

                rate: getRate(row),

                total: 0

            });

        }

        const item =
            grouped.get(key);

        item.quantity += getQty(row);

        item.total += amount;

    });

    return {

        individual,

        bundle,

        total: individual + bundle,

        rows: Array.from(
            grouped.values()
        )

    };

}

/* ================================================================
   TOP 3
================================================================ */

function calculateTop3() {

    const map = new Map();

    state.submissions.forEach(row => {

        if (
            getStatus(row).toLowerCase() !==
            "approved"
        ) {
            return;
        }

        const smc = getSMC(row);

        if (!smc) {
            return;
        }

        const amount =
            getQty(row) *
            getRate(row);

        if (!map.has(smc)) {

            map.set(smc, {

                staff: getStaff(row),

                smc: smc,

                outlet: getOutlet(row),

                amount: 0,

                count: 0

            });

        }

        const item =
            map.get(smc);

        item.amount += amount;

        item.count += 1;

    });

    return Array.from(
        map.values()
    )
        .sort(
            (a, b) =>
                b.amount -
                a.amount
        )
        .slice(0, 3);

}

/* ================================================================
   SAFE PROOF URL
================================================================ */

function safeFileUrl(value) {

    const str =
        String(value ?? "")
            .trim();

    if (!str) {
        return "";
    }

    /*
     * Prevent javascript: and other dangerous schemes.
     */

    if (
        /^javascript:/i.test(str) ||
        /^data:/i.test(str)
    ) {
        return "";
    }

    /*
     * Full URL
     */

    if (
        /^https?:\/\//i.test(str)
    ) {

        return escapeHtml(str);

    }

    /*
     * Root-relative URL
     */

    if (str.startsWith("/")) {
        return encodeURI(str);
    }

    /*
     * Relative upload path
     */

    return encodeURI(str);

}

/* ================================================================
   CLOSE MODAL WHEN CLICKING OUTSIDE
================================================================ */

document
    .querySelectorAll(".modal-overlay")
    .forEach(overlay => {

        overlay.addEventListener(
            "click",
            function(event) {

                if (
                    event.target ===
                    overlay
                ) {

                    closeModal(
                        overlay.id
                    );

                }

            }
        );

    });


/* ================================================================
   ESC KEY
================================================================ */

document.addEventListener(
    "keydown",
    function(event) {

        if (event.key !== "Escape") {
            return;
        }

        document
            .querySelectorAll(
                ".modal-overlay.show"
            )
            .forEach(modal => {

                closeModal(
                    modal.id
                );

            });

    }
);

document

    )
    .addEventListener(
        "click",
        async function() {

            try {

                await apiRequest(
                    "logout",
                    {}
                );

            } catch (err) {
                /* Still redirect even if backend response fails */
            }

            window.location.href =
                "index.html";

        }
    );


/* ================================================================
   INITIAL MONTH
================================================================ */

(function setInitialMonth() {

    const input =
        document.getElementById(
            "summaryMonth"
        );

    const now = new Date();

    input.value =
        now.getFullYear() +
        "-" +
        String(
            now.getMonth() + 1
        ).padStart(2, "0");

})();

   INITIAL LOAD
================================================================ */

(async function init() {

    try {

        await loadAdminData();

        populateSubmissionFilters();

        renderAll();

    } catch (err) {

        showMessage(
            err.message ||
            "Failed to load admin portal.",
            "error"
        );

    }

})();

</script>
