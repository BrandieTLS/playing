function showAlert(message, type = "success") {

    const container = document.getElementById("alertContainer");

    container.appendChild(alert);

/* =====================================================
   API REQUEST
===================================================== */

async function apiRequest(action, data = null) {

    let url = "admin.php?action=" + encodeURIComponent(action);

    const options = {
        method: "POST",
        headers: {
            "Content-Type": "application/json",
            "Accept": "application/json"
        },
        credentials: "same-origin"
    };

    if (data !== null) {

        options.body = JSON.stringify(data);

    }

    let response;

    try {

        response = await fetch(url, options);

    } catch (error) {

        throw new Error(
            "Unable to connect to the server."
        );

    }

    const text = await response.text();

    let result;

    try {

        result = JSON.parse(text);

    } catch (error) {

        console.error("Server response:", text);

        throw new Error(
            "Server returned invalid JSON."
        );

    }

    if (!result.success) {

        throw new Error(
            result.message || "Request failed."
        );

    }

    return result;

}

/* =====================================================
   ADD USER
===================================================== */

async function saveUser(event) {

    event.preventDefault();


    const staffName =
        document.getElementById(
            "userName"
        ).value.trim();


    const smc =
        document.getElementById(
            "userSMC"
        ).value.trim();


    const outlet =
        document.getElementById(
            "userOutlet"
        ).value.trim();


    const role =
        document.getElementById(
            "userRole"
        ).value;


    if (!staffName ||
        !smc ||
        !outlet ||
        !role) {

        showAlert(
            "Please fill in all fields.",
            "warning"
        );

        return;

    }


    try {

        await apiRequest(
            "add_user",
            {
                staff_name: staffName,
                sales_man_code: smc,
                outlet: outlet,
                role: role
            }
        );


        closeModal(
            "userModal"
        );


        document.getElementById(
            "userForm"
        ).reset();


        showAlert(
            "User added successfully."
        );


        await loadAdminData();

    } catch (error) {

        showAlert(
            error.message,
            "error"
        );

    }

}

/* =====================================================
   MONTHLY SUMMARY
===================================================== */

function renderMonthlySummary() {

    const tbody =
        document.getElementById(
            "monthlyTableBody"
        );


    const selected =
        document.getElementById(
            "monthlyFilter"
        ).value;


    let rows =
        adminData.monthlySummary || [];


    if (selected) {

        rows = rows.filter(function (item) {

            return String(
                item.month || ""
            ) === selected;

        });

    }


    if (!rows.length) {

        tbody.innerHTML = `
            <tr>
                <td colspan="7" class="empty">
                    No monthly summary available.
                </td>
            </tr>
        `;

        return;

    }


    tbody.innerHTML =
        rows.map(function (item) {

            const individual =
                Number(
                    item.individualTotal ||
                    item.individual_total ||
                    0
                );


            const bundle =
                Number(
                    item.bundleTotal ||
                    item.bundle_total ||
                    0
                );


            const total =
                Number(
                    item.total ||
                    item.totalIncentive ||
                    item.total_incentive ||
                    individual + bundle
                );


            return `

                <tr>

                    <td>
                        ${escapeHtml(item.month || "")}
                    </td>

                    <td>
                        ${escapeHtml(item.staff_name || "")}
                    </td>

                    <td>
                        ${escapeHtml(
                            item.sales_man_code ||
                            item.SMC ||
                            ""
                        )}
                    </td>

                    <td>
                        ${escapeHtml(item.outlet || "")}
                    </td>

                    <td>
                        ${money(individual)}
                    </td>

                    <td>
                        ${money(bundle)}
                    </td>

                    <td>
                        <strong>
                            ${money(total)}
                        </strong>
                    </td>

                </tr>

            `;

        }).join("");

}

function populatePromotionDropdown(
    elementId = "editSubmissionPromo",
    selectedId = null
) {

    const select =
        document.getElementById(
            elementId
        );


    if (!select) {

        return;

    }


    select.innerHTML =
        `<option value="">Select Promotion</option>`;


    (adminData.promotions || [])
        .forEach(function (promo) {

            const id =
                promo.promo_id ||
                promo.id;


            const option =
                document.createElement("option");


            option.value = id;


            option.textContent =
                promo.promo +
                " - " +
                money(
                    promo.Rate ||
                    promo.rate ||
                    0
                );


            if (
                selectedId !== null &&
                Number(id) === Number(selectedId)
            ) {

                option.selected = true;

            }


            select.appendChild(option);

        });

}


/* =====================================================
   MONTH FILTER
===================================================== */

function populateMonthlyFilter() {

    const select =
        document.getElementById(
            "monthlyFilter"
        );


    const current =
        select.value;


    const months =
        new Set();


    (adminData.monthlySummary || [])
        .forEach(function (item) {

            if (item.month) {

                months.add(
                    item.month
                );

            }

        });


    select.innerHTML =
        `<option value="">All Months</option>`;


    Array.from(months)
        .sort()
        .reverse()
        .forEach(function (month) {

            const option =
                document.createElement("option");

            option.value = month;

            option.textContent = month;

            select.appendChild(option);

        });


    if (months.has(current)) {

        select.value = current;

    }

}

/* =====================================================
   LOGOUT
===================================================== */

async function logout() {

    if (!confirm(
        "Are you sure you want to logout?"
    )) {

        return;

    }


    try {

        const result =
            await apiRequest(
                "logout"
            );


        if (result.redirect) {

            window.location.href =
                result.redirect;

            return;

        }


        window.location.href =
            "index.html";

    } catch (error) {

        /*
         * Even if logout response is not JSON,
         * clear the local page by returning to login.
         */

        window.location.href =
            "index.html";

    }

}

/* =====================================================
   STATUS BADGES
===================================================== */

function statusBadge(status) {

    const value =
        String(status || "")
        .toLowerCase();


    let cls =
        "status-pending";


    if (value === "approved") {

        cls = "status-approved";

    } else if (value === "rejected") {

        cls = "status-rejected";

    }


    return `
        <span class="status ${cls}">
            ${escapeHtml(status || "")}
        </span>
    `;

}


function promotionStatusBadge(status) {

    const value =
        String(status || "")
        .toLowerCase();


    let cls =
        "status-active";


    if (value === "upcoming") {

        cls = "status-upcoming";

    } else if (value === "expired") {

        cls = "status-expired";

    }


    return `
        <span class="status ${cls}">
            ${escapeHtml(status || "")}
        </span>
    `;

}


/* =====================================================
   DETAIL
===================================================== */

function detail(label, value) {

    return `

        <div class="form-group">

            <label>
                ${escapeHtml(label)}
            </label>

            <div>
                ${escapeHtml(
                    value === null ||
                    value === undefined ||
                    value === ""
                    ? "-"
                    : value
                )}
            </div>

        </div>

    `;

}


/* =====================================================
   MONEY
===================================================== */

function money(value) {

    const number =
        Number(value) || 0;


    return "$" +
        number.toLocaleString(
            "en-SG",
            {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            }
        );

}


/* =====================================================
   ESCAPE HTML
===================================================== */

function escapeHtml(value) {

    return String(
        value === null ||
        value === undefined
            ? ""
            : value
    )
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");

}


function escapeAttribute(value) {

    return escapeHtml(value);

}
